\section{Data Synchronization Endpoints}
\label{sec:endpoints}
The Synchronization API provides tools for managing multi-region sync operations across your Vertex datastores.

\begin{apiendpoint}{GET}{/v2/sync/jobs}{Data Sync}
Retrieve a paginated list of all active synchronization jobs associated with your account.
\end{apiendpoint}

\subsubsection{Query Parameters}
\begin{itemize}
  \item \code{status} \textit{(string, optional)}: Filter by status (\code{pending}, \code{running}, \code{completed}, \code{failed}).
  \item \code{limit} \textit{(integer, optional)}: Number of records to return. Default is 20. Max is 100.
  \item \code{cursor} \textit{(string, optional)}: Pagination cursor for the next page of results.
\end{itemize}

\noindent
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[codeboxstyle, title={Request Headers}]
\begin{lstlisting}[language=json,numbers=none]

Host: api.vertex.meridian.net
Authorization: Bearer eyJhbGc...
\end{lstlisting}
  \end{tcolorbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[codeboxstyle, title={Response Payload}]

\begin{apiendpoint}{POST}{/v2/sync/jobs}{Data Sync}
Initialize a new real-time synchronization pipeline between source and target clusters.
\end{apiendpoint}

\subsubsection{Request Body Parameters}
\begin{itemize}
  \item \code{source\_id} \textit{(string, required)}: Fictional cluster ID containing primary tables.
  \item \code{target\_id} \textit{(string, required)}: Target cluster ID to receive replicas.
  \item \code{sync\_mode} \textit{(string, required)}: The replication level (\code{snapshot}, \code{realtime}, or \code{differential}).
  \item \code{filters} \textit{(array, optional)}: Specific databases/tables to replicate.
\end{itemize}

\noindent
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[codeboxstyle, title={Example Request (JSON)}]
\begin{lstlisting}[language=json]
{
  "source_id": "clus_nexa_9910",
  "target_id": "clus_apex_4201",
  "sync_mode": "realtime",
  "filters": ["sales_data", "customers"]
}
\end{lstlisting}
  \end{tcolorbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}

\begin{apiendpoint}{DELETE}{/v2/sync/jobs/\{id\}}{Data Sync}
Gracefully terminate an active synchronization pipeline. This will stop replication but will not delete synced tables.
\end{apiendpoint}

\subsubsection{Path Parameters}
\begin{itemize}
  \item \code{id} \textit{(string, required)}: The unique job identifier, prefixed with \code{job\_vtx\_}.
\end{itemize}

\noindent
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[codeboxstyle, title={Example Request}]
\begin{lstlisting}[language=json,numbers=none]
DELETE /v2/sync/jobs/job_vtx_00192b8 HTTP/1.1
Host: api.vertex.meridian.net
Authorization: Bearer eyJhbGc...
\end{lstlisting}
  \end{tcolorbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[codeboxstyle, title={Example Response (200 OK)}]

  "terminated_at": "2026-08-03T23:27:00Z",
