% Badge drawing
\newcommand{\badge}[2]{%
  \tikz[baseline=(char.base)]{
    \node[
      fill=#1,
      text=white,
      rounded corners=2px,
      inner sep=4pt,
      font=\sffamily\bfseries\scriptsize
    ] (char) {#2};
  }%
}
\newcommand{\getbadge}{\badge{getcolor}{GET}}
\newcommand{\postbadge}{\badge{postcolor}{POST}}
\newcommand{\putbadge}{\badge{putcolor}{PUT}}
\newcommand{\deletebadge}{\badge{deletecolor}{DELETE}}

% Monospace inline code
\newcommand{\code}[1]{\texttt{\color{secondary}#1}}

\newcommand{\methodbadge}[1]{%
  \IfEqCase{#1}{%
    {GET}{\getbadge}%
    {POST}{\postbadge}%
    {PUT}{\putbadge}%
    {DELETE}{\deletebadge}%
  }[\badge{secondary}{#1}]%
}
