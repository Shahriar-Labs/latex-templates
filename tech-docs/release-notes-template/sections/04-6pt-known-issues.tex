\section{\faInfoCircle\hspace{6pt}Known Issues}

\begin{infobox}
  \small
  \begin{itemize}[leftmargin=*,nosep,itemsep=4pt,topsep=2pt]
    \item \textbf{KNW-344} --- The Custom Metric Builder may exhibit layout
      overflow on viewports narrower than 1024\,px. A responsive update is
      scheduled for v3.4.3.
    \item \textbf{KNW-351} --- SCIM provisioning does not yet support group
      push for nested Azure AD security groups. Flat groups function correctly.
    \item \textbf{KNW-358} --- Parallel execution pools default to a maximum
      concurrency of 50; exceeding this requires a support ticket to raise
      the tenant limit.
  \end{itemize}
\end{infobox}
