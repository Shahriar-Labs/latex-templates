\section{Folders}

\subsection{Create \& list contents}
\begin{routebox}
\httpbadge{POST}\hspace{6pt}\texttt{\textbf{/v2/folders}}\hspace{10pt}
\httpbadge{GET}\hspace{6pt}\texttt{\textbf{/v2/folders/\{id\}/children}}\\[3pt]
{\small Create a folder, optionally nested under a parent, or list its immediate child
files and subfolders (accepts the same \texttt{limit}/\texttt{cursor} parameters as
Section~4.1, plus \texttt{type=file\textbar folder} to filter).}
\end{routebox}

\begin{minipage}[t]{0.48\textwidth}
\begin{jsonpanel}{REQUEST BODY}{reqblue}
\begin{lstlisting}
{
  "name": "Client Deliverables",
  "parent_id": null
}
\end{lstlisting}
\end{jsonpanel}
\end{minipage}\hfill

  "parent_id": null
}
\end{lstlisting}
\end{jsonpanel}
\end{minipage}
