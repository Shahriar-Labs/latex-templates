\newcommand{\httpbadge}[1]{%
  \ifstrequal{#1}{GET}{\textcolor{white}{\colorbox{accent}{\textbf{\ttfamily\footnotesize\,GET\,}}}}{}%
  \ifstrequal{#1}{POST}{\textcolor{white}{\colorbox{postgreen}{\textbf{\ttfamily\footnotesize\,POST\,}}}}{}%
  \ifstrequal{#1}{PUT}{\textcolor{white}{\colorbox{putorange}{\textbf{\ttfamily\footnotesize\,PUT\,}}}}{}%
  \ifstrequal{#1}{PATCH}{\textcolor{white}{\colorbox{patchpurple}{\textbf{\ttfamily\footnotesize\,PATCH\,}}}}{}%
  \ifstrequal{#1}{DELETE}{\textcolor{white}{\colorbox{deletered}{\textbf{\ttfamily\footnotesize\,DELETE\,}}}}{}%
}
