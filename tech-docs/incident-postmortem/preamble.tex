% ── Helper macros ───────────────────────────────────────────
\newcommand{\pill}[2]{%
  \tikz[baseline=(n.base)]\node[rounded corners=2pt,
    fill=#1!15!white, draw=#1, inner xsep=3pt, inner ysep=2pt,
    font=\bfseries\tiny, text=#1!70!black](n){#2};%
}

\newcommand{\badge}[2]{%
  \tikz[baseline=(n.base)]\node[rounded corners=2pt,
    fill=#1, inner xsep=5pt, inner ysep=2pt,
    font=\bfseries\tiny, text=white](n){#2};%
}

\newcommand{\kv}[2]{%
  \noindent
  \begin{tabularx}{\linewidth}{@{}>{\color{accent}\bfseries\small}l@{\hspace{8mm}}>{\raggedright\arraybackslash\small}X@{}}
    #1 & #2
  \end{tabularx}%
}

\newcommand{\sectionrule}{%
  \vspace{2pt}\textcolor{rule}{\rule{\linewidth}{0.4pt}}\vspace{4pt}%
}

% ── Action item row macro ───────────────────────────────────
\newcommand{\actionrow}[5]{%
  % #1=ID #2=action #3=owner #4=deadline #5=status-colour+label
  \texttt{\small #1} &
  \small #2 &
  \small\textbf{#3} &
  \small\texttt{#4} &
  \badge{#5}{\phantom{x}}\hspace{-8pt}\badge{#5}{}%
}
