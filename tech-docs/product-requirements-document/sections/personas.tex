\section{Target Users}

\renewcommand{\arraystretch}{1.3}
\begin{tabularx}{\linewidth}{@{}p{2.9cm}X@{}}
\toprule
\rowcolor{prdpale}
\dcellaccent{Persona} & Profile \\
\midrule
\dcellaccent{Revenue Ops Lead}\newline\footnotesize\normalfont(primary) &
Owns 5--15 pipeline and conversion metrics across two or three dashboards. Checks Apex
once or twice a day at fixed times; wants to be told, not to go looking. Success is never
being surprised in a leadership review. \\
\addlinespace
\dcellaccent{Data/Platform Engineer}\newline\footnotesize\normalfont(primary) &
Responsible for the data pipelines feeding Apex. Cares about detecting \emph{feed}
failures (a metric flatlining or going to zero) as much as business anomalies. Wants
alerts routed to the team Slack channel, not personal email. \\
\addlinespace
\dcellaccent{Customer Success Manager}\newline\footnotesize\normalfont(secondary) &
Tracks account health metrics for a book of 20--40 accounts. Needs early warning with
enough context to act (which account, which metric, how unusual) without opening Apex
first. \\
\addlinespace
\dcellaccent{Workspace Admin}\newline\footnotesize\normalfont(secondary) &
Manages Apex billing and permissions. Cares about alert volume staying sane across the
org and wants visibility into who is subscribed to what, but is not the primary
subscriber. \\
\bottomrule
\end{tabularx}
