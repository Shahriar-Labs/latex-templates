\section{Glossary}

\begin{center}
\small
\begin{tabularx}{\textwidth}{@{}l X@{}}
\toprule
\rowcolor{navydark}
\textcolor{white}{\textbf{Term}} & \textcolor{white}{\textbf{Definition}} \\
\midrule
\term{Approval Chain} & The ordered sequence of Approvers a report must pass through before it
is eligible for reimbursement. \\
\rowcolor{lightgray}
\term{Batch} & A group of fully approved expense reports assembled for a single reimbursement
run and payroll export. \\
\term{Cost Center} & An organizational unit code used to route reports to the correct approval
chain and to attribute spend in reporting. \\
\rowcolor{lightgray}
\term{Escalation} & Automatic forwarding of a pending report to the next approver after 5
business days of inactivity (FR-3.4). \\
\term{FR} & Functional Requirement --- a numbered, testable statement of required system
behavior. \\
\rowcolor{lightgray}
\term{NFR} & Non-Functional Requirement --- a quality attribute constraint (performance,
security, availability, etc.). \\
\term{RBAC} & Role-Based Access Control --- authorization model in which permissions are
granted to roles rather than individual users. \\
\rowcolor{lightgray}
\term{Reconciliation} & The process of matching corporate-card-fed transactions to submitted
expense line items. \\
\term{SLA} & Service Level Agreement --- a contracted commitment on a measurable service
quality, such as uptime. \\
\rowcolor{lightgray}
\term{SSO} & Single Sign-On --- authentication scheme allowing one login to grant access across
multiple systems, here via Okta SAML. \\
\bottomrule
\end{tabularx}
\end{center}
