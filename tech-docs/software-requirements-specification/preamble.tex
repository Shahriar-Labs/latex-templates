\newcommand{\drawactor}[3]{%
  \draw[thick,steelblue] (#1,#2) circle (0.16);
  \draw[thick,steelblue] (#1,{#2-0.16}) -- (#1,{#2-0.62});
  \draw[thick,steelblue] ({#1-0.26},{#2-0.36}) -- ({#1+0.26},{#2-0.36});
  \draw[thick,steelblue] (#1,{#2-0.62}) -- ({#1-0.22},{#2-0.95});
  \draw[thick,steelblue] (#1,{#2-0.62}) -- ({#1+0.22},{#2-0.95});
  \node[below=0.14cm,align=center,font=\footnotesize\bfseries\color{navydark}] at (#1,{#2-0.95}) {#3};
}
