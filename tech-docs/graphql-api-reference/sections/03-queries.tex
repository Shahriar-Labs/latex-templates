\section*{Queries}

\begin{schemabox}[title=Root Query Fields]
type Query \{\\
\quad project(slug: String!): Project\\
\quad projects(first: Int, after: String, teamId: ID, query: String): ProjectConnection!\\
\quad task(id: ID!): Task\\
\quad tasks(first: Int, after: String, projectId: ID, status: TaskStatus, priority: TaskPriority, assigneeId: ID, query: String): TaskConnection!\\
\quad team(id: ID!): Team\\
\quad teams(first: Int, after: String, query: String): TeamConnection!\\
\quad viewer: User!\\
\}
\end{schemabox}

\subsection*{Fetch a Project by Slug}

Retrieve a project with its team name and the first five in-progress tasks.

\vspace{2pt}
\begin{querybox}[title=Query: Get Project]
query GetProject(\$slug: String!) \{\\
\quad project(slug: \$slug) \{\\
\qquad id\\
\qquad name\\
\qquad description\\
\qquad team \{ name \}\\
\qquad tasks(first: 5, status: IN\_PROGRESS) \{\\
\qquad\quad edges \{ node \{ id, title, priority, assignee \{ username \}, dueDate \} \}\\
\qquad\quad pageInfo \{ endCursor, hasNextPage \}\\
\qquad\quad totalCount\\
\qquad \}\\
\qquad createdAt\\
\quad \}\\
\}\\[2pt]
Variables: \{ "slug": "meridian-migration" \}
\end{querybox}
