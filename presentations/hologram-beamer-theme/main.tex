\begin{frame}{Perceptual stability governs fidelity}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.50\textwidth}
      \holoeyebrow{Control objective}
      \begin{equation*}
        \mathcal{J}
        = \alpha\,E_{\mathrm{pose}}
        + \beta\,E_{\mathrm{time}}
        + \gamma\,E_{\mathrm{depth}}
        + \lambda\,C_{\mathrm{compute}}
      \end{equation*}
      \vspace{3mm}
      \begin{itemize}
        \item $E_{\mathrm{pose}}$: viewpoint alignment error
        \item $E_{\mathrm{time}}$: motion-to-photon delay
        \item $E_{\mathrm{depth}}$: reconstruction inconsistency
        \item $C_{\mathrm{compute}}$: render and network cost
      \end{itemize}
    \end{column}
    \begin{column}{.44\textwidth}
      \begin{block}{Operating principle}
        Preserve stable eye contact and object registration before increasing visual detail.
      \end{block}
      \vspace{3mm}
      \begin{exampleblock}{Adaptive response}
        When bandwidth falls, PRISM reduces peripheral fidelity while protecting faces, hands, and the active object.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\holosection{02 / PROOF}{The prototype protects\\what the task needs most}
