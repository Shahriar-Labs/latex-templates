\section{Experimental Results}
\subsection{Training Throughput \& Scaling}

\begin{frame}{Performance Benchmarking}{Throughput Scaling under Model Parallelism}
  We evaluated the scaling behavior of Nexa-Base-v3 using different tensor and pipeline parallel configurations~\cite{jenkins2025tensor}.
  
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Key Takeaways:}
      \begin{itemize}
        \item \textbf{Tensor Parallelism (TP):} High intra-node performance, but scaling is constrained by NVLink limits.
        \item \textbf{Pipeline Parallelism (PP):} Effective across nodes; communication overhead is hidden.
        \item \textbf{Peak Speed:} Obtained 240 TFLOPS/GPU using TP=2, PP=4.
        \item \textbf{Efficiency:} Sustained 78\% scaling efficiency up to 128 GPUs.
      \end{itemize}
    \end{column}
