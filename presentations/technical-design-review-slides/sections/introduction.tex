\section{Introduction}

\begin{frame}{Project Overview: Nexa Cloud Gateway}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Context}
      \begin{itemize}
        \item Apex Systems is upgrading its edge infrastructure to support high-throughput cloud API routing.
        \item The Nexa Cloud Gateway acts as the unified entry point for all external client traffic.
        \item Existing reverse proxies suffer from latency spikes during peak load events.
      \end{itemize}
      \vspace{0.3cm}
      \textbf{Core Mandate}
      \begin{itemize}
        \item Consolidate authentication, routing, and rate-limiting into a single high-performance edge layer.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \begin{designcard}{Key Design Targets}
        \begin{itemize}
          \item[\faClock] \textbf{Low Latency:} $<5$\,ms p99 overhead.
          \item[\faLock] \textbf{Zero Trust:} Inline token verification.
          \item[\faChartLine] \textbf{Scale:} $100\text{k}+$ requests/sec.
          \item[\faServer] \textbf{Resilience:} Multi-region fallback.
        \end{itemize}
      \end{designcard}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Key Objectives \& Scope}
  \begin{itemize}
    \item[\faCheckCircle] \textbf{Unified Gateway Entry:} Replace the legacy regional proxies with a centralized cluster of Nexa nodes.
    \item[\faCheckCircle] \textbf{Dynamic Routing:} Route requests based on header metadata, query parameters, and geographical proximity to Vertex databases.
    \item[\faCheckCircle] \textbf{Distributed Rate Limiting:} Prevent service degradation using a global token-bucket algorithm synchronized via Nimbus caches.
    \item[\faCheckCircle] \textbf{Observability:} Export real-time metrics, logs, and distributed traces to the centralized monitoring plane.
  \end{itemize}
  \vspace{0.4cm}
  \begin{designcard}[colback=secondary!5,colframe=secondary!30]{Scope Exclusions}
    This review focuses on the API gateway runtime. Long-term log archiving databases and user billing engines are excluded from this review.
  \end{designcard}
\end{frame}
