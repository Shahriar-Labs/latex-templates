% Slide 4: Bullet Builds (sequential list)
\begin{frame}{Orchestration Challenges in Edge AI}
  \textbf{Goal:} Train a unified model across decentralized data resources without raw upload.
  \vskip3ex
  \begin{itemize}
    \item<1-> \textbf{Communication Latency:} High-dimensional updates saturate shared channels.
    \item<2-> \textbf{Resource Asymmetry:} Vertex nodes span low-power IoT to compute clusters.
    \item<3-> \textbf{Statistical Variance:} Non-IID local data distributions delay convergence.
  \end{itemize}
\end{frame}

% Slide 5: Two-Column Slide
\begin{frame}{System Topology Paradigm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Traditional Centralized Training}
      \vskip1.5ex
      \begin{itemize}
        \item Direct raw upload to centralized database.
        \item Saturated uplink networks during peak times.
        \item Single-point-of-failure under breach.
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \textbf{Synapse Federated Protocol}
      \vskip1.5ex
      \begin{itemize}
        \item Local optimization on local device storage.
        \item Only parameterized weights are transmitted.
        \item Data remains isolated inside the local edge.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 7: Mathematical Formulation Slide
\begin{frame}{Optimization Objective}
  We formulate the distributed empirical risk minimization as:
  \vskip1ex
  \begin{equation*}
    \min_{\theta \in \mathbb{R}^d} F(\theta) = \sum_{k=1}^K p_k F_k(\theta)
  \end{equation*}
  \vskip1ex
  where the local loss function $F_k(\theta)$ for client $k$ is computed over dataset $\mathcal{D}_k$:
  \vskip1ex
  \begin{equation*}
    F_k(\theta) = \frac{1}{|\mathcal{D}_k|} \sum_{i \in \mathcal{D}_k} \ell(f(x_i; \theta), y_i)
  \end{equation*}
  \vskip1.5ex
  Here, client weights are proportional to dataset sizes: $p_k = |\mathcal{D}_k| / \sum_j |\mathcal{D}_j|$.
\end{frame}

% Slide 10: Block / Alertblock / Exampleblock Trio
\begin{frame}{Synapse Security Architecture}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Differential Privacy}
        Strict bounding of local update sensitivity to protect user membership information.
      \end{block}
      \vskip2.5ex
      \begin{exampleblock}{Local Perturbation}
        Additive Gaussian noise is parameterized per-epoch based on privacy budget constraints.
      \end{exampleblock}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Aggregator Threat Model}
        The coordinator is considered curious but honest. Synapse deploys cryptographic secret sharing to prevent weight snooping.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
