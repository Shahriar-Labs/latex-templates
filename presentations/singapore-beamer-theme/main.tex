\section{Motivation \& Challenges}

% Slide 3: Challenges (Bullet builds)
\begin{frame}{Challenges in Large-Scale LLM Training}
    \begin{itemize}[<+->]
        \item \textbf{High Communication Overhead}: Dense parameters result in all-reduce bottlenecks on large clusters.
        \item \textbf{Single-Point Failure}: Centralized parameter server topologies pose massive reliability risks.
        \item \textbf{Resource Underutilization}: Asymmetric compute nodes lead to idle wait states during barrier synchronization.
        \item \textbf{Scalability Limits}: Traditional SGD efficiency drops drastically beyond a few dozen nodes.
    \end{itemize}
\end{frame}

% Slide 4: Centralized vs. Decentralized (Two Columns)
\begin{frame}{Centralized vs. Decentralized Optimization}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Centralized Paradigm}
                \begin{itemize}
                    \item Rely on dedicated parameter servers.
                    \item High bandwidth required at the root.
                    \item Synchronous gradient aggregation.
                    \item Vulnerable to straggler nodes.
                \end{itemize}
            \end{block}
        \end{column}
        \hfill
        \begin{column}{0.48\textwidth}
            \begin{block}{Decentralized (Vertex-SGD)}
                \begin{itemize}
                    \item Local consensus updates via neighbors.
                    \item Uniform network traffic distribution.
                    \item Asynchronous execution models.
                    \item High robustness to node failures.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{The Vertex-SGD Algorithm}

% Slide 5: The Optimization Loop (Equations)
\begin{frame}{The Vertex-SGD Optimization Protocol}
    Let $x_i^{(t)}$ represent the local parameter weights of node $i$ at step $t$.
    
    The local update consists of a gradient step followed by neighbor consensus:
    
    \begin{equation}
        x_i^{(t+1/2)} = x_i^{(t)} - \eta g_i\left(x_i^{(t)}\right)
    \end{equation}
    
    \begin{equation}
        x_i^{(t+1)} = \sum_{j \in \mathcal{N}_i} W_{ij} x_j^{(t+1/2)}
    \end{equation}
    
    \vskip 0.5cm
    \begin{itemize}
        \item $\eta > 0$ is the local learning rate.
        \item $g_i$ is the stochastic gradient computed on node $i$'s local batch.
        \item $W \in \mathbb{R}^{n \times n}$ is a doubly stochastic gossip matrix.
        \item $\mathcal{N}_i$ is the set of physical neighbors of node $i$.
    \end{itemize}
\end{frame}

% Slide 9: Block, Alertblock, Exampleblock Trio
\begin{frame}{Key Observations \& Robustness}
    \begin{block}{Theoretical Convergence}
        Local parameters approach the true global optimum sublinearly for non-convex loss surfaces.
    \end{block}
    
    \begin{alertblock}{Straggler Vulnerability}
        Strict consensus bounds might delay updates if individual node computation speeds degrade.
    \end{alertblock}
    
    \begin{exampleblock}{Empirical Speedup}
        Over a wide range of tasks, Vertex-SGD yields a $2.1\times$ to $4.2\times$ reduction in end-to-end wall-clock training times.
    \end{exampleblock}
\end{frame}

% Slide 10: Future Directions
\begin{frame}{Future Work \& Outlook}
    We plan to extend Vertex-SGD to address the following challenges:
    \vskip 0.5cm
    \begin{itemize}
        \item \textbf{Asynchronous Updates}: Relaxing consensus synchrony to allow fully lock-free training.
        \item \textbf{Adaptive Topologies}: Modifying matrix weights dynamically based on live throughput metrics.
        \item \textbf{Quantized Gossip}: Lowering communication bandwidth via lossy gradient quantization.
    \end{itemize}
\end{frame}
