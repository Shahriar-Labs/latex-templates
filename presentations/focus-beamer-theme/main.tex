\section{1. Background \& Motivation}

% Slide 3: Motivation
\begin{frame}{Motivation: The Rise of Edge Computing}
  Edge computing shifts computation closer to data sources.
  \begin{itemize}[<+->]
    \item \textbf{Reduced Latency:} Faster response times for critical applications.
    \item \textbf{Bandwidth Conservation:} Minimizes heavy cloud data transfers.
    \item \textbf{Privacy Preservation:} Keeps sensitive data local to the device.
    \item \textbf{Resilience:} Decentralized architecture prevents single points of failure.
  \end{itemize}
\end{frame}

% Slide 4: Challenges (Two Columns)
\begin{frame}{Challenges in Current Architectures}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Resource Heterogeneity}
        Edge devices vary widely in performance:
        \begin{itemize}
          \item Low-power microcontrollers (Apex Nano)
          \item Smart gateways (Vertex Pro)
          \item High-end local servers (Synapse Max)
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{block}{Network Volatility}
        Connections at the edge are unstable:
        \begin{itemize}
          \item Intermittent connectivity
          \item Asymmetric bandwidth limits
          \item High packet loss rates
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 6: Resource Allocation Protocol
\begin{frame}{Resource Allocation Protocol}
  Our protocol maximizes throughput across heterogeneous nodes:
  \begin{enumerate}
    \item \textbf{Discovery:} Edge nodes broadcast resource availability (RAM, CPU).
    \item \textbf{Estimation:} The orchestrator models latency based on network telemetry.
    \item \textbf{Partitioning:} Vertex-partitioning divides the workload dynamically.
    \item \textbf{Execution:} Jobs are dispatched with optimistic scheduling.
  \end{enumerate}
\end{frame}
