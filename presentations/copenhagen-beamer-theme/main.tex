\section{Introduction \& Motivation}

% Slide 3: Background: Centralized vs. Decentralized Compute
\begin{frame}{Background: Centralized vs. Decentralized Compute}
  \begin{itemize}
    \item<1-> \textbf{Centralized Systems:} Heavy reliance on hyper-scale data centers (e.g., cloud platforms).
    \begin{itemize}
      \item High throughput and low latency within the cluster.
      \item Single point of failure and high operational costs.
    \end{itemize}
    \item<2-> \textbf{Decentralized Architecture:} Leveraging edge nodes and idle resource pools.
    \begin{itemize}
      \item Lower network egress and geo-distributed latency benefits.
      \item High heterogeneity in hardware capabilities and network bandwidth.
    \end{itemize}
    \item<3-> \textbf{The Synapse Vision:} Orchestrate model training across unstable, low-power edge nodes without sacrificing convergence speed.
  \end{itemize}
\end{frame}

% Slide 4: Challenges in Distributed Edge Networks
\begin{frame}{Challenges in Distributed Edge Networks}
  Training neural networks over consumer-grade nodes introduces severe bottleneck vectors:
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Network Fluctuation:} Slower uplinks lead to synchronization lag.
    \pause
    \item \textbf{Hardware Heterogeneity:} Devices ranging from low-power edge accelerators to high-end workstations.
    \pause
    \item \textbf{Security \& Trust:} Nodes can join/leave arbitrarily, raising data privacy issues.
    \pause
    \item \textbf{Straggler Nodes:} Slow nodes delay gradient aggregation under traditional synchronous protocols.
  \end{itemize}
\end{frame}

% Slide 6: Mathematical Formulation
\begin{frame}{Mathematical Formulation}
  In Synapse ML, we formulate the global objective function as:
  \begin{equation}
    \min_{w \in \mathbb{R}^d} F(w) = \sum_{k=1}^K p_k F_k(w)
  \end{equation}
  where $K$ is the total active edge nodes, and $p_k = n_k / N$ (with $N = \sum_{k=1}^K n_k$) represents the node weighting factor.
  Under our asynchronous protocol, the weight update at step $t+1$ is defined as:
  \begin{equation}
    w_{t+1} = w_t - \eta \nabla F_k(w_{t - \tau_k(t)})
  \end{equation}
  where $\tau_k(t)$ is the gradient staleness factor.
\end{frame}

% Slide 7: Algorithmic Framework
\begin{frame}{Asynchronous Aggregation Algorithm}
  \textbf{Synapse Server-Side Update Protocol}
  \hrule\vspace{0.2cm}
  \begin{enumerate}
    \item \textbf{Initialize} global model parameters $w_0$ and node contribution vector $c_0$.
    \item \textbf{On receiving} local update $g_k$ from node $k$ at time $t$:
    \begin{enumerate}
      \item Calculate the dynamic staleness parameter: $\tau = t - t_{sent}$.
      \item Compute learning rate scaling factor: $\alpha(\tau) = \frac{\alpha_0}{1 + \gamma \tau}$.
      \item Update global parameters:
      \begin{equation*}
        w_{t+1} \leftarrow w_t - \alpha(\tau) \cdot g_k
      \end{equation*}
    \end{enumerate}
    \item \textbf{Broadcast} the updated global weights $w_{t+1}$ back to node $k$.
    \item \textbf{Repeat} until target validation accuracy $\epsilon$ is reached.
  \end{enumerate}
  \hrule
\end{frame}
