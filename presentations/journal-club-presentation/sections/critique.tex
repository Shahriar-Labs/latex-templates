\begin{frame}{Strengths \& Limitations Critique}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{strengthsbox}{Strengths}
        \begin{itemize}
          \item \textbf{Realistic Formulation:} The inclusion of network propagation latency in the state representation is a major step forward for decentralized systems.
          \item \textbf{Excellent Scalability:} Gossip-based consensus keeps the communication overhead to $O(\log N)$ where $N$ is the number of switches.
          \item \textbf{Open Source Commitment:} The authors provided fully reproducible simulator code and scripts.
        \end{itemize}
      \end{strengthsbox}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{critiquebox}{Limitations \& Concerns}
        \begin{itemize}
          \item \textbf{Cold-Start Vulnerability:} PPO algorithm exhibits severe latency spikes during the initial 50 epochs of training.
          \item \textbf{Sim-to-Real Gap:} The network simulator uses simplified Poisson arrivals; real-world bursty internet traffic may behave differently.
          \item \textbf{Hardware Constraints:} Deep neural net inference on edge switches is slow without dedicated TPU accelerators.
        \end{itemize}
      \end{critiquebox}
    \end{column}
  \end{columns}
\end{frame}
