% Citation Banner Command
\newcommand{\citationbanner}[4]{%
  \begin{tcolorbox}[
    colback=primary,
    coltext=white,
    boxrule=0pt,
    arc=4pt,
    left=8pt, right=8pt, top=6pt, bottom=6pt,
    width=\textwidth
  ]
    \small \textbf{#1} \hfill \textit{#2} (\textbf{#3})\\
    \footnotesize DOI: #4
  \end{tcolorbox}%
}
