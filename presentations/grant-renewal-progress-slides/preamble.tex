% #1 = percent complete (0-100), #2 = bar color
\newcommand{\progressbar}[2]{%
  \begin{tikzpicture}[baseline=-0.5ex]
    \fill[mist, rounded corners=1.5pt] (0,0) rectangle (2.6,0.34);
    \fill[#2, rounded corners=1.5pt] (0,0) rectangle ({2.6*#1/100},0.34);
    \draw[navy, line width=0.3pt, rounded corners=1.5pt] (0,0) rectangle (2.6,0.34);
    \node[anchor=west, font=\scriptsize\bfseries, text=navy] at (2.75,0.17) {#1\%};
  \end{tikzpicture}%
}

% #1 = label text, #2 = fill color, for status chips
\newcommand{\statuschip}[2]{%
  \tikz[baseline=-0.5ex]{\node[fill=#2, text=white, font=\scriptsize\bfseries,
    rounded corners=2pt, inner sep=3pt] {#1};}%
}
