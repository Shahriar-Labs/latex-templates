      \textbf{Step-by-Step Task List}
      \begin{enumerate}
        \item \textbf{Access GitLab:} Clone the baseline repository: \\
          {\scriptsize\texttt{git clone git@git.nimbus.io:vt/flow.git}}
        \item \textbf{Start Slurm Session:} Request a GPU allocation: \\
          {\scriptsize\texttt{srun -p gpu --gpus=1 --pty bash}}
        \item \textbf{Run Replication:} Run the onboarding test script: \\
          {\scriptsize\texttt{python run\_onboard.py --config cfg.json}}
        \item \textbf{Publish Results:} Paste performance metrics on the Nimbus Wiki.
      \end{enumerate}
    \end{column}
