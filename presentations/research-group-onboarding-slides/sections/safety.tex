\section{Lab Safety \& Protocols}

\begin{frame}{Lab Safety \& Protocols}{Physical Lab Guidelines}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Physical Access \& Hardware Safety}
      \begin{itemize}
        \item \textbf{ESD Protection:} Always wear Electrostatic Discharge (ESD) wrist straps when handling custom boards and FPGAs.
        \item \textbf{Cleanroom Rules:} No food or open drinks are allowed in the server closet or custom testing benches.
        \item \textbf{Emergency Shutdown:} Locate the red power-off buttons on the side walls in case of hardware malfunction.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.46\textwidth}
      \begin{alertbox}
        \textbf{\faExclamationTriangle\ Warning: Laser Safety} \\
        Our communication testbed uses Class 3B lasers. Never operate the testbed without wearing laser safety goggles (located in Drawer 4).
      \end{alertbox}
      
      \vspace{0.3cm}
      \begin{highlightbox}
        \textbf{Incident Reporting} \\
        Report equipment failures or safety hazards in the \texttt{\#lab-hardware} Slack channel.
      \end{highlightbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Data Safety \& Ethics}{Protecting Code \& Datasets}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Data Backup Policies}
      \begin{itemize}
        \item All raw data must be mirrored to the Nimbus storage cluster (which undergoes nightly backups).
        \item Do not store critical experimental data solely on local lab computer hard drives.
      \end{itemize}
      
      \vspace{0.3cm}
      \textbf{Responsible Code Sharing}
      \begin{itemize}
        \item Always keep private repositories private until publication.
        \item Never commit credentials, server IPs, or API tokens to Git.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.46\textwidth}
      \begin{accentbox}
        \textbf{Ethical Publication Rules}
        \begin{itemize}
          \item Follow IEEE guidelines for authorship and credit.
          \item Ensure all open-source models published on Vertex Registry are sanitized.
        \end{itemize}
      \end{accentbox}
    \end{column}
  \end{columns}
\end{frame}
