\section{Expectations \& Culture}

\begin{frame}{Expectations \& Culture}{Weekly Research Rhythms}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Weekly Meetings}
      \begin{itemize}
        \item \textbf{Lab Meeting:} Mondays 10:00 AM. Attendance is mandatory. We discuss high-level project statuses and administrative updates.
        \item \textbf{Reading Group:} Wednesdays 4:00 PM. We present and critique recent SOTA papers on edge machine learning.
      \end{itemize}
      
      \vspace{0.3cm}
      \textbf{Individual Synchronizations}
      \begin{itemize}
        \item Weekly 1-on-1 with your direct mentor/advisor. Schedule via Nimbus Wiki calendar.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.46\textwidth}
      \begin{highlightbox}
        \textbf{Working Hours} \\
        We support flexible schedules, but core collaboration hours are between \textbf{10:00 AM and 4:00 PM} (when team members are expected to be responsive on Slack and email).
      \end{highlightbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Onboarding Milestones}{First Year Timeline}
  \begin{columns}[T]

      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
