\section{Who We Are}

\begin{frame}{Welcome to the Meridian Lab}{Mission \& Core Values}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Our Core Mission}
      \begin{itemize}
        \item We design high-throughput machine learning architectures for distributed, heterogeneous edge environments.
        \item We focus on bridging the gap between algorithmic complexity and actual hardware deployment constraints.
      \end{itemize}
      
      \vspace{0.4cm}
      \textbf{Core Values}
      \begin{itemize}
        \item \textbf{Methodological Rigor:} Code and math must be verified and fully reproducible.
        \item \textbf{Collaboration:} Open communication and mutual help.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.46\textwidth}
      \begin{highlightbox}
        \textbf{Primary Objective} \\
        To build robust intelligence that operates seamlessly within constrained environments (see VertexFlow architecture \cite{meridian2025vertexflow} and Synapse systems \cite{synapse2024decentralized}).
      \end{highlightbox}
