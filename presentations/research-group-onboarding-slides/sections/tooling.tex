\section{Tooling \& Infrastructure}

\begin{frame}{Core Compute Resources}{The Apex GPU Cluster}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Cluster Architecture}
      \begin{itemize}
        \item \textbf{GPU Nodes:} 16x DGX H100 nodes, each with 8x 80GB H100 GPUs.
        \item \textbf{CPU Nodes:} 32x dual-socket EPYC nodes for preprocessing.
        \item \textbf{Storage Pool:} 2PB high-speed Ceph storage.
      \end{itemize}
      
      \vspace{0.3cm}
      \textbf{Scheduling Policies}
      \begin{itemize}
        \item Always use Slurm to request sessions.
        \item Keep interactive jobs under 2 hours.
      \end{itemize}
    \end{column}
    
    \begin{column}{0.46\textwidth}
      \begin{highlightbox}
        \textbf{\faServer\ Compute Allocations} \\
        Compute queues are allocated weekly. Post requests in the \texttt{\#meridian-compute} Slack channel by Sunday night.
      \end{highlightbox}
