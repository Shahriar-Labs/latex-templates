% 06 — Optimisation -----------------------------------------------------------
\begin{frame}{Optimising flexibility within production limits}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.51\textwidth}
      \eyebrow{Rolling-horizon objective}
      \begin{equation*}
        \min_{u_{1:H}}\;
        \sum_{t=1}^{H}
        \underbrace{p_t P_t}_{\text{energy}}
        + \underbrace{c_t^{\mathrm{peak}} z_t}_{\text{capacity}}
        + \underbrace{\lambda\lVert \Delta u_t\rVert_1}_{\text{wear}}
      \end{equation*}
      \vspace{-1mm}
      \begin{align*}
        x_{t+1} &= A x_t + B u_t + E w_t,\\[-1mm]
        x_t &\in \mathcal{X}_{\mathrm{safe}},\qquad
        u_t \in \mathcal{U}_{\mathrm{asset}}.
      \end{align*}
      {\small\color{SintefMuted}
        Forecast uncertainty tightens the safe envelope before dispatch.}
    \end{column}
    \begin{column}{.44\textwidth}
      \begin{exampleblock}{Hard constraints}
        Process temperature, pressure, state-of-charge, ramp rate, and operator lockouts.
      \end{exampleblock}
      \vspace{2mm}
      \begin{block}{Soft objectives}
        Cost, emissions, reserve value, asset wear, and schedule preference.
      \end{block}
      \vspace{2mm}
      \begin{alertblock}{Fail-safe behaviour}
        If data quality drops, the site returns to its last verified local policy.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
