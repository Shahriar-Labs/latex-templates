% Slide 4: Bullet Builds
\begin{frame}{Introduction to Edge-Cloud Dynamics}
  Managing workloads across decentralized networks requires addressing core constraints:
  \begin{itemize}[<+-| alert@+>]
    \item \faServer\ \textbf{System Heterogeneity:} Node capacity varies by orders of magnitude.
    \item \faNetworkWired\ \textbf{Workload Volatility:} Network demand fluctuates unpredictably.
    \item \faClock\ \textbf{Telemetry Latency:} Centralized scheduling suffers from sync delay.
  \end{itemize}
\end{frame}
