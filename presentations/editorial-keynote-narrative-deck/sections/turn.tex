\sectionslide{The objection}

\begin{frame}{Why a reasonable person disagrees}
  \eyebrow{Turn}
  \begin{pullquote}
  Put the strongest counter-argument on a slide, in its best form. A talk that only defeats
  weak versions of the objection convinces nobody who held it.
  \end{pullquote}
\end{frame}

\begin{frame}{What settles it}
  \bignumber{One figure}{that a sceptic would accept. Not the most impressive number you have,
  the most persuasive one.}
\end{frame}
