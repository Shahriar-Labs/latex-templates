\begin{frame}{Calibration Metric}
  \small
  \begin{block}{Expected Calibration Error}
    \[
      \mathrm{ECE} = \sum_{m=1}^{M} \frac{|B_m|}{n}
        \left| \mathrm{acc}(B_m) - \mathrm{conf}(B_m) \right|
    \]
  \end{block}
  \par\vspace{2pt}
  Notes are grouped into $M{=}15$ equal-width confidence bins $B_m$;
  $\mathrm{acc}(B_m)$ is the fraction of correct triage calls in the bin
  and $\mathrm{conf}(B_m)$ is the bin's mean predicted confidence.
  \par\vspace{4pt}
  \begin{block}{Conformal coverage guarantee (Aim 2)}
    \[
      \mathbb{P}\big[Y \in \mathcal{C}_\lambda(X)\big] \;\geq\; 1 - \alpha
    \]
  \end{block}
  Threshold $\lambda$ is selected on a calibration split so the
  escalate-to-human tier satisfies this bound at $\alpha = 0.05$.
\end{frame}
