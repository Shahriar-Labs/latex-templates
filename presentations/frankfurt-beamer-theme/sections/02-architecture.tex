% Slide 6: Distributed Pathfinding Algorithm
\begin{frame}{Decentralized MPC and Coordination}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Algorithmic Pipeline:}
            \begin{itemize}
                \item Local trajectory generation via MPC
                \item Peer-to-peer conflict detection
                \item Consensus-based priority assignment
                \item Trajectory repair and adjustment
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Local Optimization Model:}
            \begin{equation*}
                \min_{u} \sum_{t=0}^{H} \|x_t - x_{\text{goal}}\|^2_Q + \|u_t\|^2_R
            \end{equation*}
            Subject to:
            \begin{align*}
                x_{t+1} &= f(x_t, u_t) \\
                \|x_t - x^j_t\| &\ge d_{\text{min}}, \quad \forall j
            \end{align*}
        \end{column}
    \end{columns}
\end{frame}
