\section{Motivation}

\begin{frame}{The Problem}{Learned Policies Meet Unstructured Terrain}
  \begin{minipage}[t]{0.32\textwidth}
    \begin{navybox}{\faExclamationTriangle\ The Failure Mode}
      \footnotesize
      Policies trained in the \textbf{Nexa} simulator encounter terrain
      statistics at deployment that the training distribution never covered.
    \end{navybox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \begin{navybox}{\faRobot\ Where It Bites}
      \footnotesize
      Agricultural inspection and disaster-response rovers built on the
      \textbf{Vertex} platform must act under this mismatch \emph{without}
      a human in the loop.
    \end{navybox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \begin{navybox}{\faChartLine\ The Cost}
      \footnotesize
      Overconfident actions on out-of-distribution terrain produce
      near-misses and, in field trials, outright rollovers.
    \end{navybox}
  \end{minipage}
  \par\vspace{10pt}

  \begin{block}{Central claim of this proposal}
    A field robot's policy should know \emph{when its own dynamics model is
    guessing} -- and slow down, not merely act on the mean prediction.
  \end{block}
\end{frame}

\begin{frame}{Research Aims}{Three Aims, One Deployable System}
  \begin{minipage}[t]{0.32\textwidth}
    \begin{copperbox}{Aim 1}
      \footnotesize
      Design a \textbf{calibrated, onboard-feasible} uncertainty estimator
      for learned terrain-interaction dynamics models.
    \end{copperbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \begin{copperbox}{Aim 2}
      \footnotesize
      Develop a \textbf{risk-aware policy optimization} framework that
      trades exploration against a formal safety budget.
    \end{copperbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \begin{copperbox}{Aim 3}
      \footnotesize
      \textbf{Validate in the field} on the Vertex rover across three
      terrain classes at the Meridian Outcrop Test Range.
    \end{copperbox}
  \end{minipage}
  \par\vspace{12pt}

  \begin{itemize}
    \item Aims 1--2 are trained and benchmarked in simulation before any
      hardware commitment.
    \item Aim 3 closes the loop: the same uncertainty estimate that shapes
      the policy in simulation is the one logged on the physical rover.
  \end{itemize}
\end{frame}
