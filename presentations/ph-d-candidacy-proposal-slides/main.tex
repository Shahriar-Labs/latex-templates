\sectionframe{1}{Problem and thesis}{The reliability gap appears where changing data meets changing human behavior.}

% 4 -- Motivation
\begin{frame}{Silent failure sustains trust}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \eyebrow{Operational pattern}
      \begin{enumerate}
        \item A model is validated on historical data.
        \item Deployment changes the data and workflow.
        \item Accuracy erodes before labels arrive.
        \item Users cannot see when reliance is unsafe.
      \end{enumerate}
    \column{0.40\textwidth}
      \begin{thesisbox}
        \eyebrow{Core problem}
        Current monitoring asks whether the \emph{model} shifted.
        Safe deployment requires knowing whether the
        \textbf{joint human--AI decision} became unreliable.
      \end{thesisbox}
  \end{columns}

  \vspace{0.20cm}
  {\color{Slate}\fontsize{14}{17}\selectfont
   Scope: decision-support settings with consequential outcomes, delayed labels,
   and a human who can accept, override, or ignore an AI recommendation.}
\end{frame}

% 5 -- Literature gap
\begin{frame}{The gap spans three fields}
  \begin{columns}[T,onlytextwidth]
    \column{0.30\textwidth}
      \eyebrow{Shift detection}
      Measures changes in inputs, outputs, or representations.

      \vspace{0.22cm}{\color{Coral}\bfseries Missing:} downstream human response.
    \column{0.04\textwidth}
      \centering\color{Gold}\rule{1pt}{3.75cm}
    \column{0.30\textwidth}
      \eyebrow{Human reliance}
      Studies trust, calibration, explanations, and overrides.

      \vspace{0.22cm}{\color{Coral}\bfseries Missing:} changing deployment data.
    \column{0.04\textwidth}
      \centering\color{Gold}\rule{1pt}{3.75cm}
    \column{0.30\textwidth}
      \eyebrow{Adaptive systems}
      Updates models using retraining or test-time adaptation.

      \vspace{0.22cm}{\color{Coral}\bfseries Missing:} behavioral safety objectives.
  \end{columns}

  \vspace{0.22cm}
  \begin{thesisbox}
    \textbf{Research gap:} no unified account links detectable shift,
    changing reliance, and an intervention that improves the final decision.
  \end{thesisbox}
\end{frame}

% 6 -- Thesis and RQs
\begin{frame}{One thesis and three questions}
  \begin{thesisbox}
    \textbf{Thesis.} Deployment risk can be identified earlier and reduced more reliably
    when monitoring combines model uncertainty with human reliance signals, then triggers
    an intervention matched to the observed failure mode.
  \end{thesisbox}

  \vspace{0.22cm}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries\color{Ocean}}l@{\hspace{0.32cm}}>{\RaggedRight\arraybackslash}X@{}}
    RQ1 & Which unlabeled signals predict degradation in joint decision quality?\\[0.22cm]
    RQ2 & How does distribution shift causally change acceptance and override behavior?\\[0.22cm]
    RQ3 & Which targeted intervention restores calibration with the least added burden?\\
  \end{tabularx}
\end{frame}

% 10 -- Study 1
\begin{frame}{Study 1: Detect risk early}
  \studylabel{Ocean}{STUDY 1}\hspace{0.18cm}
  {\color{Midnight}\fontsize{20}{24}\selectfont\bfseries Prospective monitoring study}

  \vspace{0.34cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \eyebrow{Design}
      Replay timestamped cases in order; expose the system to gradual and abrupt shifts;
      delay outcome labels to mirror deployment.

      \vspace{0.36cm}
      \eyebrow{Signals}
      Predictive entropy, representation drift, disagreement, acceptance rate, and response time.
    \column{0.47\textwidth}
      \eyebrow{Primary test}
      Compare signal combinations on lead time and precision for detecting a clinically or
      operationally meaningful rise in joint error.

      \vspace{0.36cm}
      \eyebrow{Falsification}
      Reject H1 if behavioral signals add no out-of-sample value beyond model-only monitoring.
  \end{columns}
\end{frame}

% 11 -- Study 2
\begin{frame}{Study 2: Explain reliance}
  \studylabel{Coral}{STUDY 2}\hspace{0.18cm}
  {\color{Midnight}\fontsize{20}{24}\selectfont\bfseries Controlled mechanism experiment}

  \vspace{0.34cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \eyebrow{Manipulation}
      Randomize shift visibility, model confidence, and explanation quality in a
      preregistered $2\times2\times2$ design.

      \vspace{0.36cm}
      \eyebrow{Measured behavior}
      Acceptance, appropriate override, decision time, confidence, and perceived workload.
    \column{0.47\textwidth}
      \eyebrow{Causal estimand}
      Average treatment effect of visible shift cues on appropriate reliance,
      with expertise as a prespecified moderator.

      \vspace{0.36cm}
      \eyebrow{Falsification}
      Reject H2 if shift cues change confidence but do not improve appropriate reliance.
  \end{columns}
\end{frame}

% 12 -- Study 3
\begin{frame}{Study 3: Target intervention}
  \studylabel{Success}{STUDY 3}\hspace{0.18cm}
  {\color{Midnight}\fontsize{20}{24}\selectfont\bfseries Randomized intervention trial}

  \vspace{0.20cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \eyebrow{Policy arms}
      \begin{enumerate}
        \item Standard confidence display
        \item Contextual shift alert
        \item Selective deferral to human review
      \end{enumerate}
    \column{0.46\textwidth}
      \eyebrow{Success criterion}
      Lower joint decision loss without a practically unacceptable increase in time or workload.

      \vspace{0.22cm}
      \eyebrow{Falsification}
      Reject H3 if the best policy fails the preregistered benefit--burden threshold.
  \end{columns}

  \vspace{0.16cm}
  {\color{Coral}\bfseries
   Trigger the intervention by failure mode, not a generic uncertainty threshold.}
\end{frame}

\sectionframe{3}{Evidence and execution}{Validity comes from converging domains, preregistered tests, and staged decisions.}

% 14 -- Data and domains
\begin{frame}{Two domains test generality}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \eyebrow{Domain A: clinical triage}
      Consequential decisions, delayed outcomes, specialist users, and strong governance constraints.

      \vspace{0.34cm}
      \textbf{Role in thesis:} high-stakes validation and expert-reliance analysis.
    \column{0.47\textwidth}
      \eyebrow{Domain B: infrastructure incidents}
      Rapid decisions, rich event logs, recurring operational shifts, and measurable response cost.

      \vspace{0.34cm}
      \textbf{Role in thesis:} replication, stress testing, and external validity.
  \end{columns}

  \vspace{0.26cm}
  \begin{thesisbox}
    \textbf{Transfer claim:} the monitoring logic should transfer; domain-specific harm weights
    and intervention thresholds should not.
  \end{thesisbox}
\end{frame}

% 15 -- Evaluation
\begin{frame}{Three claims need three tests}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries\color{Ocean}}l@{\hspace{0.35cm}}>{\RaggedRight\arraybackslash}X>{\RaggedRight\arraybackslash}X@{}}

  \end{thesisbox}
\end{frame}

% 17 -- Ethics and risks
\begin{frame}{Each risk has a fallback}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries\color{Coral}}l@{\hspace{0.35cm}}>{\RaggedRight\arraybackslash}X>{\RaggedRight\arraybackslash}X@{}}
    \toprule
    Risk & Prevention & Fallback\\
    \midrule
    Restricted data & De-identification, minimum access, audit log & Public benchmark plus synthetic shift\\[0.20cm]
    Weak behavior signal & Validated task, pilot reliability threshold & Narrow claim to observable actions\\[0.20cm]
    Alert fatigue & Burden cap and human-factors review & Selective deferral only\\[0.20cm]
    Unequal benefit & Prespecified subgroup analysis & Group-conditional threshold or no deployment\\
    \bottomrule
  \end{tabularx}

  \vspace{0.20cm}
  {\color{Slate}\fontsize{14}{17}\selectfont
   No live clinical recommendation is made during the dissertation; all intervention studies
   use retrospective replay or controlled simulation until separate deployment approval.}
\end{frame}
