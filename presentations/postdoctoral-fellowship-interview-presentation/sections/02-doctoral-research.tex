\section{Doctoral Research}

\begin{frame}
\frametitle{The Problem}
\framesubtitle{Ab initio accuracy is too slow for the size of the search space}

\begin{columns}[T]
\begin{column}{0.52\linewidth}
\textbf{Solid-state electrolytes} are the bottleneck for safe, high-energy-density batteries, but:
\begin{itemize}
  \item Ab initio molecular dynamics (AIMD) is accurate but far too slow to screen ionic conductivity across candidate chemistries
  \item Classical force fields are fast but too inaccurate for the disordered, defect-rich structures that make good conductors
  \item Existing machine-learned potentials need thousands of labeled structures per chemistry -- a cost few labs can absorb
\end{itemize}
\end{column}
\begin{column}{0.44\linewidth}
\begin{tcolorbox}[colback=primary!6,colframe=primary,title={\small Research Question},fonttitle=\bfseries,coltitle=white]
\small Can we build a machine-learned potential that reaches AIMD-level accuracy on ionic transport while needing an order of magnitude less training data?
\end{tcolorbox}
\end{column}
\end{columns}
\end{frame}
