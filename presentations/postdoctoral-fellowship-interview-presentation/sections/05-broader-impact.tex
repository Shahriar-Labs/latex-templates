\section{Broader Impact}

\begin{frame}
\frametitle{Funding, Dissemination \& Community Value}
\framesubtitle{Positioning the project beyond a single lab}

\begin{columns}[T]
\begin{column}{0.5\linewidth}
\textbf{Funding fit}
\begin{itemize}
  \item Directly addresses the Meridian Foundation Energy Grand Challenge priority area on solid-state battery materials
  \item Builds a competitive case for a joint Nimbus Lab / Vertex University follow-on proposal
\end{itemize}
\end{column}
\begin{column}{0.5\linewidth}
\textbf{Dissemination}
\begin{itemize}
  \item Open dataset and code release under a permissive license
  \item Workshop contribution at the Synapse Solid-State Energy Symposium
  \item Shared benchmark suite to accelerate reproducible comparisons across labs
\end{itemize}
\end{column}
\end{columns}

\vspace{0.25cm}
\begin{block}{Why this matters for the field}
\footnotesize Halide solid electrolytes are under-explored relative to oxides and sulfides largely because no fast, transferable screening tool exists. Closing that gap benefits every group working on solid-state batteries, not just ours.
\end{block}
\end{frame}
