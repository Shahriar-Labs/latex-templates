\section{Doctoral Research}

\begin{frame}
\frametitle{The Problem}
\framesubtitle{Ab initio accuracy is too slow for the size of the search space}

\begin{columns}[T]
\begin{column}{0.52\linewidth}
\textbf{Solid-state electrolytes} are the bottleneck for safe, high-energy-density batteries, but:
\begin{itemize}
  \item Ab initio molecular dynamics (AIMD) is accurate but far too slow to screen ionic conductivity across candidate chemistries
  \item Classical force fields are fast but too inaccurate for the disordered, defect-rich structures that make good conductors
  \item Existing machine-learned potentials need thousands of labeled structures per chemistry -- a cost few labs can absorb
\end{itemize}
\end{column}
\begin{column}{0.44\linewidth}
\begin{tcolorbox}[colback=primary!6,colframe=primary,title={\small Research Question},fonttitle=\bfseries,coltitle=white]
\small Can we build a machine-learned potential that reaches AIMD-level accuracy on ionic transport while needing an order of magnitude less training data?
\end{tcolorbox}
\end{column}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}
\frametitle{Key Contributions}
\framesubtitle{Three results from the doctoral thesis}

\footnotesize
\begin{tabularx}{\linewidth}{@{}l X X@{}}
\toprule
Contribution & Venue / Validation & Impact \\
\midrule
IonForge machine-learned potential framework & \emph{Journal of Computational Materials Science}, 2024 & 40$\times$ speedup vs.\ AIMD; $<$2\,meV/atom force error \\
Active-learning sampling protocol & \emph{Computational Energy Materials}, 2025 & Cut required training structures by 65\% \\
Discovery of 12 candidate garnet-type conductors & Synapse Battery Materials Consortium, Annual Meeting 2025 & 3 candidates validated experimentally by partner labs \\
\bottomrule
\end{tabularx}

\vspace{0.05cm}
\begin{block}{Headline result}
\footnotesize IonForge matches AIMD force accuracy on Li-ion conductors while training on 65\% fewer structures than the strongest published baseline.
\end{block}
\end{frame}
