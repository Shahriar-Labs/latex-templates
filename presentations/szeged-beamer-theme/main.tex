\section{The gap}
\subsection{Why trails break}

% -----------------------------------------------------------------------------
% 3. Problem
% -----------------------------------------------------------------------------
\begin{frame}{A saved script does not preserve the experiment}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}
        \item Parameters drift between exploratory runs and reported results.
        \item Data snapshots change while filenames remain the same.
        \item Package, hardware, and operating-system details disappear.
        \item Manual exclusions survive only in a notebook margin or chat thread.
      \end{itemize}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{alertblock}{The hidden failure}
        The chart can remain visible even after the path that produced it has been
        lost.
      \end{alertblock}
      \vspace{0.15cm}
      \begin{exampleblock}{The required shift}
        Treat provenance as part of the result, not as documentation added later.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\subsection{Design principles}

% -----------------------------------------------------------------------------
% 4. Principles
% -----------------------------------------------------------------------------
\begin{frame}{Three principles keep the evidence attached}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{Capture by default}
        Record configuration, code identity, data identity, environment, and seed
        at execution time.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Make runs immutable}
        Create a new run record for every change; never revise history in place.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Verify independently}
        Test replayability and interpretation with checks separate from the analysis
        code.
      \end{block}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \color{SzegedBlue}\large\bfseries
  The trail should be cheaper to create than to reconstruct.
\end{frame}

\subsection{Identity}

% -----------------------------------------------------------------------------
% 6. Run identity
% -----------------------------------------------------------------------------
\begin{frame}{A run fingerprint makes silent changes visible}
  For a computational run $r$, define its identity from the resolved experiment
  state:
  \begin{equation*}
    \phi(r)
      = H\!\left(
          h_{\mathrm{code}}\,\Vert\,
          h_{\mathrm{data}}\,\Vert\,
          h_{\mathrm{env}}\,\Vert\,
          \theta\,\Vert\,s
        \right),
  \end{equation*}
  where $\theta$ is the canonical parameter set, $s$ is the random seed, and
  $H$ is a collision-resistant hash.

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Same fingerprint}
        The declared computational state is identical and the run is eligible for
        a replay comparison.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Different fingerprint}
        Something material changed; compare manifests before comparing outcomes.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vfill
  \small\color{SzegedInk!82}
  A fingerprint identifies the declared state; validation still determines whether
  the run behaved as expected.
\end{frame}

\section{The assurance case}
\subsection{Validation ladder}

% -----------------------------------------------------------------------------
% 8. Validation
% -----------------------------------------------------------------------------
\begin{frame}{Validation moves from execution to interpretation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{enumerate}
        \item \textbf{Integrity}\quad Inputs and outputs match their checksums.
        \item \textbf{Executability}\quad The environment rebuilds and the workflow completes.
        \item \textbf{Determinism}\quad Repeated runs meet the declared tolerance.
      \end{enumerate}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{enumerate}
        \setcounter{enumi}{3}
        \item \textbf{Robustness}\quad Controlled perturbations reveal sensitivity.
        \item \textbf{Traceability}\quad Every claim links to a checked artifact.
        \item \textbf{Reviewability}\quad A new reader can follow the trail without private context.
      \end{enumerate}
    \end{column}
  \end{columns}

  \vfill
  \begin{exampleblock}{Acceptance rule}
    Publish only when every required rung passes; otherwise place the owned
    exception beside the claim.
  \end{exampleblock}
\end{frame}

\section{Adoption}
\subsection{Implementation path}

% -----------------------------------------------------------------------------
% 10. Roadmap
% -----------------------------------------------------------------------------
\begin{frame}{Adopt the trail in four bounded steps}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.23\textwidth}
      \taglabel{STEP 1}\par\vspace{0.12cm}
      \textbf{Inventory}\par
      \small Map where inputs, parameters, environments, and outputs currently live.
    \end{column}
    \begin{column}{0.23\textwidth}
      \taglabel{STEP 2}\par\vspace{0.12cm}
      \textbf{Standardize}\par
      \small Define one manifest schema and one immutable run directory convention.
    \end{column}
    \begin{column}{0.23\textwidth}
      \taglabel{STEP 3}\par\vspace{0.12cm}
      \textbf{Automate}\par
      \small Capture identity and validation evidence inside the execution path.
    \end{column}
    \begin{column}{0.23\textwidth}
      \taglabel{STEP 4}\par\vspace{0.12cm}
      \textbf{Govern}\par
      \small Require evidence links at review and rehearse replay with a new operator.
    \end{column}
  \end{columns}

  \vspace{0.65cm}
  \begin{block}{Start with the highest-value workflow}
    Choose one result that is frequently revised, reused, or challenged. Make its
    trail complete before scaling the convention across projects.
  \end{block}
\end{frame}

\subsection{Decision}

% -----------------------------------------------------------------------------
% 11. Close
% -----------------------------------------------------------------------------
\begin{frame}{Before the next run, decide what must be replayable}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item Put the experiment contract in a machine-readable manifest.
        \item Give every run an immutable identity and evidence bundle.
        \item Validate the trail independently of the analysis that produced it.
        \item Link each released claim to the artifact that supports it.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{exampleblock}{The practical standard}
        A credible result is not merely repeatable by its creator; it is inspectable
        and replayable by someone new.
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vfill
  \centering
  {\color{SzegedBlue}\Large\bfseries Preserve the path, not only the picture.}
\end{frame}
