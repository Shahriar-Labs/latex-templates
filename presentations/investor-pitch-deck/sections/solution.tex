\section{Solution}

\begin{frame}{Meet Nimbus}{One Forecasting Layer Across the Business}

  Nimbus plugs into a company's existing ERP, POS, and inventory systems and replaces
  manual demand planning with a continuously-updating forecast -- and the actions that
  follow from it.

  \vspace{0.45cm}

  \begin{minipage}[t]{0.31\textwidth}
    \begin{panelboxalt}{\textcolor{accentViolet}{\faChartLine}\ \ Forecast}
      \small SKU-level demand forecasts refreshed every 15 minutes, accurate to
      \textbf{93\% at the 4-week horizon}.
    \end{panelboxalt}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.31\textwidth}
    \begin{panelboxalt}{\textcolor{accentViolet}{\faCogs}\ \ Optimize}
      \small Replenishment and safety-stock targets recalculated automatically as the
      forecast moves.
    \end{panelboxalt}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.31\textwidth}
    \begin{panelboxalt}{\textcolor{accentViolet}{\faBell}\ \ Act}
      \small Purchase-order and transfer recommendations pushed straight into the
      customer's ERP, no rip-and-replace.
    \end{panelboxalt}
  \end{minipage}
  \par

\end{frame}
