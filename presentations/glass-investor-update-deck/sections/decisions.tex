\sectionslide{Decisions}

\begin{frame}{What we decided}
  \begin{glass}
  One decision per update, with the alternative you rejected. An update that lists activity
  rather than decisions is a status report, and nobody reads those twice.
  \end{glass}
\end{frame}

\begin{frame}{What went wrong}
  \begin{glass}
  Say it plainly and say what changed as a result. Investors who find out later from someone
  else discount everything else in the letter.
  \end{glass}
\end{frame}
