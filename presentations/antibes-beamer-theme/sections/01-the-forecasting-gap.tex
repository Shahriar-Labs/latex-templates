\section{The forecasting gap}
\subsection{Motivation}

% -----------------------------------------------------------------------------
% 3. Motivation
% -----------------------------------------------------------------------------
\begin{frame}{Citywide forecasts arrive too late for street-level decisions}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}[<+->]
        \item Traffic states can change within minutes after an incident, event, or downpour.
        \item Central pipelines must first upload, reconcile, and process thousands of sensor streams.
        \item Raw trajectories create privacy exposure and costly network demand.
        \item A useful forecast must be \alert{local, timely, and trustworthy under missing data}.
      \end{itemize}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{alertblock}{Operational mismatch}
        A 15-minute forecast has little value when computation and transfer consume most of that horizon.
      \end{alertblock}
      \vspace{0.25cm}
      \begin{exampleblock}{Research question}
        Can intersections collaborate without exporting individual journeys?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\subsection{Design target}

% -----------------------------------------------------------------------------
% 4. Design principles
% -----------------------------------------------------------------------------
\begin{frame}{Three commitments define a deployable forecasting service}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{1. Learn locally}
        Gateways retain raw counts and trajectories; only clipped model updates leave a district.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{2. Act quickly}
        Inference stays on the signal-control path, independent of a cloud round trip.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{3. Fail visibly}
        Confidence falls when inputs are stale, turning uncertainty into an explicit operator signal.
      \end{block}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \color{PulseNavy}\large\bfseries
  Coordinate models globally; keep observations and urgent decisions local.
\end{frame}
