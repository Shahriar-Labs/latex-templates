\begin{frame}{The design earns speed by accepting explicit trade-offs}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Strong fit}
        \begin{itemize}
          \item recurring corridors
          \item constrained uplinks
          \item strict data boundaries
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Manage carefully}
        \begin{itemize}
          \item uneven district volume
          \item concept drift
          \item model-version skew
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Escalate}
        \begin{itemize}
          \item mass sensor outage
          \item unseen event pattern
          \item unsafe phase proposal
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \small
  Privacy does not remove accountability: every promoted model retains a signed audit trail and rollback target.
\end{frame}
