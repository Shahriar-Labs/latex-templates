\section{The forecasting gap}
\subsection{Motivation}

% -----------------------------------------------------------------------------
% 3. Motivation
% -----------------------------------------------------------------------------
\begin{frame}{Citywide forecasts arrive too late for street-level decisions}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}[<+->]
        \item Traffic states can change within minutes after an incident, event, or downpour.
        \item Central pipelines must first upload, reconcile, and process thousands of sensor streams.
        \item Raw trajectories create privacy exposure and costly network demand.
        \item A useful forecast must be \alert{local, timely, and trustworthy under missing data}.
      \end{itemize}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{alertblock}{Operational mismatch}
        A 15-minute forecast has little value when computation and transfer consume most of that horizon.
      \end{alertblock}
      \vspace{0.25cm}
      \begin{exampleblock}{Research question}
        Can intersections collaborate without exporting individual journeys?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\subsection{Design target}

% -----------------------------------------------------------------------------
% 4. Design principles
% -----------------------------------------------------------------------------
\begin{frame}{Three commitments define a deployable forecasting service}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{1. Learn locally}
        Gateways retain raw counts and trajectories; only clipped model updates leave a district.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{2. Act quickly}
        Inference stays on the signal-control path, independent of a cloud round trip.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{3. Fail visibly}
        Confidence falls when inputs are stale, turning uncertainty into an explicit operator signal.
      \end{block}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \color{PulseNavy}\large\bfseries
  Coordinate models globally; keep observations and urgent decisions local.
\end{frame}

\subsection{Learning objective}

% -----------------------------------------------------------------------------
% 6. Objective
% -----------------------------------------------------------------------------
\begin{frame}{One objective balances accuracy, delay, and privacy}
  District $k$ optimizes a local forecasting loss $F_k$; the coordinator combines updates as
  \begin{equation*}
    \min_{\boldsymbol{w}}
    \underbrace{\sum_{k=1}^{K} p_k F_k(\boldsymbol{w})}_{\text{forecast error}}
    + \lambda\underbrace{\lVert\boldsymbol{w}\rVert_2^2}_{\text{stability}},
    \qquad
    \widetilde{\boldsymbol{g}}_k
    = \operatorname{clip}_{C}(\boldsymbol{g}_k)
      + \mathcal{N}(\boldsymbol{0},\sigma^2 C^2\mathbf{I}).
  \end{equation*}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $p_k$: weight based on usable observations,
        \item $C$: bound on one district's influence,
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $\sigma$: calibrated privacy noise,
        \item $\lambda$: resistance to unstable updates.
      \end{itemize}
    \end{column}
  \end{columns}

  \begin{exampleblock}{Control contract}
    Publish a forecast only when latency is below $2$ seconds and calibrated confidence exceeds the district threshold.
  \end{exampleblock}
\end{frame}

% -----------------------------------------------------------------------------
% 7. Update cycle
% -----------------------------------------------------------------------------
\begin{frame}{The update cycle separates urgent inference from patient learning}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{enumerate}
        \item \textbf{Observe:} align recent counts, speeds, and signal states.
        \item \textbf{Infer:} predict the next three 5-minute intervals locally.
        \item \textbf{Protect:} clip, noise, and encrypt the scheduled update.
        \item \textbf{Aggregate:} combine updates robustly across available districts.
        \item \textbf{Adopt:} promote a model only after local safety checks pass.
      \end{enumerate}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{tikzpicture}[x=0.78cm,y=0.78cm]

\subsection{Takeaways}

% -----------------------------------------------------------------------------
% 13. Conclusion
% -----------------------------------------------------------------------------
\begin{frame}{Three ideas to take home}
  \begin{enumerate}
    \item \textbf{Move decisions, not raw journeys.} Edge inference shortens the operational path and limits exposure.
    \item \textbf{Treat uncertainty as product behavior.} Missing evidence must change confidence, escalation, and control authority.
    \item \textbf{Deploy through measurable gates.} Accuracy alone is insufficient; latency, privacy, calibration, and rollback complete the contract.
  \end{enumerate}

  \vfill
  \begin{exampleblock}{Result in one sentence}
    CivicPulse approaches centralized accuracy with a 6.7$\times$ faster response and 97\% less uplink traffic in the illustrative replay.
  \end{exampleblock}
\end{frame}
