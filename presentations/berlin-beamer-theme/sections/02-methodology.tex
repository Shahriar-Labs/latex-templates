\begin{frame}{Vertex Framework Architecture}
  Decentralized control loop split between localized edge agents and a cloud orchestrator:
  \vspace{0.1cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Core Components:}
      \vspace{0.1cm}
      \begin{itemize}
        \item \textbf{Edge Observers}: Monitor local power draw and voltage fluctuations.
        \item \textbf{Decision Agents}: Localized policy execution via reinforcement learning.
        \item \textbf{Global Orchestrator}: Asynchronously updates global weights.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \begin{tikzpicture}[node distance=0.7cm, every node/.style={fill=white, draw=SynapseBlue, rounded corners, thick, align=center, font=\scriptsize}, scale=0.65, transform shape]

        \node (agent) [below=of global, fill=SynapseBlue!10] {Vertex Decision Agent};
        \node (edge) [below=of agent, fill=gray!10] {Edge Observer};
        
        \draw[->, thick, >=stealth, draw=SynapseBlue] (global.300) -- node[right, draw=none, fill=none, font=\tiny] {Model Weights} (agent.60);
