% Slide 3: Motivation & Research Challenges (with bullet builds)
\begin{frame}{Motivation: Data Orchestration Challenges}
    Distributed systems face significant orchestration overhead:
    \begin{itemize}
        \item<1-> \textbf{High Latency:} Communication delays between remote nodes.
        \item<2-> \textbf{Data Inconsistency:} Conflicting updates in replicated databases.
        \item<3-> \textbf{Resource Over-allocation:} Excess computing power used for routing tasks.
        \item<4-> \textbf{Dynamic Topology:} Nodes joining and leaving the network seamlessly.
    \end{itemize}
\end{frame}

% Slide 4: Research Objectives
\begin{frame}{Research Objectives}
    We focus on addressing these challenges through the Nexa framework:
    \begin{itemize}
        \item Optimize task distribution dynamically across networks.
        \item Reduce node synchronization overhead by up to 40\%.
        \item Support self-healing network structures in real-time.
    \end{itemize}
\end{frame}
