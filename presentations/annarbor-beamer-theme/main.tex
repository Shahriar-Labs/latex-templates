\section{The trust gap}

% -----------------------------------------------------------------------------
% 3. Motivation
% -----------------------------------------------------------------------------
\begin{frame}{Public-service answers carry consequences beyond the chat window}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}[<+->]
        \item Policies change across jurisdictions, programs, and effective dates.
        \item A fluent answer can conceal missing evidence or an outdated source.
        \item Staff need a short response; reviewers need the complete decision trail.
        \item The design target is therefore \alert{useful under time pressure and auditable afterward}.
      \end{itemize}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{alertblock}{Failure mode}
        Treating retrieval as background context allows unsupported claims to survive into the final answer.
      \end{alertblock}
      \vspace{0.18cm}
      \begin{exampleblock}{Core question}
        Can every consequential sentence be traced to current, authorized evidence?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% 4. Requirements
% -----------------------------------------------------------------------------
\begin{frame}{Three requirements turn retrieval into an assurance layer}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{1. Bound the corpus}
        Index only approved sources, preserve effective dates, and attach ownership metadata.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{2. Verify each claim}
        Separate candidate generation from evidence checking before any answer is released.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{3. Preserve the trace}
        Store query, source version, passages, verdicts, and the final response together.
      \end{block}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \color{ArborNavy}\large\bfseries
  Retrieval finds evidence; verification decides whether that evidence is enough.
\end{frame}

% -----------------------------------------------------------------------------
% 6. Decision rule
% -----------------------------------------------------------------------------
\begin{frame}{A bounded score makes the release policy explicit}
  For answer $a$ with claims $c_1,\ldots,c_m$, define
  \begin{equation*}
    Q(a)
    = \sum_{i=1}^{m} w_i\,s(c_i,e_i)
      - \lambda U(a)
      - \gamma A(a),
    \qquad
    \sum_{i=1}^{m} w_i=1,
  \end{equation*}
  where
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $s(c_i,e_i)\in[0,1]$: support from cited evidence,
        \item $U(a)$: unresolved ambiguity or missing context,
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $A(a)$: source-age penalty after a policy update,
        \item $\lambda,\gamma$: risk-calibrated penalties.
      \end{itemize}
    \end{column}
  \end{columns}

  \begin{block}{Release rule}
    Deliver only when $Q(a)\geq\tau$, every high-impact claim has direct support, and no source is outside its validity window; otherwise ask for context or escalate.
  \end{block}
\end{frame}

\section{Evaluation}

% -----------------------------------------------------------------------------
% 8. Evaluation design
% -----------------------------------------------------------------------------
\begin{frame}{Evaluation separates answer quality from operational safety}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item \textbf{1,200 scenarios}: routine, ambiguous, outdated, and adversarial queries.
        \item \textbf{Three systems}: generation only, retrieval augmented, and gated verification.
        \item \textbf{Double review}: policy specialists label support and acceptable escalation.
        \item \textbf{Update drill}: 10\% of source documents change midway through the test.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Primary measures}
        \begin{itemize}
          \item claim support rate,
          \item unsafe-release rate,
          \item useful-answer rate,
          \item median latency.
        \end{itemize}
      \end{block}
      \begin{exampleblock}{Evaluation principle}
        A safe escalation counts as correct behavior, not as an unanswered query.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Deployment}

% -----------------------------------------------------------------------------
% 11. Limitations
% -----------------------------------------------------------------------------
\begin{frame}{A verified answer can still fail the person receiving it}
  \begin{block}{Coverage}
    Missing policy documents cannot be repaired by a stronger verifier; corpus owners need explicit coverage targets and update service levels.
  \end{block}
  \begin{alertblock}{Interpretation}
    Evidence may support a sentence while the overall guidance remains misleading because of omitted exceptions or inaccessible language.
  \end{alertblock}
  \begin{exampleblock}{Accountability}
    Human review remains mandatory for appeals, conflicting authorities, and decisions that materially affect rights or benefits.
  \end{exampleblock}
\end{frame}

% -----------------------------------------------------------------------------
% 13. Summary
% -----------------------------------------------------------------------------
\begin{frame}{Evidence-first design changes what ``good'' looks like}
  \begin{enumerate}
    \item \textbf{Constrain the evidence.} Authorized, versioned sources define the assistant's knowledge boundary.
    \item \textbf{Verify the claims.} A separate release gate checks support, ambiguity, and freshness.
    \item \textbf{Keep the trace.} Every consequential output remains reproducible and reviewable.
    \item \textbf{Escalate deliberately.} Uncertainty triggers clarification or human judgment, not improvisation.
  \end{enumerate}

  \vfill
  \begin{exampleblock}{Takeaway}
    The safest assistant is not the one that always answers; it is the one that knows what evidence permits it to say.
  \end{exampleblock}
\end{frame}
