% 8. Evaluation design
% -----------------------------------------------------------------------------
\begin{frame}{Evaluation separates answer quality from operational safety}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item \textbf{1,200 scenarios}: routine, ambiguous, outdated, and adversarial queries.
        \item \textbf{Three systems}: generation only, retrieval augmented, and gated verification.
        \item \textbf{Double review}: policy specialists label support and acceptable escalation.
        \item \textbf{Update drill}: 10\% of source documents change midway through the test.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Primary measures}
        \begin{itemize}
          \item claim support rate,
          \item unsafe-release rate,
          \item useful-answer rate,
          \item median latency.
        \end{itemize}
      \end{block}
      \begin{exampleblock}{Evaluation principle}
        A safe escalation counts as correct behavior, not as an unanswered query.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
