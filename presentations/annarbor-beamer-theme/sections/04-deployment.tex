% 11. Limitations
% -----------------------------------------------------------------------------
\begin{frame}{A verified answer can still fail the person receiving it}
  \begin{block}{Coverage}
    Missing policy documents cannot be repaired by a stronger verifier; corpus owners need explicit coverage targets and update service levels.
  \end{block}
  \begin{alertblock}{Interpretation}
    Evidence may support a sentence while the overall guidance remains misleading because of omitted exceptions or inaccessible language.
  \end{alertblock}
  \begin{exampleblock}{Accountability}
    Human review remains mandatory for appeals, conflicting authorities, and decisions that materially affect rights or benefits.
  \end{exampleblock}
\end{frame}

% 13. Summary
% -----------------------------------------------------------------------------
\begin{frame}{Evidence-first design changes what ``good'' looks like}
  \begin{enumerate}
    \item \textbf{Constrain the evidence.} Authorized, versioned sources define the assistant's knowledge boundary.
    \item \textbf{Verify the claims.} A separate release gate checks support, ambiguity, and freshness.
    \item \textbf{Keep the trace.} Every consequential output remains reproducible and reviewable.
    \item \textbf{Escalate deliberately.} Uncertainty triggers clarification or human judgment, not improvisation.
  \end{enumerate}

  \vfill
  \begin{exampleblock}{Takeaway}
    The safest assistant is not the one that always answers; it is the one that knows what evidence permits it to say.
  \end{exampleblock}
\end{frame}
