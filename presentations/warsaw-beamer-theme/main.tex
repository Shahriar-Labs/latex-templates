\section{The decision gap}

% -----------------------------------------------------------------------------
% 3. Motivation
% -----------------------------------------------------------------------------
\begin{frame}{An alert is useful only if it arrives before conditions change}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \begin{itemize}[<+->]
        \item Sensors observe rainfall, channel depth, and flow at different rates.
        \item Central processing adds a network dependency exactly when links are least reliable.
        \item A late or unstable alert erodes operator trust, even when the model is accurate.
        \item The design target is therefore \alert{timely, explainable, and graceful degradation}.
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{alertblock}{Decision gap}
        Raw measurements do not protect a neighborhood. A dependable pipeline must turn partial evidence into a bounded-risk action.
      \end{alertblock}
      \vspace{0.25cm}
      \begin{exampleblock}{Core question}
        How much intelligence must remain local when cloud connectivity disappears?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -----------------------------------------------------------------------------
% 4. Design principles
% -----------------------------------------------------------------------------
\begin{frame}{Three principles keep the system useful during disruption}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{1. Decide locally}
        Each gateway maintains a compact risk model and can issue a provisional alert without a round trip.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{2. Fuse uncertainty}
        Missing sensors reduce confidence explicitly instead of silently changing the prediction.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{3. Reconcile later}
        The cloud coordinates models and audit history, but it is not on the critical alert path.
      \end{block}
    \end{column}
  \end{columns}

  \vfill
  \centering
  \color{AstraNavy}\large\bfseries
  Local autonomy shortens the path; explicit uncertainty keeps that autonomy accountable.
\end{frame}

% -----------------------------------------------------------------------------
% 6. Decision model
% -----------------------------------------------------------------------------
\begin{frame}{Risk combines hazard, confidence, and the cost of waiting}
  For gateway $g$ at time $t$, the policy evaluates a bounded risk score
  \begin{equation*}
    R_{g,t}
    = \sigma\!\left(
      \boldsymbol{w}^{\mathsf T}\boldsymbol{x}_{g,t}
      - \lambda u_{g,t}
      + \gamma \Delta_{g,t}
    \right),
    \qquad \sigma(z)=\frac{1}{1+e^{-z}},
  \end{equation*}
  where
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $\boldsymbol{x}_{g,t}$: recent sensor features,
        \item $u_{g,t}\in[0,1]$: uncertainty from missing or stale evidence,
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $\Delta_{g,t}$: estimated cost of deferring a decision,
        \item $\lambda,\gamma$: policy-controlled trade-offs.
      \end{itemize}
    \end{column}
  \end{columns}

  \begin{block}{Action rule}
    Issue a local alert when $R_{g,t}\geq\tau$ and confidence exceeds the operational floor; otherwise request corroboration or escalate to a human operator.
  \end{block}
\end{frame}

% -----------------------------------------------------------------------------
% 10. Trade-offs
% -----------------------------------------------------------------------------
\begin{frame}{Resilience improves, but deployment still requires guardrails}
  \begin{block}{What the architecture handles well}
    It maintains a short decision path, exposes uncertainty, and records why each alert crossed the policy threshold.
  \end{block}

  \begin{alertblock}{What can still fail}
    Correlated sensor faults can create confident but wrong evidence; local models can also drift if reconnection is rare.
  \end{alertblock}

  \begin{exampleblock}{Operational guardrail}
    Combine plausibility checks, model-expiry limits, and a human-review path for ambiguous or unusually severe events.
  \end{exampleblock}
\end{frame}

\section{Deployment path}

% -----------------------------------------------------------------------------
% 11. Roadmap
% -----------------------------------------------------------------------------
\begin{frame}{A staged rollout earns trust before expanding autonomy}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 01}\par
      \vspace{0.15cm}
      \textbf{Shadow}\par
      \small Run beside the current process; compare recommendations without issuing alerts.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 02}\par
      \vspace{0.15cm}
      \textbf{Supervised}\par
      \small Let operators approve alerts while monitoring calibration and disagreement.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 03}\par
      \vspace{0.15cm}
      \textbf{Bounded autonomy}\par
      \small Automate only verified scenarios; escalate low-confidence cases by policy.
    \end{column}
  \end{columns}

  \vfill
  \begin{exampleblock}{Promotion gate}
    Advance only when latency, calibration, false-alert burden, and operator acceptance meet predefined thresholds together.
  \end{exampleblock}
\end{frame}

% -----------------------------------------------------------------------------
% 12. Close
% -----------------------------------------------------------------------------
\begin{frame}{Reliable alerts come from architecture, not accuracy alone}
  \begin{enumerate}
    \item Keep the critical decision path local and bounded.
    \item Represent missing evidence as uncertainty, not as silence.
    \item Use the cloud for coordination and learning, not mandatory availability.
    \item Expand autonomy gradually, with measurable promotion gates.
  \end{enumerate}

  \vfill
  \begin{block}{Takeaway}
    A resilient edge system does more than predict risk: it preserves a trustworthy decision process when infrastructure is under stress.
  \end{block}
