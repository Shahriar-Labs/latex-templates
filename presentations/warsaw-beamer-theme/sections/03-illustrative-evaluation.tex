\begin{frame}{The evaluation stresses the failure modes that matter operationally}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{enumerate}
        \item Replay 500 synthetic storm sequences across 24 virtual gateways.
        \item Vary sensor loss, packet delay, and cloud availability independently.
        \item Compare cloud-only, threshold-only, and edge-fusion policies.
        \item Record decision latency, alert stability, and calibration error.
      \end{enumerate}
    \end{column}
    \begin{column}{0.43\textwidth}

          \bottomrule
        \end{tabular}
      \end{block}
      \vspace{0.2cm}
      \scriptsize\color{AstraInk!75}

% -----------------------------------------------------------------------------
% 10. Trade-offs
% -----------------------------------------------------------------------------
\begin{frame}{Resilience improves, but deployment still requires guardrails}
  \begin{block}{What the architecture handles well}
    It maintains a short decision path, exposes uncertainty, and records why each alert crossed the policy threshold.
  \end{block}

  \begin{alertblock}{What can still fail}
    Correlated sensor faults can create confident but wrong evidence; local models can also drift if reconnection is rare.
  \end{alertblock}

  \begin{exampleblock}{Operational guardrail}
    Combine plausibility checks, model-expiry limits, and a human-review path for ambiguous or unusually severe events.
  \end{exampleblock}
\end{frame}
