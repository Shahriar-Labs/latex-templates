\begin{frame}{A staged rollout earns trust before expanding autonomy}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 01}\par
      \vspace{0.15cm}
      \textbf{Shadow}\par
      \small Run beside the current process; compare recommendations without issuing alerts.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 02}\par
      \vspace{0.15cm}
      \textbf{Supervised}\par
      \small Let operators approve alerts while monitoring calibration and disagreement.
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering
      {\usebeamercolor[fg]{structure}\Large\bfseries 03}\par
      \vspace{0.15cm}
      \textbf{Bounded autonomy}\par
      \small Automate only verified scenarios; escalate low-confidence cases by policy.
    \end{column}
  \end{columns}

  \vfill
  \begin{exampleblock}{Promotion gate}
    Advance only when latency, calibration, false-alert burden, and operator acceptance meet predefined thresholds together.
  \end{exampleblock}
\end{frame}

% -----------------------------------------------------------------------------
% 12. Close
% -----------------------------------------------------------------------------
\begin{frame}{Reliable alerts come from architecture, not accuracy alone}
  \begin{enumerate}
    \item Keep the critical decision path local and bounded.
    \item Represent missing evidence as uncertainty, not as silence.
    \item Use the cloud for coordination and learning, not mandatory availability.
    \item Expand autonomy gradually, with measurable promotion gates.
  \end{enumerate}

  \vfill
  \begin{block}{Takeaway}
    A resilient edge system does more than predict risk: it preserves a trustworthy decision process when infrastructure is under stress.
  \end{block}
