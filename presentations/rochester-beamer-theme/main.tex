\section{Introduction}

% Slide 3: Context
\begin{frame}{Introduction}{Decentralized Microgrid Decarbonization}
  \begin{itemize}[<+->]
    \item \textbf{Transition to Renewables}: Microgrids integrate distributed energy resources (DERs) like solar PV, wind turbines, and battery storage.
    \item \textbf{High Volatility}: Renewable generation fluctuates rapidly due to weather conditions, challenging grid stability.
    \item \textbf{Scalability Limits}: Centralized control systems struggle to manage thousands of active endpoints in real-time.
    \item \textbf{Autonomous Optimization}: Shift toward localized, edge-computing intelligence using multi-agent architectures.
  \end{itemize}
\end{frame}

% Slide 4: Problem Statement
\begin{frame}{Key Challenges}{Operational Hurdles in Modern Microgrids}
  \begin{alertblock}{Problem Statement}
    Dynamic power balancing in microgrids requires real-time coordination under stochastic supply/demand profiles without exposing private local consumer data.
  \end{alertblock}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Dimensionality}: Large action spaces (battery charging/discharging, load shedding, grid import/export).
    \item \textbf{Communication Bottlenecks}: High latency prevents centralized optimization models from converging.
    \item \textbf{Data Privacy}: Consumers are unwilling to share high-resolution load profiles with a central entity.
  \end{itemize}
\end{frame}

% Slide 6: Mathematical Formulation
\begin{frame}{Mathematical Formulation}{Objective Function and Constraints}
  \small
  We formulate the microgrid optimization as a Decentralized Partially Observable Markov Decision Process (Dec-POMDP).
  
  \begin{block}{Objective Function}
    Minimize total operational cost across all nodes $N$ over planning horizon $T$:
    \[
      \min_{\boldsymbol{a}} \mathbb{E} \left[ \sum_{t=1}^{T} \sum_{i=1}^{N} \left( C_i^{\text{grid}}(P_{i,t}^{\text{grid}}) + C_i^{\text{deg}}(P_{i,t}^{\text{bat}}) + \beta_i L_{i,t}^{\text{shed}} \right) \right]
    \]
  \end{block}
  
  \begin{block}{Key Operational Constraints}
    \begin{align*}
      P_{i,t}^{\text{gen}} + P_{i,t}^{\text{bat}} + P_{i,t}^{\text{grid}} &= P_{i,t}^{\text{load}} - L_{i,t}^{\text{shed}} \quad &\forall i, t \quad &\text{(Power Balance)} \\
      S_{i,t+1} &= S_{i,t} + \eta_i P_{i,t}^{\text{bat}} \Delta t \quad &\forall i, t \quad &\text{(Battery SoC Dynamics)}
    \end{align*}
  \end{block}
\end{frame}

% Slide 7: Algorithmic Approach
\begin{frame}{Algorithmic Approach}{Decentralized Actor, Centralized Critic}
  To solve the Dec-POMDP, we utilize a Multi-Agent Deep Deterministic Policy Gradient (MADDPG) variant:
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Centralized Training}: Critics have access to the joint state $\boldsymbol{s} = (s_1, \dots, s_N)$ and joint action $\boldsymbol{a} = (a_1, \dots, a_N)$ during training.
    \item \textbf{Decentralized Execution}: During deployment, each agent $\pi_{\theta_i}$ acts using only its local observation $o_i$.
    \item \textbf{Policy Update}:
    \[
      \nabla_{\theta_i} J(\theta_i) = \mathbb{E}_{\boldsymbol{s}, \boldsymbol{a} \sim \mathcal{D}} \left[ \nabla_{\theta_i} \pi_{\theta_i}(a_i | o_i) \nabla_{a_i} Q_i^{\boldsymbol{\psi}}(\boldsymbol{s}, a_1, \dots, a_N) \bigr|_{a_i = \pi_{\theta_i}(o_i)} \right]
    \]
  \end{itemize}
\end{frame}
