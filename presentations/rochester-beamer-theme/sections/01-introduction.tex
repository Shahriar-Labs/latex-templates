\section{Introduction}

% Slide 3: Context
\begin{frame}{Introduction}{Decentralized Microgrid Decarbonization}
  \begin{itemize}[<+->]
    \item \textbf{Transition to Renewables}: Microgrids integrate distributed energy resources (DERs) like solar PV, wind turbines, and battery storage.
    \item \textbf{High Volatility}: Renewable generation fluctuates rapidly due to weather conditions, challenging grid stability.
    \item \textbf{Scalability Limits}: Centralized control systems struggle to manage thousands of active endpoints in real-time.
    \item \textbf{Autonomous Optimization}: Shift toward localized, edge-computing intelligence using multi-agent architectures.
  \end{itemize}
\end{frame}

% Slide 4: Problem Statement
\begin{frame}{Key Challenges}{Operational Hurdles in Modern Microgrids}
  \begin{alertblock}{Problem Statement}
    Dynamic power balancing in microgrids requires real-time coordination under stochastic supply/demand profiles without exposing private local consumer data.
  \end{alertblock}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Dimensionality}: Large action spaces (battery charging/discharging, load shedding, grid import/export).
    \item \textbf{Communication Bottlenecks}: High latency prevents centralized optimization models from converging.
    \item \textbf{Data Privacy}: Consumers are unwilling to share high-resolution load profiles with a central entity.
  \end{itemize}
\end{frame}
