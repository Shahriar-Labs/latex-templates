%--- custom frame title: bold text + teal rule -------------------------
\setbeamertemplate{frametitle}{
  \vskip 0.35cm
  \hspace{0.6cm}\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle\\[2pt]
  \hspace{0.6cm}\textcolor{paletext}{\small\insertframesubtitle}
  \begin{tikzpicture}[remember picture,overlay]
    \draw[signalteal,line width=1.4pt] (0.6,-0.15) -- (12.9,-0.15);
  \end{tikzpicture}
  \vskip 0.15cm
}

%--- footline: five progress dots + slide number ------------------------
\defbeamertemplate*{footline}{progressdots}{
  \leavevmode
  \hbox{\begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (12.8,0.42);
    \foreach \i in {1,...,5}{
      \pgfmathsetmacro{\xpos}{5.6+(\i-1)*0.4}
      \ifnum\i<\thesecprog
        \fill[signalteal] (\xpos,0.21) circle (0.045);
      \else
        \ifnum\i=\thesecprog
          \fill[emberamber] (\xpos,0.21) circle (0.065);
        \else
          \draw[paletext,line width=0.5pt] (\xpos,0.21) circle (0.045);
        \fi
      \fi
    }
    \node[anchor=east,paletext,font=\scriptsize] at (12.55,0.21) {\insertframenumber/\inserttotalframenumber};
  \end{tikzpicture}}
}
