\section{Result I}

\begin{frame}
  \frametitle{Result I --- latency drops hardest where it hurt most}
  \framesubtitle{p50 / p95 inference latency, cache vs. no cache, 4k hot / 0.10 warm radius}
  \begin{table}
    \centering
    \small
    \begin{tabular}{@{}lrrrr@{}}
      \toprule
      \textbf{Device tier} & \textbf{p50 base} & \textbf{p50 cached} & \textbf{p95 base} & \textbf{p95 cached} \\
      \midrule
      Entry (\$90)   & 211\,ms & 68\,ms  & 340\,ms & 129\,ms \\
      Mid-range      & 104\,ms & 41\,ms  & 176\,ms & 74\,ms  \\
      Flagship       & 52\,ms  & 24\,ms  & 88\,ms  & 41\,ms  \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{2mm}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Reading the table}
        \begin{itemize}
          \item Entry-tier p50 falls \textbf{68\%}, closing most of the gap to flagship
          \item p95 improves even more --- the cache absorbs worst-case stalls, not just the average
          \item Accuracy moved by $<$0.3 NDCG points
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{tcolorbox}[colback=mist,colframe=emberamber,boxrule=0.6pt,arc=2mm,title={\color{white}Budget check},colbacktitle=emberamber]
        Entry-tier requests under the 120\,ms budget:\\
        \statword{32\%}{$\rightarrow$} \statword{91\%}{after caching}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}
