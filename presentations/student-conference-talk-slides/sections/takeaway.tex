\section{Takeaway}

\begin{frame}
  \frametitle{Takeaway}
  \framesubtitle{One sentence, three numbers, one open question}
  \begin{alertblock}{The sentence}
    A 4k-entry hot cache with a locality-sensitive warm tier cuts entry-tier p95 latency by \textbf{62\%} for \textbf{under 0.3 NDCG points} of ranking quality.
  \end{alertblock}
  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{tcolorbox}[colback=mist,colframe=signalteal,arc=2mm,boxrule=0.6pt,valign=center,halign=center]
        {\Large\bfseries\color{signalteal}68\%}\\[2pt]
        {\scriptsize entry-tier p50 drop}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{tcolorbox}[colback=mist,colframe=signalteal,arc=2mm,boxrule=0.6pt,valign=center,halign=center]
        {\Large\bfseries\color{signalteal}91\%}\\[2pt]
        {\scriptsize requests inside budget}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{tcolorbox}[colback=mist,colframe=signalteal,arc=2mm,boxrule=0.6pt,valign=center,halign=center]
        {\Large\bfseries\color{signalteal}77\%}\\[2pt]
        {\scriptsize steady-state hit rate}
      \end{tcolorbox}
    \end{column}
  \end{columns}
  \vspace{3mm}
  \begin{block}{Open question for the room}
    \small The warm tier's LSH radius was tuned per device tier by hand. Does an online bandit beat our fixed schedule on hardware we haven't tested?
  \end{block}
\end{frame}
