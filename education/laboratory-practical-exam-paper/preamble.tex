\newcommand{\pointmarks}[1]{%
    \marginnote{{%
        \hfill%
        \tikz[baseline=(char.base)]\node[%
            draw=secondary!60,%
            fill=secondary!5,%
            rounded corners=2pt,%
            inner sep=3.5pt,%
            text=secondary!80!primary,%
            font=\sffamily\small\bfseries,%
            line width=0.6pt%
        ] (char) {[#1]};%
    }}%
}

% Drawing Grid Box
\newcommand{\drawinggrid}[1]{%
    \begin{tcolorbox}[%
        colback=white,%
        colframe=primary!25,%
        width=\textwidth,%
        height=5.2cm,%
        arc=4pt,%
        boxrule=0.6pt,%
        title={#1},%
        coltitle=primary!90!black,%
        fonttitle=\sffamily\bfseries\small,%
        attach boxed title to top left={yshift=-2mm, xshift=4mm},%
        boxed title style={colback=primary!8, colframe=primary!25, boxrule=0.6pt}%
    ]%
    \begin{center}
    \begin{tikzpicture}
        \draw[step=0.5cm,primary!12,very thin] (0,0) grid (14.5, 3.8);
    \end{tikzpicture}
    \end{center}
    \end{tcolorbox}%
}
