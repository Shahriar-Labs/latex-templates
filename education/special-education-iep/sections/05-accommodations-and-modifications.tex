\section{Accommodations and Modifications}

\begin{minipage}{0.48\textwidth}
\begin{iepbox}
{\bfseries\color{IEPNavy}\Large Classroom Accommodations}\\
\begin{itemize}[leftmargin=*,nosep,label=\textcolor{IEPTeal}{\faCheckSquare}]
  \item Preferential seating near point of instruction
  \item Text-to-speech for all reading tasks exceeding 100 words
  \item Audio versions of core content-area textbooks
  \item Extended time (150\%) on reading and writing assessments
  \item Directions read aloud and clarified
  \item Graphic organizers provided for writing tasks
  \item Word prediction software (Co:Writer)
  \item Chunking of multi-step assignments
  \item Study guides and notes provided in advance
  \item Frequent comprehension checks during independent reading
\end{itemize}
\end{iepbox}
\end{minipage}%
\hfill
\begin{minipage}{0.48\textwidth}
\begin{iepbox}
{\bfseries\color{IEPNavy}\Large Assessment Accommodations}\\
\begin{itemize}[leftmargin=*,nosep,label=\textcolor{IEPTeal}{\faCheckSquare}]
  \item Small-group setting for all state and district assessments
  \item Text-to-speech for reading passages (IAR and NWEA MAP)
  \item Extended time (double time) for reading and writing sections
  \item Scribe for extended written responses on IAR
  \item Read-aloud for math word problems
  \item Frequent supervised breaks (every 30 minutes)
  \item Human reader for science and social studies assessments
\end{itemize}
\vspace{4pt}
{\bfseries\color{IEPNavy} Modifications}\\[2pt]
\begin{itemize}[leftmargin=*,nosep,label=\textcolor{IEPGold}{$\bullet$}]
  \item Reduced written response length (50\% of grade-level expectation)
  \item Spelling requirements waived on content-area writing
\end{itemize}
\end{iepbox}
\end{minipage}
