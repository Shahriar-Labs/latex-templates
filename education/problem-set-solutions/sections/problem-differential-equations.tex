\section{Problem 1 --- Second-Order Linear ODE (10 points)}

\begin{problembox}{Problem Statement}
A damped oscillator is modelled by
\[
  y'' + 4y' + 13y = 0, \qquad y(0) = 2, \qquad y'(0) = -1.
\]
Find $y(t)$ and classify the damping regime. \cite{boyce2017elementary}
\end{problembox}

\begin{steps}

\item \textbf{Set up the characteristic equation.} Substituting
$y = e^{rt}$ gives
\[
  r^2 + 4r + 13 = 0.
\]

\rubric{2}{Correct characteristic equation from the ODE coefficients.}

\item \textbf{Solve for the roots.} By the quadratic formula,
\[
  r = \frac{-4 \pm \sqrt{16 - 52}}{2} = \frac{-4 \pm \sqrt{-36}}{2}
    = -2 \pm 3i.
\]
The discriminant is negative, so the roots are a complex-conjugate pair
$r = -2 \pm 3i$, which places the system in the \emph{underdamped} regime.

\rubric{3}{Correct roots \emph{and} explicit statement of the damping
regime (underdamped, since the discriminant $< 0$).}

\item \textbf{Write the general solution.} For roots $\alpha \pm i\beta$
with $\alpha = -2$, $\beta = 3$, the general real solution is
\[
  y(t) = e^{-2t}\big(C_1 \cos 3t + C_2 \sin 3t\big).
\]

\item \textbf{Apply the initial conditions.} At $t = 0$:
\[
  y(0) = C_1 = 2.
\]
Differentiating,
\[
  y'(t) = e^{-2t}\big[(-2C_1 + 3C_2)\cos 3t + (-2C_2 - 3C_1)\sin 3t\big],
\]
so $y'(0) = -2C_1 + 3C_2 = -1$. With $C_1 = 2$ this gives
$3C_2 = -1 + 4 = 3$, hence $C_2 = 1$.

\rubric{3}{Correct product rule on $y'(t)$ --- the most common slip is
dropping the $-2$ factor from the envelope term when differentiating.}

\end{steps}

\begin{finalbox}
\[
  y(t) = e^{-2t}\big(2\cos 3t + \sin 3t\big), \qquad \text{underdamped, }
  \omega_d = 3 \text{ rad/s}.
\]
\end{finalbox}

\rubric{2}{Final expression assembled correctly with both constants
substituted; units/label for $\omega_d$ not required but rewarded.}
