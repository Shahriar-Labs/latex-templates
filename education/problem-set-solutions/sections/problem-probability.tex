\section{Problem 3 --- Conditional Probability (10 points)}

\begin{problembox}{Problem Statement}
A component in a Nimbus sensor array is supplied by one of two vendors:
Vendor A supplies 60\% of units and 3\% of those are defective; Vendor B
supplies 40\% of units and 5\% of those are defective. A unit is drawn at
random and found to be defective. What is the probability it came from
Vendor B?
\end{problembox}

\begin{steps}

\item \textbf{Define events and given probabilities.} Let $B_A$, $B_B$ be
the events ``unit from Vendor A/B'' and $D$ the event ``unit defective''.
\begin{align*}
  P(B_A) &= 0.60, & P(B_B) &= 0.40, \\
  P(D \mid B_A) &= 0.03, & P(D \mid B_B) &= 0.05.
\end{align*}

\rubric{2}{Events correctly labelled and conditional probabilities matched
to the right vendor.}

\item \textbf{Compute the total probability of a defect.} By the law of
total probability,
\begin{align*}
  P(D) &= P(D \mid B_A)P(B_A) + P(D \mid B_B)P(B_B) \\
       &= (0.03)(0.60) + (0.05)(0.40) = 0.018 + 0.020 = 0.038.
\end{align*}

\rubric{3}{Both terms present and correctly weighted; a common error is
using $P(B_A \mid D)$ in place of the prior $P(B_A)$.}

\item \textbf{Apply Bayes' theorem.}
\[
  P(B_B \mid D) = \frac{P(D \mid B_B)\,P(B_B)}{P(D)}
    = \frac{(0.05)(0.40)}{0.038} = \frac{0.020}{0.038}.
\]

\rubric{3}{Correct Bayes' theorem set-up with the Step 2 result used as
the denominator.}

\item \textbf{Simplify.}
\[
  P(B_B \mid D) = \frac{0.020}{0.038} \approx 0.526.
\]

\end{steps}

\begin{finalbox}
\[
  P(\text{Vendor B} \mid \text{defective}) = \frac{0.020}{0.038}
    \approx 0.526 \;\; (52.6\%).
\]
\end{finalbox}

\rubric{2}{Final probability reported to at least three significant
figures with correct interpretation (posterior, not prior).}
