\section{Problem 2 --- Eigenvalues and Diagonalization (10 points)}

\begin{problembox}{Problem Statement}
Let
\[
  A = \begin{pmatrix} 4 & 2 \\{} 1 & 3 \end{pmatrix}.
\]
Find the eigenvalues and eigenvectors of $A$, and use them to write
$A = PDP^{-1}$.
\end{problembox}

\begin{steps}

\item \textbf{Characteristic polynomial.} Compute
\[
  \det(A - \lambda I) = \det\begin{pmatrix} 4-\lambda & 2 \\{} 1 & 3-\lambda
  \end{pmatrix} = (4-\lambda)(3-\lambda) - 2 = \lambda^2 - 7\lambda + 10.
\]

\rubric{2}{Determinant expanded correctly; sign error on the off-diagonal
product is the most frequent deduction here.}

\item \textbf{Solve for eigenvalues.} Factoring,
\[
  \lambda^2 - 7\lambda + 10 = (\lambda - 5)(\lambda - 2) = 0
  \;\Longrightarrow\; \lambda_1 = 5,\ \lambda_2 = 2.
\]

\item \textbf{Eigenvector for $\lambda_1 = 5$.} Solve $(A - 5I)v = 0$:
\[
  \begin{pmatrix} -1 & 2 \\{} 1 & -2 \end{pmatrix}
  \begin{pmatrix} v_1 \\{} v_2 \end{pmatrix} = \begin{pmatrix} 0 \\{} 0
  \end{pmatrix} \;\Longrightarrow\; v_1 = 2v_2.
\]
Taking $v_2 = 1$ gives $v^{(1)} = (2, 1)^{\mathsf T}$.

\rubric{3}{Row reduction and free-variable choice; either eigenvector
scaling is accepted as long as the direction is correct.}

\item \textbf{Eigenvector for $\lambda_2 = 2$.} Solve $(A - 2I)v = 0$:
\[
  \begin{pmatrix} 2 & 2 \\{} 1 & 1 \end{pmatrix}
  \begin{pmatrix} v_1 \\{} v_2 \end{pmatrix} = \begin{pmatrix} 0 \\{} 0
  \end{pmatrix} \;\Longrightarrow\; v_1 = -v_2.
\]
Taking $v_2 = 1$ gives $v^{(2)} = (-1, 1)^{\mathsf T}$.

\rubric{3}{Same criterion as Step 3, applied to the second eigenspace.}

\end{steps}

  \draw[->,inkblue,thick] (0,0) -- (2,1) node[above right,font=\small]
    {$v^{(1)},\ \lambda_1=5$};
  \draw[->,accentgold,thick] (0,0) -- (-1,1) node[above left,font=\small]
    {$v^{(2)},\ \lambda_2=2$};
  \node[circle,fill=slate,inner sep=1.1pt] at (0,0) {};
\end{tikzpicture}
\end{center}

\begin{finalbox}
\[
  P = \begin{pmatrix} 2 & -1 \\{} 1 & 1 \end{pmatrix}, \qquad
  D = \begin{pmatrix} 5 & 0 \\{} 0 & 2 \end{pmatrix}, \qquad
  A = PDP^{-1}.
\]
\end{finalbox}

\rubric{2}{$P$ built from eigenvectors as columns \emph{in the same order}
as the eigenvalues appear in $D$ --- a very common source of lost marks.}
