\section{Problem 4 --- Constrained Optimization (10 points)}

\begin{problembox}{Problem Statement}
The Apex fabrication line uses a rectangular enclosure of fixed perimeter
$P = 40$ m for a cooling unit. Find the dimensions that maximize the
enclosed area, and confirm the result is a maximum.
\end{problembox}

\begin{steps}

\item \textbf{Set up the objective and constraint.} Let the sides be $x$
and $y$. The area to maximize is $A = xy$, subject to
\[
  2x + 2y = 40 \quad \Longrightarrow \quad y = 20 - x.
\]

\rubric{2}{Constraint correctly solved for one variable and substituted
in the next step (not left as a Lagrange multiplier problem, which is
also accepted but not required here).}

\item \textbf{Reduce to a single-variable function.} Substitute $y$ into
$A$:
\[
  A(x) = x(20 - x) = 20x - x^2, \qquad 0 < x < 20.
\]

\item \textbf{Find critical points.} Differentiate and set to zero:
\[
  A'(x) = 20 - 2x = 0 \quad \Longrightarrow \quad x = 10.
\]

\rubric{3}{Derivative taken correctly and set to zero; domain restriction
$0 < x < 20$ should be stated so the endpoints are ruled out.}

\item \textbf{Confirm it is a maximum.} The second derivative is
\[
  A''(x) = -2 < 0 \quad \text{for all } x,
\]
so $A(x)$ is concave down everywhere and $x = 10$ is a global maximum on
the interval. Then $y = 20 - 10 = 10$, so the optimal enclosure is a
square.

\rubric{3}{Second-derivative test (or a sign chart for $A'$) explicitly
invoked --- a bare ``this is the max'' with no justification loses this
mark.}

\end{steps}

\begin{center}
\begin{tikzpicture}[scale=0.9]
  \draw[->,slate,thin] (-0.3,0) -- (5.6,0) node[right,font=\small] {$x$};
  \draw[->,slate,thin] (0,-0.3) -- (0,3.6) node[above,font=\small] {$A(x)$};
  \draw[inkblue,thick,domain=0:20,samples=60,smooth,variable=\t]
    plot ({\t/4},{ (20*\t - \t*\t)/28 });
  \draw[dashed,accentgold] (2.5,0) -- (2.5,{(20*2.5-2.5*2.5)/28})
    -- (0,{(20*2.5-2.5*2.5)/28});
  \node[circle,fill=accentgold,inner sep=1.3pt] at
    (2.5,{(20*2.5-2.5*2.5)/28}) {};
  \node[below,font=\footnotesize] at (2.5,0) {$x=10$};
  \node[left,font=\footnotesize] at (0,{(20*2.5-2.5*2.5)/28}) {$A=100$};
\end{tikzpicture}
\end{center}

\begin{finalbox}
\[
  x = y = 10 \text{ m}, \qquad A_{\max} = 100 \text{ m}^2 \quad
  \text{(the optimal rectangle is a square)}.
\]
\end{finalbox}

\rubric{2}{Both dimensions and the maximum area value reported, with the
observation that the optimal shape is a square.}
