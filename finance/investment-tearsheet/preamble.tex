% === Custom commands ===
% Label over value. The stack lives in its own tabular because a bare \\ at the
% top level of an X (paragraph) column ends the TABLE ROW rather than breaking
% the line — so every metric used to terminate its row early and the values
% ended up displayed against the wrong labels.
\newcommand{\metriclabel}[2]{%
  \begin{tabular}[t]{@{}l@{}}
    {\footnotesize\color{slate}#1}\\[1pt]
    {\normalsize\color{navy}\bfseries#2}
  \end{tabular}%
}

\newcommand{\risktile}[2]{%
  \fcolorbox{mdgray}{ltgray}{%
    \begin{minipage}{0.148\textwidth}
      \centering\vspace{3pt}
      {\footnotesize\color{slate}#1\vspace{2pt}\par}
      {\normalsize\color{navy}\bfseries#2\vspace{3pt}\par}
    \end{minipage}%
  }%
}

\newcommand{\quartile}[1]{%
  \ifnum#1=1
    \textcolor{posgreen}{\bfseries Q1}%
  \else\ifnum#1=2
    \textcolor{warning}{\bfseries Q2}%
  \else\ifnum#1=3
    \textcolor{accent}{\bfseries Q3}%
  \else
    \textcolor{neggred}{\bfseries Q4}%
  \fi\fi\fi
}
