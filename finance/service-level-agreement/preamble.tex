%% Title block helpers
\newcommand{\fieldlabel}[1]{%
  {\color{MidGrey}\footnotesize\bfseries\MakeUppercase{#1}}}

\newcommand{\fieldvalue}[1]{%
  {\color{DeepNavy}\small #1}}

%% Clause numbering
\newcounter{clausectr}
\renewcommand{\theclausectr}{\arabic{clausectr}}
\newcommand{\clause}[1]{%
  \stepcounter{clausectr}%
  \par\noindent\textbf{\color{SignalTeal}\theclausectr.\quad #1}\par}

%% Inline highlight box (wraps to full width; use for multi-line callouts)
\newcommand{\notebox}[1]{%
  \begin{tcolorbox}[
    enhanced, breakable,
    colback=PaleSlate, colframe=SignalTeal,
    boxrule=0pt, leftrule=3pt,
    arc=2pt, outer arc=2pt,
    top=6pt, bottom=6pt, left=8pt, right=8pt
  ]
  \small #1
  \end{tcolorbox}}

%% Amber warning box (exclusions, caps)
\newcommand{\warnbox}[1]{%
  \begin{tcolorbox}[
    enhanced, breakable,
    colback=PaleSlate, colframe=AmberFlag,
    boxrule=0pt, leftrule=3pt,
    arc=2pt, outer arc=2pt,
    top=6pt, bottom=6pt, left=8pt, right=8pt
  ]
  \small #1
  \end{tcolorbox}}

%% Definition box (used for key terms)

  enhanced, breakable,

  title={\faBook\quad Definitions},
  fonttitle={\bfseries\color{DeepNavy}\small},
  coltitle=DeepNavy,
  attach boxed title to top left={yshift=-2mm, xshift=8mm},
  boxed title style={colback=white, colframe=RuleGrey, boxrule=0.5pt, arc=3pt},
  top=10pt, bottom=8pt, left=10pt, right=10pt
}

%% Tier / metric box

  colback=white,

}
