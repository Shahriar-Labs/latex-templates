% Custom Command for Risk Score Cell
\newcommand{\riskRating}[2]{%
  \pgfmathtruncatemacro{\rating}{#1 * #2}%
  \ifnum\rating>15\relax
    \cellcolor{riskRedBg}\textcolor{riskRed}{\bfseries\rating}%
  \else\ifnum\rating>9\relax
    \cellcolor{riskOrangeBg}\textcolor{riskOrange}{\bfseries\rating}%
  \else\ifnum\rating>4\relax
    \cellcolor{riskYellowBg}\textcolor{riskYellow}{\bfseries\rating}%
  \else
    \cellcolor{riskGreenBg}\textcolor{riskGreen}{\bfseries\rating}%
  \fi\fi\fi
}
