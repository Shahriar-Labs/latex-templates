\section{Recommended supporting evidence}
\begin{itemize}
  \item signed agreements and amendments in force during the covered period;
  \item organization charts, job descriptions, and delegated-authority matrices;
  \item board papers and evidence showing who made significant risk decisions;
  \item invoices, purchase orders, delivery terms, and counterparty confirmations;
  \item service catalogues, deliverables, time records, allocation drivers, and
    benefits-test evidence;
  \item trademark registrations, licence schedules, and royalty-base reports;
  \item loan approval papers, credit analysis, cash forecasts, and interest
    calculations;
  \item benchmark search strategies, exports, screening notes, and comparable
    company financials; and
  \item signed financial statements, trial balances, segment accounts, and
    reconciliation workbooks.
\end{itemize}
