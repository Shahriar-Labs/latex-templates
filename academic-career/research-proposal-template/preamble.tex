\usepackage{amsmath,amssymb}

% ── Encoding & Fonts ──────────────────────────────────────────
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}

% ── Page geometry ─────────────────────────────────────────────
\usepackage[margin=2.2cm,top=2cm,bottom=2.2cm]{geometry}

% ── Colors ────────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{accent}{HTML}{1B5E7B}       % deep teal-blue
\definecolor{accentlight}{HTML}{D6EAF5}  % pale accent tint
\definecolor{dark}{HTML}{2D2D2D}
\definecolor{midgray}{HTML}{777777}
\definecolor{lightgray}{HTML}{EEEEEE}
\definecolor{softbg}{HTML}{FAFBFC}

% ── Tables ────────────────────────────────────────────────────
\usepackage{tabularx}
\usepackage{booktabs}
\usepackage{array}
\usepackage{colortbl}
\raggedbottom

% ── Graphics & TikZ ───────────────────────────────────────────
\usepackage{tikz}
\usepackage{tcolorbox}
\tcbuselibrary{skins}

% ── Section formatting ────────────────────────────────────────
\usepackage{titlesec}
\titleformat{\section}
  {\Large\bfseries\color{accent}}
  {\thesection}{1em}{}[\vspace{-0.5em}\rule{\textwidth}{0.4pt}\vspace{0.5em}]
\titleformat{\subsection}
  {\large\bfseries\color{dark}}
  {\thesubsection}{1em}{}

% ── Lists ─────────────────────────────────────────────────────
\usepackage{enumitem}
\setlist{nosep,leftmargin=*,itemsep=2pt,topsep=4pt}

% ── Icons ─────────────────────────────────────────────────────
\usepackage{fontawesome5}

% ── Headers / footers ────────────────────────────────────────
\usepackage{fancyhdr}
\usepackage{lastpage}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\color{midgray}\footnotesize\sffamily Page \thepage\ of \pageref{LastPage}}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% ── Hyperlinks ────────────────────────────────────────────────
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=accent,
  urlcolor=accent!80!dark,
  citecolor=accent,
  pdfborder={0 0 0},
}

% ── Paragraph setup ───────────────────────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{4pt plus 1pt minus 1pt}

  colframe=accent,
  arc=0mm,
  boxrule=0pt,
  leftrule=3.5pt,
  top=10pt,
  bottom=10pt,
  left=14pt,
  right=10pt,
  before skip=10pt,
  after skip=16pt,
}

% --- Contact-info key/value row ---
\newcommand{\contactrow}[2]{%
  \raisebox{-0.5pt}{\textcolor{accent}{#1}}\hspace{4pt}{\footnotesize #2}%
}

% --- Budget table row ---
\newcommand{\budgetrow}[4]{%
  #1 & #2 & #3 & #4 \\[4pt]
}

% --- Gantt bar dimensions ---
\newlength{\ganttunit}
\setlength{\ganttunit}{5.8mm}
