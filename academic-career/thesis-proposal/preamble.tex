% Packages, colours and custom commands for this template.
% Edited and compiled online at https://letx.app

% Page geometry
\usepackage[margin=1.0in]{geometry}

% Typography & Fonts
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella} % TeX Gyre Pagella (Palatino clone) for elegant academic serif
\usepackage{microtype} % Microtypography styling

% Colors
\usepackage{xcolor}
\definecolor{accent}{HTML}{0D5C75} % Deep Teal Accent
\definecolor{neutraldark}{HTML}{2B2D42} % Slate Dark for headings/text
\definecolor{neutrallight}{HTML}{F4F4F9} % Soft grey-blue for tables/boxes

% Table formatting
\usepackage{booktabs}
\usepackage{tabularx}

% Headings customization
\usepackage{titlesec}
\titleformat{\section}
  {\color{accent}\normalfont\large\bfseries}
  {\thesection}{1em}{}
  [{\color{accent}\titlerule[0.8pt]}]
\titleformat{\subsection}
  {\color{accent}\normalfont\normalsize\bfseries}
  {\thesubsection}{1em}{}

% Lists & spacing
\usepackage{enumitem}
\setlist[itemize]{leftmargin=*,noitemsep,topsep=2pt}
\setlist[enumerate]{leftmargin=*,noitemsep,topsep=2pt}

% TikZ for beautiful diagrams
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta}

% Bibliography setup
\usepackage{filecontents}


\usepackage[backend=biber,style=numeric-comp,sorting=none]{biblatex}
\addbibresource{references.bib}

% Hyperref (loaded last)
\usepackage[hidelinks]{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=accent,
    citecolor=accent,
    urlcolor=accent
}
