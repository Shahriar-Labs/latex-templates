\dossierpart{III}{Service}{Departmental, university, and professional contributions}

\section{Departmental Service}
\begin{itemize}
  \item Member, Graduate Admissions Committee, Department of Computer Science (2021--2024); Chair (2023--2024), overseeing review of approximately 340 applications per cycle.
  \item Member, Faculty Search Committee for the Machine Learning Systems cluster hire (2022--2023), resulting in two successful offers.
  \item Departmental Colloquium Coordinator (2021--present), scheduling and hosting 6--8 external speakers per academic year.
  \item Co-author of the department's revised graduate qualifying-exam syllabus (see Part II).
\end{itemize}

\section{University Service}
\begin{itemize}

  \item Faculty mentor, Meridian University Women in Computing chapter (2021--present).
  \item Panelist, Meridian University Postdoctoral Research Symposium (2022, 2024).
\end{itemize}

\section{Professional Service}
\begin{itemize}
  \item Program Committee member: International Conference on Machine Learning (ICML), 2022--2026; Conference on Neural Information Processing Systems (NeurIPS), 2023--2025; International Conference on Learning Representations (ICLR), 2024--2026.
  \item Reviewer: \emph{Journal of Machine Learning Research}, \emph{IEEE Transactions on Computers}, \emph{ACM Transactions on Sensor Networks} (2021--present), averaging 12 manuscript reviews per year.
  \item Guest co-editor, special issue on Federated and Decentralized Learning, \emph{ACM Transactions on Interactive Intelligent Systems} (2025), with Dr.\ Sarah Jenkins.
  \item Session chair, Workshop on Machine Learning for Health (ML4H), 2022 and 2023.
\end{itemize}

\section{Outreach and Broader Impact}
\begin{itemize}
  \item Founded and continues to run \emph{Models That Fail}, a public lecture series at Meridian University introducing distribution-shift and robustness concepts to non-specialist audiences; nine sessions since 2022 with a combined attendance of approximately 640.
  \item Volunteer instructor, Meridian regional high-school computing outreach program (2021--present), leading annual one-day workshops on introductory machine learning for underrepresented students.
  \item Open-source maintainer of the \texttt{robustfed} library used to implement the aggregation methods described in \citep{rostova2022robust} and \citep{rostova2021federated}, with contributions from four external research groups.
\end{itemize}
