% Packages, colours and custom commands for this template.

% --- Geometry & Margins ---
\usepackage[margin=0.65in, top=0.65in, bottom=0.65in]{geometry}

% --- Typography ---
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella} % Beautiful Palatino-based serif font
\usepackage{microtype} % Improved spacing and justification
\usepackage{setspace}
\setstretch{1.00} % Clean, compact line spacing

% --- Colors ---
\usepackage{xcolor}
\definecolor{accent}{HTML}{1B365D} % Deep Navy Accent
\definecolor{highlight}{HTML}{2E4A62} % Muted Slate Blue
\definecolor{textgray}{HTML}{4A4A4A} % Muted text color for sub-info

% --- Paragraph Spacing ---
\setlength{\parindent}{0pt}
\setlength{\parskip}{4.5pt plus 0.5pt minus 0.5pt}

% --- Section Styling ---
\usepackage{titlesec}
\titleformat{\section}
  {\color{accent}\normalfont\Large\bfseries}
  {\thesection}{1em}{}
  [\vspace{-0.4em}\color{accent}\titlerule]

\titleformat{\subsection}
  {\color{accent}\normalfont\large\bfseries}
  {\thesubsection}{0.8em}{}

\titlespacing*{\section}{0pt}{0.7em}{0.35em}
\titlespacing*{\subsection}{0pt}{0.5em}{0.25em}

% --- Callout Box for Executive Summary ---
\usepackage[most]{tcolorbox}
\newtcolorbox{summarybox}{
  colback=accent!5,
  colframe=accent,
  arc=0mm,
  outer arc=0mm,
  top=2.2mm,
  bottom=2.2mm,
  left=3.5mm,
  right=3.5mm,
  boxrule=0pt,
  leftrule=3.5pt,
  before=\par\noindent\vspace{0.3em},
  after=\vspace{0.3em}
}

% --- Lists and Spacing ---
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.5em, itemsep=0.15em, topsep=0.1em, parsep=0pt}
\setlist[enumerate]{leftmargin=1.5em, itemsep=0.15em, topsep=0.1em, parsep=0pt}

% --- Tables ---
\usepackage{booktabs}
\usepackage{tabularx}

% --- Hyperlinks ---
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=accent,
  citecolor=accent,
  urlcolor=accent
}
