% --- colour palette --------------------------------------------------
\definecolor{inkblue}{HTML}{1B2A4A}
\definecolor{claret}{HTML}{7A1F2B}
\definecolor{parchment}{HTML}{F5F0E6}
\definecolor{slategray}{HTML}{55606E}

% --- notes apparatus ---------------------------------------------------
\let\footnote=\endnote
\renewcommand{\notesname}{Notes}

% --- chapter / part styling --------------------------------------------
\titleformat{\part}[display]
  {\centering\Huge\bfseries\color{inkblue}}
  {\fontsize{14pt}{16pt}\selectfont\textcolor{claret}{PART \Roman{part}}}
  {8pt}
  {\vspace{6pt}}
\titlespacing*{\part}{0pt}{0pt}{40pt}

\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{inkblue}}
  {\fontsize{11pt}{13pt}\selectfont\textcolor{slategray}{CHAPTER \thechapter}}
  {6pt}
  {\Huge}
\titlespacing*{\chapter}{0pt}{20pt}{28pt}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{inkblue}}{\thesection}{0.6em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{slategray}}{\thesubsection}{0.6em}{}

% --- running headers ----------------------------------------------------
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}

\fancyhead[LE]{\small\scshape\color{slategray}\leftmark}
\fancyhead[RO]{\small\scshape\color{slategray}\rightmark}
\fancyfoot[C]{\small\thepage}
