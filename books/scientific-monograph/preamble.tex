% Side note command
\newcommand{\sidenote}[1]{%
  \marginnote{%
    \vspace{2pt}%
    \footnotesize\raggedright\color{accent!75!black}%
    {\color{accent!30}\rule{\marginparwidth}{0.4pt}}\\[2pt]%
    #1%
  }%
}

% Index term convenience
\newcommand{\idxterm}[1]{#1\index{#1}}
\newcommand{\idxsub}[2]{#1\index{#2!#1}}

% Metadata block (DOI, ISBN, etc.)
\newcommand{\metadatablock}[5]{%
  \begin{tcolorbox}[
    enhanced,
    colback=accent!6,
    colframe=accent!25,
    sharp corners,
    boxrule=0.5pt,
    width=\textwidth,
    before skip=24pt,
    after skip=0pt
  ]
  \footnotesize\sffamily
  \begin{tabularx}{\linewidth}{@{}>{\color{accent}\bfseries}l@{\hspace{6mm}}>{\raggedright\arraybackslash}X@{}}
  DOI       & #1 \\
  ISBN      & #2 \\
  LCCN      & #3 \\
  Subjects  & #4 \\
  Edition   & #5 \\
  \end{tabularx}
  \end{tcolorbox}
}

% Color plate environment
\newcounter{plate}
\newtcolorbox{colorplate}[1]{
  enhanced,
  colback=#1!10,
  colframe=#1!50,
  sharp corners,
  boxrule=0.6pt,
  code={\refstepcounter{plate}},
  title={Plate \theplate},
  fonttitle=\color{#1!80!black}\bfseries,
  before skip=14pt,
  after skip=14pt,
  attach boxed title to top left={xshift=6mm,yshift=-3mm},
  boxed title style={colback=bg,colframe=#1!50,boxrule=0.5pt,sharp corners}
}
