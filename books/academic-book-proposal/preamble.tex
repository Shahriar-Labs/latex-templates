% ── Encoding & font ──────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella}          % Palatino-style serif — classic, scholarly
\usepackage{microtype}

% ── Page geometry ────────────────────────────────────────────
\usepackage[
  top=28mm, bottom=28mm,
  left=32mm, right=32mm
]{geometry}

% ── Colour ───────────────────────────────────────────────────
\usepackage[table]{xcolor}
\definecolor{accent}{HTML}{2D5F8A}    % slate blue — scholarly accent
\definecolor{rule}{HTML}{B0C4D8}      % pale steel — thin rules
\definecolor{lightbg}{HTML}{F4F7FA}   % near-white tint for table header

% ── Spacing & paragraph ──────────────────────────────────────
\usepackage{parskip}
\setlength{\parindent}{0pt}
\setlength{\parskip}{5pt}

% ── Section headings ─────────────────────────────────────────
\usepackage{titlesec}
\titleformat{\section}
  {\color{accent}\normalfont\large\bfseries}
  {}{0pt}{}[{\color{rule}\titlerule[0.6pt]}]
\titlespacing*{\section}{0pt}{14pt}{6pt}

\titleformat{\subsection}
  {\color{accent!80!black}\normalfont\normalsize\bfseries\itshape}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{8pt}{3pt}

% ── Tables ───────────────────────────────────────────────────
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{array}
\renewcommand{\arraystretch}{1.35}

% ── Lists ────────────────────────────────────────────────────
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.4em, itemsep=2pt, topsep=3pt}

% ── Hyperref (last) ──────────────────────────────────────────
\usepackage[
  colorlinks=true,
  urlcolor=accent,
  linkcolor=accent,
  citecolor=accent,
  pdftitle={Academic Book Proposal: The Architecture of Uncertainty},
  pdfauthor={Eleanor M. Hartwell}
]{hyperref}

% ── Helper macros ────────────────────────────────────────────
\newcommand{\proposalrule}{%
  {\color{rule}\rule{\linewidth}{0.6pt}}}

\newcommand{\kv}[2]{%
  \begin{tabularx}{\linewidth}{%
    @{}>{\color{accent}\bfseries}l@{\hspace{8mm}}>{\raggedright\arraybackslash}X@{}}
    #1 & #2
  \end{tabularx}}

% chapter entry: \chentry{num}{title}{summary}
\newcommand{\chentry}[3]{%
  \noindent
  {\color{accent}\bfseries Chapter #1.\enspace}{\bfseries #2}\\
  {\small #3}\par\vspace{3pt}}
