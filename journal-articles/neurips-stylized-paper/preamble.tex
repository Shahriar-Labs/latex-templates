\newcommand{\figplaceholder}[3][accentlt]{%
  % #1=fill colour, #2=width, #3=height
  \begin{tikzpicture}
    \fill[#1] (0,0) rectangle (#2,#3);
    \draw[accent, line width=0.8pt] (0,0) rectangle (#2,#3);
    \node[font=\small\color{neutralmid}] at ({#2/2},{#3/2})
      {[Figure placeholder]};
  \end{tikzpicture}}
