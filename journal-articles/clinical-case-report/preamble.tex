% ---------- Page geometry ----------
\usepackage[
  top=22mm, bottom=24mm,
  left=28mm, right=28mm,
  headheight=14pt
]{geometry}

% ---------- Font ----------
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}

% ---------- Colour palette ----------
\usepackage{xcolor}
\definecolor{accent}{HTML}{0B6E8F}   % clinical teal-blue
\definecolor{accentlt}{HTML}{E4F3F8} % pale blue tint
\definecolor{neutral}{HTML}{3D3D3D}  % near-black text
\definecolor{rule}{HTML}{BBDDE8}     % light rule

% ---------- Typography helpers ----------
\usepackage{microtype}
\usepackage{ragged2e}
\usepackage{setspace}
\setstretch{1.15}

% ---------- Spacing ----------
\setlength{\parindent}{0pt}
\setlength{\parskip}{5pt}

% ---------- Section styling ----------
\usepackage{titlesec}
\titleformat{\section}
  {\normalfont\bfseries\color{accent}\large}
  {}{0em}{}
  [\vspace{1pt}\textcolor{rule}{\hrule height 0.6pt}\vspace{2pt}]
\titlespacing*{\section}{0pt}{10pt}{4pt}

\titleformat{\subsection}
  {\normalfont\bfseries\color{neutral}}
  {}{0em}{}
\titlespacing*{\subsection}{0pt}{7pt}{2pt}

% ---------- Tables ----------
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{array}
\renewcommand{\arraystretch}{1.25}

% ---------- Lists ----------
\usepackage{enumitem}
\setlist{itemsep=2pt, topsep=3pt, leftmargin=14pt}

% ---------- Header / footer ----------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{rule}\leaders\hrule height \headrulewidth\hfill}}
\fancyhead[L]{\footnotesize\color{accent}\textit{Case Report} \textbullet\ \textit{Journal of Internal Medicine Case Reports}}
\fancyhead[R]{\footnotesize\color{neutral}\thepage}
\fancypagestyle{plain}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\footnotesize\color{neutral}\thepage}
}

% ---------- Hyperlinks ----------
\usepackage[
  colorlinks=true,
  linkcolor=accent,
  citecolor=accent,
  urlcolor=accent,
  pdfauthor={Smith et al.},
  pdftitle={Acute Aortic Dissection Presenting as Atypical Chest Pain: A Case Report}
]{hyperref}

% ---------- Misc ----------
\usepackage{graphicx}
\usepackage{amsmath}

% ---------- tcolorbox for abstract ----------
\usepackage{tcolorbox}
\tcbuselibrary{skins, breakable}

\newtcolorbox{abstractbox}{
  enhanced,
  breakable,
  colback=accentlt,
  colframe=accent,
  arc=3pt,
  boxrule=0.8pt,
  left=10pt, right=10pt,
  top=8pt, bottom=8pt,
  fonttitle=\bfseries\color{accent},
  title=Abstract,
}

% ---------- Convenience macros ----------
\newcommand{\keyword}[1]{\textbf{\textcolor{accent}{#1}}}
\newcommand{\etal}{\textit{et al}.}
