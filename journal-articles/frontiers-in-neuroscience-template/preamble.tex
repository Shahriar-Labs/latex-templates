% Draw the journal running head at shipout so it remains identical on ordinary,
% float-heavy, title, and bibliography pages.
\AddToHook{shipout/foreground}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=west,font=\sffamily\footnotesize,text=frontgray]
      at ([xshift=24mm,yshift=-13.5mm]current page.north west) {Chen et al.};
    \node[anchor=east,font=\sffamily\footnotesize,text=frontgray]
      at ([xshift=-24mm,yshift=-13.5mm]current page.north east) {Closed-loop sleep stimulation};
    \draw[frontcyan,line width=0.65pt]
      ([xshift=24mm,yshift=-18mm]current page.north west) --
      ([xshift=-24mm,yshift=-18mm]current page.north east);
  \end{tikzpicture}%
}
