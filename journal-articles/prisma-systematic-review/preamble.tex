\usepackage{amsmath}

% ---------- geometry & typography ----------
\usepackage[top=22mm,bottom=22mm,left=26mm,right=26mm]{geometry}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{microtype}
\setlength{\parindent}{0pt}
\setlength{\parskip}{5pt}

% ---------- colour ----------
\usepackage[dvipsnames,table]{xcolor}

\definecolor{boxbg}{HTML}{EDF4FB}    % very light blue tint
\definecolor{neutral}{HTML}{444444}

% ---------- tables ----------
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{array}
\renewcommand{\arraystretch}{1.25}

\usepackage{enumitem}

% ---------- TikZ for PRISMA diagram ----------
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc}

% ---------- misc ----------
\usepackage{hyperref}
\hypersetup{colorlinks=true,linkcolor=accent,urlcolor=accent,citecolor=accent,
            pdftitle={PRISMA 2020 Systematic Review},
            pdfauthor={Research Team}}

\captionsetup{font=small,labelfont={bf,color=accent},skip=6pt}
