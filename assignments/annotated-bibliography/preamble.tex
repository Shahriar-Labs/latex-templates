% Packages, colours and custom commands for this template.
% Annotated Bibliography Template — edit & compile online at https://letx.app

% ── Packages ────────────────────────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[
  top=1.1in, bottom=1.1in,
  left=1.15in, right=1.15in
]{geometry}
\usepackage{xcolor}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{hyperref}
\usepackage{microtype}
\usepackage{parskip}

% ── Parskip safety ──────────────────────────────────────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% ── Colours ─────────────────────────────────────────────────────────────────
\definecolor{accent}{HTML}{2E6B8A}   % steel blue — restrained academic accent
\definecolor{neutral}{HTML}{4A4A4A}  % dark charcoal for body text
\definecolor{light}{HTML}{AABDC8}    % pale blue for rules

% ── Hyperref ────────────────────────────────────────────────────────────────
\hypersetup{
  colorlinks=true,
  urlcolor=accent,
  linkcolor=accent,
  hidelinks
}

% ── Section styling (titlesec) ───────────────────────────────────────────────
% Thematic subheadings: small-caps, accent colour, thin rule below
\titleformat{\section}
  {\normalfont\large\bfseries\color{accent}}
  {}
  {0pt}
  {}
  [\vspace{2pt}{\color{light}\hrule height 0.6pt}\vspace{4pt}]

\titlespacing*{\section}{0pt}{18pt}{6pt}

% ── Bibliography entry macro ─────────────────────────────────────────────────
% Usage: \bibentry{<hanging-indent citation>}{<annotation paragraph>}
\newcommand{\bibentry}[2]{%
  \begin{list}{}{%
    \setlength{\leftmargin}{0.5in}%
    \setlength{\itemindent}{-0.5in}%
    \setlength{\topsep}{6pt}%
    \setlength{\itemsep}{0pt}%
    \setlength{\parsep}{0pt}%
  }%
  \item \textcolor{neutral}{#1}%
  \end{list}%
  \vspace{4pt}%
  \begin{list}{}{%
    \setlength{\leftmargin}{0.5in}%
    \setlength{\topsep}{0pt}%
    \setlength{\itemsep}{0pt}%
    \setlength{\parsep}{0pt}%
  }%
  \item \textcolor{neutral}{#2}%
  \end{list}%
  \vspace{10pt}%
}

% ── Title block macro ────────────────────────────────────────────────────────
\newcommand{\titleblock}[3]{%
  % #1 = topic/title, #2 = author, #3 = course + semester
  \begin{center}
    {\LARGE\bfseries\color{accent} #1}\\[6pt]
    {\large\color{neutral} Annotated Bibliography}\\[10pt]
    {\normalsize\color{neutral} #2}\\[3pt]
    {\normalsize\color{neutral} #3}
  \end{center}%
  \vspace{4pt}%
  {\color{light}\hrule height 0.8pt}%
  \vspace{14pt}%
}
