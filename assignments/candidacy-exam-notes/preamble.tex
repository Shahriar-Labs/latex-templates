% ── Packages ──────────────────────────────────────────────────────────────────
\usepackage[top=1in, bottom=1in, left=1in, right=1in]{geometry}
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{microtype}
\usepackage{amsmath, amssymb, amsthm}
\usepackage[dvipsnames]{xcolor}
\usepackage[most]{tcolorbox}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{booktabs}
\usepackage{hyperref}
\usepackage{parskip}
\setlength{\parindent}{0pt}
\setlength{\parskip}{4pt}

% ── Colors ────────────────────────────────────────────────────────────────────
\definecolor{accent}{HTML}{2E5FA3}       % steel blue — one accent
\definecolor{accentlight}{HTML}{EAF0FB}  % very pale blue tint
\definecolor{neutral}{HTML}{4A4A4A}      % dark grey body
\definecolor{rulecolor}{HTML}{BFCFE8}    % soft rule
\definecolor{tipcolor}{HTML}{1A7A4A}     % green for exam tips
\definecolor{tipbg}{HTML}{EAF7EF}
\definecolor{defbg}{HTML}{F5F5F5}

% ── Hyperref setup ────────────────────────────────────────────────────────────
\hypersetup{
  colorlinks=true,
  urlcolor=accent,
  linkcolor=accent,
  pdfauthor={PhD Candidate},
  pdftitle={Candidacy Exam Study Notes}
}

% ── Section styling ───────────────────────────────────────────────────────────
\titleformat{\section}
  {\large\bfseries\color{accent}}
  {}
  {0pt}
  {}
  [\vspace{1pt}\textcolor{rulecolor}{\rule{\linewidth}{0.8pt}}\vspace{2pt}]

\titleformat{\subsection}
  {\normalsize\bfseries\color{neutral}}
  {}
  {0pt}
  {}

\titlespacing{\section}{0pt}{10pt}{4pt}
\titlespacing{\subsection}{0pt}{6pt}{2pt}

% ── tcolorbox styles ──────────────────────────────────────────────────────────
\tcbuselibrary{skins, breakable}

% Key theorem / result box
\newtcolorbox{keyresult}[1]{
  enhanced,
  breakable,
  colback=accentlight,
  colframe=accent,
  fonttitle=\bfseries\small,
  title={#1},
  left=6pt, right=6pt, top=4pt, bottom=4pt,
  arc=3pt,
  boxrule=0.7pt,
  toptitle=2pt, bottomtitle=2pt
}

% Definition box
\newtcolorbox{defbox}[1]{
  enhanced,
  breakable,
  colback=defbg,
  colframe=neutral!40,
  fonttitle=\bfseries\small\color{neutral},
  title={#1},
  left=6pt, right=6pt, top=4pt, bottom=4pt,
  arc=3pt,
  boxrule=0.5pt,
  toptitle=2pt, bottomtitle=2pt
}

% Exam tip box
\newtcolorbox{examtip}{
  enhanced,
  breakable,
  colback=tipbg,
  colframe=tipcolor,
  fontupper=\small,
  left=6pt, right=6pt, top=3pt, bottom=3pt,
  arc=3pt,
  boxrule=0.5pt,
  overlay={
    \node[anchor=west, font=\bfseries\small\color{tipcolor}]
      at ([xshift=6pt]frame.north west) {\strut Exam Tip};
  },
  top=12pt
}

% ── Convenience macros ────────────────────────────────────────────────────────

\setlist[enumerate]{leftmargin=1.6em, itemsep=1pt, topsep=2pt, parsep=0pt}
