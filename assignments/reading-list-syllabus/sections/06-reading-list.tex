\section{Reading List}

\subsection*{\color{accent}Required Texts}

\begin{itemize}[leftmargin=*, itemsep=3pt]
  \reading{David Chalmers}{The Conscious Mind: In Search of a Fundamental Theory}
          {Oxford University Press, 1996. ISBN 978-0195117899.}
  \reading{Andy Clark}{Supersizing the Mind: Embodiment, Action, and Cognitive Extension}
          {Oxford University Press, 2008. ISBN 978-0195333213.}
  \reading{Daniel C.\ Dennett}{Brainstorms: Philosophical Essays on Mind and Psychology}
          {MIT Press, 1978 [repr.\ 1981]. ISBN 978-0262540441.}
  \reading{Derek Parfit}{Reasons and Persons, Part III: Personal Identity}
          {Oxford University Press, 1984. Selections distributed via course portal.}
  \reading{Course Reader}{\textup{Compiled Selections}}{Available for purchase at the campus copy centre or as PDF on the course portal. Includes all journal articles and book chapters listed in the weekly schedule.}
\end{itemize}

\subsection*{\color{accent}Recommended / Further Reading}

\begin{itemize}[leftmargin=*, itemsep=3pt]
  \reading{Ned Block, Owen Flanagan, and Güven Güzeldere (eds.)}{The Nature of Consciousness: Philosophical Debates}
          {MIT Press, 1997. An indispensable anthology.}
  \reading{Patricia Smith Churchland}{Neurophilosophy: Toward a Unified Science of the Mind/Brain}
          {MIT Press, 1986. Essential background for the eliminativist tradition.}
  \reading{Jerry A.\ Fodor}{The Language of Thought}
          {Harvard University Press, 1975. Foundational for computational theories of mind.}
  \reading{Thomas Nagel}{Mortal Questions}
          {Cambridge University Press, 1979. Contains ``What Is It Like to Be a Bat?''}
  \reading{John R.\ Searle}{The Rediscovery of the Mind}
          {MIT Press, 1992. Develops biological naturalism beyond the Chinese Room.}
  \reading{Francisco J.\ Varela, Evan Thompson, and Eleanor Rosch}{The Embodied Mind: Cognitive Science and Human Experience}
          {MIT Press, 1991. Foundational text for enactivism.}
  \reading{Gilbert Ryle}{The Concept of Mind}
          {Hutchinson, 1949 [Routledge, 2009]. The locus classicus of philosophical behaviourism.}
\end{itemize}
