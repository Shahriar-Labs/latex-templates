% ── Colours ─────────────────────────────────────────────────
\definecolor{accent}{HTML}{2D5F8A}   % steel blue — restrained academic accent
\definecolor{rule}{HTML}{A8BFCF}     % muted blue-grey for rules
\definecolor{muted}{HTML}{666666}    % secondary text

% ── Spacing reset (required with parskip + tables) ──────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{5pt}

% ── Header / Footer ─────────────────────────────────────────
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{rule}\leaders\hrule height \headrulewidth\hfill}}
\fancyhead[L]{\small\color{muted}PHIL 3820 — Philosophy of Mind}
\fancyhead[R]{\small\color{muted}Spring 2025}
\fancyfoot[C]{\small\color{muted}\thepage}

% ── Section style (titlesec) ────────────────────────────────
\titleformat{\section}
  {\large\bfseries\color{accent}}
  {}
  {0pt}
  {}
  [\vspace{-2pt}{\color{rule}\titlerule[0.6pt]}\vspace{4pt}]

\titlespacing*{\section}{0pt}{16pt}{6pt}

% ── Accent rule macro ────────────────────────────────────────
\newcommand{\accentrule}{%
  {\color{accent}\rule{\linewidth}{1.8pt}}%
}

% ── Info-row table helper ───────────────────────────────────
% KEY = l column coloured bold; VALUE = X column; gap = 8 mm
\newcolumntype{K}{>{\color{accent}\bfseries}l}

% ── Reading-list item macro ─────────────────────────────────
\newcommand{\reading}[3]{%
  \item \textbf{#1} — \textit{#2}. #3%
}
