% ── Geometry ──────────────────────────────────────────────
\usepackage[top=22mm,bottom=24mm,left=22mm,right=22mm]{geometry}

% ── Typography ────────────────────────────────────────────
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{microtype}

% ── Colour ────────────────────────────────────────────────
\usepackage[dvipsnames,svgnames,x11names]{xcolor}
\definecolor{accent}{HTML}{2D6A9F}   % steel blue — single accent
\definecolor{accentbg}{HTML}{EBF3FB} % very pale blue — tinted boxes
\definecolor{neutral}{HTML}{4A4A4A}  % body text
\definecolor{rulecol}{HTML}{BBCCDD}  % hairlines

% ── Layout helpers ────────────────────────────────────────
\usepackage{parskip}
\setlength{\parindent}{0pt}
\setlength{\parskip}{4pt}

\usepackage{titlesec}
\titleformat{\section}
  {\color{accent}\large\bfseries}{}{0em}{}
  [\vspace{1pt}\color{rulecol}\titlerule\vspace{2pt}]
\titlespacing*{\section}{0pt}{10pt}{4pt}

\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.4em, itemsep=1pt, topsep=3pt, parsep=0pt}
\setlist[enumerate]{leftmargin=1.6em, itemsep=1pt, topsep=3pt, parsep=0pt}

% ── Tables ────────────────────────────────────────────────
\usepackage{tabularx}
\usepackage{booktabs}
\renewcommand{\arraystretch}{1.15}

% ── Boxes ─────────────────────────────────────────────────
\usepackage[most]{tcolorbox}

\newtcolorbox{keybox}[1]{
  colback=accentbg,
  colframe=accent,
  colbacktitle=accent,
  coltitle=white,
  fonttitle=\bfseries\small,
  title={#1},
  boxrule=0.5pt,
  arc=3pt,
  left=7pt, right=7pt, top=4pt, bottom=4pt,
  titlerule=0pt,
  toptitle=3pt, bottomtitle=3pt,
  before upper={\setlength{\parskip}{3pt}\setlength{\parindent}{0pt}},
}

% ── Misc ──────────────────────────────────────────────────
\usepackage{fontawesome5}
\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning}
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  urlcolor=accent,
  linkcolor=accent,
  pdfauthor={Dr.\ Elena Vasquez},
  pdftitle={Seminar Handout — The Transformer Architecture},
}

% ── Header macro ──────────────────────────────────────────
\newcommand{\handoutheader}[5]{%
  {\color{accent}\rule{\linewidth}{1.5pt}}\par\vspace{4pt}
  {\fontsize{16}{19}\selectfont\bfseries\color{neutral}#2}\par\vspace{3pt}
  \begingroup\small\color{neutral}
    \faUniversity\enspace\textbf{#1}\quad
    \faUser\enspace#3\quad
    \faCalendar\enspace#4\quad
    \faHashtag\enspace Session~#5
  \endgroup\par\vspace{4pt}
  {\color{accent}\rule{\linewidth}{0.6pt}}
  \vspace{6pt}
}

% ── Discussion-question counter ───────────────────────────
\newcounter{dqcount}
\newcommand{\dq}[1]{%
  \stepcounter{dqcount}%
  \noindent
  \makebox[1.8em][l]{\textbf{\color{accent}Q\thedqcount.}}%
  #1\par\vspace{2pt}%
}
