% Base packages
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}

% Page layout and margins
\usepackage[margin=1in, headheight=14pt]{geometry}
\usepackage{fancyhdr}

% Color definition
\usepackage{xcolor}
\definecolor{accent}{HTML}{0D9488}       % Modern teal
\definecolor{accentdark}{HTML}{0F766E}   % Dark teal for headings
\definecolor{neutraldark}{HTML}{1E293B}   % Slate 800 for body text
\definecolor{neutrallight}{HTML}{F8FAFC}  % Slate 50 for code/summary backgrounds
\definecolor{bordergray}{HTML}{E2E8F0}    % Slate 200 for lines/borders

% Set body text color
\color{neutraldark}

% Graphics and plots
\usepackage{graphicx}

% Tables
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{enumitem}

% Code snippets formatting
\usepackage{listings}
\lstset{
    backgroundcolor=\color{neutrallight},
    basicstyle=\ttfamily\small\color{neutraldark},
    breakatwhitespace=false,         
    breaklines=true,                 
    captionpos=b,                    
    commentstyle=\color{gray},
    keywordstyle=\color{accentdark}\bfseries,
    numberstyle=\tiny\color{gray},
    numbers=left,

    showstringspaces=false,
    showtabs=false,                  
    tabsize=2,
    frame=tb,
    rulecolor=\color{bordergray},
    framerule=0.8pt
}

  {\color{accentdark}\normalfont\Large\bfseries}

    coltitle=accentdark,
    attach title to upper,
    after title={\par\medskip}
}
