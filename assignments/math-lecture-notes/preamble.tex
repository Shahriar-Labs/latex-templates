% Custom styled Proof environment using tcolorbox
\newtcolorbox{proofbox}[1]{
    enhanced,
    breakable,
    colback=proofcolor!2!white,
    colframe=proofcolor!60!black,
    arc=0mm,
    outer arc=0mm,
    boxrule=0pt,
    leftrule=2.5pt,
    top=2mm,
    bottom=2mm,
    left=3.5mm,
    right=3.5mm,
    title={#1},
    coltitle=proofcolor!80!black,
    fonttitle=\sffamily\bfseries\itshape,
    attach title to upper,
    after title={\par\smallskip\noindent},
    fontupper=\normalfont,
    after upper=\hfill\qedsymbol,
}

\makeatletter
\renewenvironment{proof}[1][\proofname]{%
  \begin{proofbox}{#1}%
}{%
  \end{proofbox}%
}
\makeatother
