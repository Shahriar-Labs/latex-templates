% ─── Encoding & Fonts ────────────────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella}          % Palatino-style serif — classic, elegant
\usepackage{microtype}

% ─── Page geometry ───────────────────────────────────────────────────────────
\usepackage[
  top=28mm, bottom=28mm,
  left=32mm, right=32mm,
  headheight=14pt
]{geometry}

% ─── Colour palette (one accent + neutrals) ──────────────────────────────────
\usepackage{xcolor}
\definecolor{accent}{HTML}{2D5FA6}   % deep cobalt — restrained academic blue
\definecolor{lightbg}{HTML}{F0F4FB}  % very pale blue tint for theorem boxes
\definecolor{neutralgray}{HTML}{6B7280}
\definecolor{rulecol}{HTML}{BECDE8}

% ─── Math packages ───────────────────────────────────────────────────────────
\usepackage{amsmath, amssymb, amsthm}
\usepackage{mathtools}

% ─── tcolorbox theorem environments ──────────────────────────────────────────
\usepackage[most]{tcolorbox}

% Generic styled box factory
\tcbset{
  proofbox/.style={
    enhanced,
    breakable,
    colback   = lightbg,
    colframe  = accent,
    arc       = 3pt,
    boxrule   = 0.6pt,
    left      = 8pt,
    right     = 8pt,
    top       = 5pt,
    bottom    = 5pt,
    fonttitle = \bfseries\color{white},
    attach boxed title to top left = {yshift=-2mm, xshift=8pt},
    boxed title style = {
      colback  = accent,
      colframe = accent,
      arc      = 2pt,
      boxrule  = 0pt,
      fontupper = \small\bfseries\color{white},
      left=4pt, right=4pt, top=2pt, bottom=2pt
    },
  },
  lemmabox/.style={
    enhanced,
    breakable,
    colback   = white,
    colframe  = rulecol,
    arc       = 3pt,
    boxrule   = 0.6pt,
    left=8pt, right=8pt, top=5pt, bottom=5pt,
    fonttitle = \bfseries\color{accent},
    attach boxed title to top left = {yshift=-2mm, xshift=8pt},
    boxed title style = {
      colback  = rulecol,
      colframe = rulecol,
      arc=2pt, boxrule=0pt,
      fontupper = \small\bfseries\color{accent},
      left=4pt, right=4pt, top=2pt, bottom=2pt
    },
  },
  defbox/.style={
    enhanced,
    breakable,
    colback   = white,
    colframe  = neutralgray,
    arc       = 3pt,
    boxrule   = 0.5pt,
    left=8pt, right=8pt, top=5pt, bottom=5pt,
    fonttitle = \bfseries\color{white},
    attach boxed title to top left = {yshift=-2mm, xshift=8pt},
    boxed title style = {
      colback  = neutralgray,
      colframe = neutralgray,
      arc=2pt, boxrule=0pt,
      fontupper = \small\bfseries\color{white},
      left=4pt, right=4pt, top=2pt, bottom=2pt
    },
  },
}

% Theorem counter
\newcounter{thmcount}[section]
\renewcommand{\thethmcount}{\arabic{thmcount}}

% ── Theorem
\newenvironment{Theorem}[1][]{%
  \refstepcounter{thmcount}%
  \begin{tcolorbox}[proofbox,
    title={Theorem~\thethmcount\ifx\relax#1\relax\else\ (#1)\fi}]%
}{%
  \end{tcolorbox}%
}

% ── Lemma
\newenvironment{Lemma}[1][]{%
  \refstepcounter{thmcount}%
  \begin{tcolorbox}[lemmabox,
    title={Lemma~\thethmcount\ifx\relax#1\relax\else\ (#1)\fi}]%
}{%
  \end{tcolorbox}%
}

% ── Proposition
\newenvironment{Proposition}[1][]{%
  \refstepcounter{thmcount}%
  \begin{tcolorbox}[lemmabox,
    title={Proposition~\thethmcount\ifx\relax#1\relax\else\ (#1)\fi}]%
}{%
  \end{tcolorbox}%
}

% ── Corollary
\newenvironment{Corollary}[1][]{%
  \refstepcounter{thmcount}%
  \begin{tcolorbox}[lemmabox,
    title={Corollary~\thethmcount\ifx\relax#1\relax\else\ (#1)\fi}]%
}{%
  \end{tcolorbox}%
}

% ── Definition
\newenvironment{Definition}[1][]{%
  \refstepcounter{thmcount}%
  \begin{tcolorbox}[defbox,
    title={Definition~\thethmcount\ifx\relax#1\relax\else\ (#1)\fi}]%
}{%
  \end{tcolorbox}%
}

% ── Proof environment with coloured QED box
\renewcommand{\qedsymbol}{%
  \tikz[baseline=-1.5pt]%
    \filldraw[accent] (0,0) rectangle (5.5pt,5.5pt);%
}
\renewenvironment{proof}[1][\proofname]{%
  \par\smallskip
  \noindent\textit{#1.}\enspace\ignorespaces
}{%
  \hfill\qedsymbol\par\smallskip
}
