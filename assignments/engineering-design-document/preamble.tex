%% ── Helper macros ────────────────────────────────────────────────────────────
\newcommand{\kvrow}[2]{#1 & #2 \\[2pt]}

\newcommand{\badge}[2]{%
  \tikz[baseline=(b.base)]\node[
    fill=#1!15, draw=#1!40, rounded corners=2pt,
    inner xsep=4pt, inner ysep=1.5pt,
    font=\footnotesize\bfseries\color{#1}
  ](b){#2};%
}

%% ── Title block macro ────────────────────────────────────────────────────────
\newcommand{\doctitle}[2]{%

    enhanced, sharp corners,
    colback=accentlt, colframe=accent,
    leftrule=4pt, toprule=0pt, bottomrule=0pt, rightrule=0pt,
    boxsep=0pt, left=10pt, right=10pt, top=10pt, bottom=10pt,
    before skip=0pt, after skip=14pt
  ]
    {\Large\bfseries\color{accent}#1}\\[4pt]
    {\normalsize\color{neutral}#2}
  \end{tcolorbox}%
}
