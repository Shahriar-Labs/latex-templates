\usepackage{amsmath}

% ── Core layout ─────────────────────────────────────────────

  top=1.15in, bottom=1.15in,
  left=1.25in, right=1.15in,
  headheight=28pt
]{geometry}

% ── Colour palette ───────────────────────────────────────────
\usepackage[dvipsnames,svgnames,x11names]{xcolor}
\definecolor{accent}{HTML}{1B3A5C}    % deep navy
\definecolor{rule}{HTML}{3A6491}      % mid navy rule
\definecolor{subtle}{HTML}{4A4A4A}    % body text
\definecolor{light}{HTML}{F5F4F0}     % page tint (parchment-warm)
\definecolor{keycolor}{HTML}{1B3A5C}

% ── Page tint ────────────────────────────────────────────────
\usepackage{pagecolor}
\pagecolor{light}

% ── Header / footer ──────────────────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\color{subtle}\textit{PRIVILEGED AND CONFIDENTIAL — ATTORNEY-CLIENT COMMUNICATION}}
\fancyhead[R]{\footnotesize\color{subtle}\thepage}
\renewcommand{\headrulewidth}{0.35pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{rule}\leaders\hrule height \headrulewidth\hfill}}

% ── Utilities ────────────────────────────────────────────────
\usepackage{tabularx}
\usepackage{booktabs}

\usepackage{enumitem}
\usepackage{titlesec}

\usepackage{fontawesome5}
\usepackage{ragged2e}

% ── hyperref setup ───────────────────────────────────────────
\hypersetup{
  colorlinks=true,
  urlcolor=rule,
  linkcolor=accent,
  pdfauthor={Harrington \& Associates LLP},
  pdftitle={Legal Memorandum — Greenfield Capital Partners},
  pdfsubject={Duty of Care and Fiduciary Obligations},

% ── Section styling ──────────────────────────────────────────
\titleformat{\section}
  {\normalfont\large\bfseries\color{accent}}
  {\thesection.}{0.55em}{}[{\color{rule}\vspace{2pt}\hrule height 0.5pt\vspace{3pt}}]

\titleformat{\subsection}
  {\normalfont\normalsize\bfseries\color{accent}}
  {\thesubsection.}{0.5em}{}

\titlespacing*{\section}{0pt}{16pt}{6pt}
\titlespacing*{\subsection}{0pt}{10pt}{4pt}

% ── List styles ──────────────────────────────────────────────
\setlist[itemize]{
  leftmargin=1.4em, itemsep=2pt, parsep=0pt,
  topsep=4pt, label={\color{accent}\textbullet}
}
\setlist[enumerate]{
  leftmargin=1.6em, itemsep=2pt, parsep=0pt, topsep=4pt,
  label={\color{accent}\arabic*.}
}

% ── Defined terms command ────────────────────────────────────
\newcommand{\term}[1]{\textbf{``#1''}}
\newcommand{\defined}[1]{\textit{\color{accent}#1}}

% ── Thin horizontal rule ─────────────────────────────────────
\newcommand{\hairrule}{\vspace{4pt}{\color{rule!50}\hrule height 0.3pt}\vspace{4pt}}

% ── Heading block macro (TO/FROM/DATE/RE) ────────────────────
% Key column = plain l (baseline), value = X
\newcommand{\memoheader}[4]{%
  \begin{tabularx}{\linewidth}{%
    @{}>{\color{keycolor}\bfseries\small}l@{\hspace{10mm}}>{\small\raggedright\arraybackslash}X@{}%
  }%
    TO:   & #1 \\[3pt]
    FROM: & #2 \\[3pt]
    DATE: & #3 \\[3pt]
    RE:   & #4 \\
  \end{tabularx}%
}

% ── Paragraph label (CREAC labels) ───────────────────────────
\newcommand{\creac}[1]{\vspace{6pt}\noindent{\color{accent}\bfseries\small #1\quad}\ignorespaces}

% ── Signature line ────────────────────────────────────────────
\newcommand{\sigline}[2]{%
  \vspace{18pt}
  \noindent\rule{6cm}{0.4pt}\\
  {\small\textbf{#1}}\\
  {\small\color{subtle}#2}%
}
