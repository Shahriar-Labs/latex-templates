\section{Related Authority \& Further Reading}
% ─────────────────────────────────────────────────────────────

\begin{itemize}[leftmargin=1.2em, itemsep=1pt, topsep=2pt]
  \item \textit{Plessy v.\ Ferguson}, 163 U.S.\ 537 (1896) --- overruled \textit{in re} public education.
  \item \textit{Sweatt v.\ Painter}, 339 U.S.\ 629 (1950) --- intangible inequalities in law school admissions.
  \item \textit{McLaurin v.\ Oklahoma State Regents}, 339 U.S.\ 637 (1950) --- impairment of graduate instruction.
  \item \textit{Brown v.\ Board of Education II}, 349 U.S.\ 294 (1955) --- remedial decree; ``all deliberate speed.''
  \item \textit{Green v.\ County School Bd.\ of New Kent Cty.}, 391 U.S.\ 430 (1968) --- affirmative duty to desegregate.
  \item \textit{Swann v.\ Charlotte-Mecklenburg Bd.\ of Ed.}, 402 U.S.\ 1 (1971) --- busing as remedial tool.
  \item \textit{Parents Involved in Community Schools v.\ Seattle School Dist.\ No.\ 1}, 551 U.S.\ 701 (2007) --- limits on voluntary race-conscious school assignments.
  \item Richard Kluger, \textit{Simple Justice: The History of Brown v.\ Board of Education and Black America's Struggle for Equality} (1975).
\end{itemize}
