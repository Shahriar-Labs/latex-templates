% Inline label used in key/value citation table
\newcommand{\kv}[1]{\textcolor{accent}{\textbf{#1}}}

% Small caps label for inline call-outs
\newcommand{\legalterm}[1]{\textsc{#1}}

% Numbered reasoning step
\newcounter{reasonstep}
\newcommand{\reasonstep}[1]{%
  \stepcounter{reasonstep}%
  \smallskip
  \noindent
  \begin{tikzpicture}[baseline=(char.base)]
    \node[circle, fill=navyblue, text=white,
          inner sep=1.8pt, font=\tiny\bfseries] (char) {\thereasoningstep};
  \end{tikzpicture}\hspace{4pt}\textbf{#1}%
  \par
}

% Alias for cleaner reference
\let\thereasoningstep\thereasoningstep
\newcommand{\thereasoningstep}{\arabic{reasonstep}}

% Dissent/concurrence callout
\newenvironment{rvcomment}[1]{%
  \par\smallskip
  \noindent{\color{midgray}\rule{0.3\linewidth}{0.3pt}}
  \par\noindent{\footnotesize\color{midgray}\textbf{#1}}\par\small\color{midgray}%
}{%
  \par\smallskip
}
