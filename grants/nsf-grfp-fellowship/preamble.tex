% Packages, colours and custom commands for this template.
% This template can be edited and compiled online at https://letx.app

% --- Margins & Page Setup ---
\usepackage[margin=1.0in]{geometry}

% --- Typography & Font ---
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella} % Elegant Palatino-clone font compliant with NSF guidelines
\usepackage{microtype} % Improves hyphenation and spacing

% --- Colors ---
\usepackage{xcolor}
\definecolor{accent}{HTML}{003366} % Restrained Academic Navy Blue
\definecolor{darkgray}{HTML}{222222} % Charcoal for main text

% --- Headers & Footers ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\fontsize{9pt}{11pt}\selectfont \textbf{\textcolor{accent}{Jane M. Doe}}}
\fancyhead[R]{\fontsize{9pt}{11pt}\selectfont \textcolor{gray}{\leftmark}}
\fancyfoot[C]{\fontsize{10pt}{12pt}\selectfont \thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% --- Section Customization ---
\usepackage{titlesec}
\setcounter{secnumdepth}{0} % Unnumbered sections

\titleformat{\section}
  {\color{accent}\normalfont\large\bfseries}
  {}{0em}{}
\titlespacing*{\section}{0pt}{6pt}{3pt}

\titleformat{\subsection}
  {\color{accent}\normalfont\normalsize\bfseries}
  {}{0em}{}
\titlespacing*{\subsection}{0pt}{4pt}{2pt}

% --- Tables & Lists ---
\usepackage{booktabs}
\usepackage{tabularx}

% --- Math & Hyperlinks ---
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage[hidelinks]{hyperref}

% --- Strict LetX Alignment Rules ---
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\newcommand{\newpar}{\par\vspace{4pt}\noindent}
