% Packages, colours and custom commands for this template.
% =========================================================================
% Data Management Plan (DMP) Template
% NSF/NIH-style Supplementary Document
% This template can be edited and compiled online at https://letx.app
% =========================================================================


\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{geometry}
\geometry{margin=0.85in}

% Color definitions
\usepackage[table]{xcolor}
\definecolor{accent}{HTML}{1A365D}      % Deep Navy Blue
\definecolor{neutraldark}{HTML}{2D3748}   % Charcoal for body text
\definecolor{lightgray}{HTML}{F7FAFC}     % Soft background for table headers
\definecolor{linkcolor}{HTML}{2B6CB0}     % Steel blue for links

% Font configuration
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}

% Table packages
\usepackage{tabularx}
\usepackage{booktabs}

% Headings styling
\usepackage{titlesec}
\titleformat{\section}
  {\color{accent}\normalfont\large\bfseries}
  {\thesection.}
  {0.5em}
  {}
  [\titlerule]
\titlespacing*{\section}{0pt}{1.4ex plus 0.4ex minus 0.2ex}{0.8ex plus 0.1ex}

\titleformat{\subsection}
  {\color{accent}\normalfont\normalsize\bfseries}
  {\thesubsection.}
  {0.5em}
  {}
\titlespacing*{\subsection}{0pt}{1.0ex plus 0.3ex minus 0.2ex}{0.5ex plus 0.1ex}

% List configuration
\usepackage{enumitem}
\setlist[itemize]{leftmargin=*,labelsep=0.4em,topsep=4pt,itemsep=3pt,parsep=0pt,partopsep=0pt}
\setlist[enumerate]{leftmargin=*,labelsep=0.4em,topsep=4pt,itemsep=3pt,parsep=0pt,partopsep=0pt}

% Captions
\usepackage[labelfont={color=accent,bf},font=small]{caption}

% Header and Footer
\usepackage{fancyhdr}
\usepackage{lastpage}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\fancyhead[L]{\color{neutraldark}\small Data Management Plan}
\fancyhead[R]{\color{neutraldark}\small NSF/NIH Supplementary Document}
\fancyfoot[C]{\color{neutraldark}\small Page \thepage\ of \pageref{LastPage}}

% Hyperlinks
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=linkcolor,
    filecolor=linkcolor,      
    urlcolor=linkcolor,
    citecolor=linkcolor,
    pdfauthor={Dr. Eleanor Vance},
    pdftitle={Data Management Plan},
}

% Hard rules - parskip & parindent
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
