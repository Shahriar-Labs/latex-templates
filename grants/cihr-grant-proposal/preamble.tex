%% ─── Helper macros ────────────────────────────────────────────────────────────
\newcommand{\cihrfield}[2]{%
  \vspace{2pt}%
  {\fontsize{7.5}{9}\selectfont\color{cihrDarkGray}\textbf{#1}}\par
  {\fontsize{9}{11}\selectfont #2}\vspace{4pt}%
}

\newcommand{\milestone}[3]{%
  \noindent
  \begin{minipage}[c]{0.06\linewidth}
    \centering
    \begin{tikzpicture}[baseline=(n.base)]
      \node[circle, fill=cihrNavy, text=white, inner sep=2.5pt,
            font=\bfseries\fontsize{7}{9}\selectfont] (n) {#1};
    \end{tikzpicture}%
  \end{minipage}%
  \begin{minipage}[c]{0.94\linewidth}
    \fontsize{9}{11}\selectfont\raggedright
    \textbf{#2}%
    \ifx\empty#3\empty\else\enspace---\enspace#3\fi
  \end{minipage}\par\vspace{3pt}%
}
