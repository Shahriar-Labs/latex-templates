\begin{tcolorbox}[notebox, title={\faClipboardList\quad Objectives \& Hypothesis}]

  \textbf{Primary Objective}\\[3pt]
  Synthesise aluminium-doped LLZO pellets at two dopant concentrations
  (0.2\,mol\% and 0.4\,mol\% Al\textsuperscript{3+}) and measure their room-temperature
  ionic conductivities via electrochemical impedance spectroscopy (EIS) using
  Au-sputtered blocking electrodes.

  \vspace{6pt}
  \textbf{Secondary Objectives}
  \begin{itemize}[leftmargin=16pt, itemsep=2pt, topsep=2pt]
    \item Confirm cubic-phase LLZO formation via X-ray diffraction (XRD) patterns.
    \item Compare pellet relative densities (target $\geq 94\,\%$ of theoretical).
    \item Record sintering mass loss and attribute to volatile Li\textsubscript{2}O evaporation.
  \end{itemize}

  \vspace{8pt}
  \textbf{Hypothesis}\\[3pt]
  Al\textsuperscript{3+} substitution for Li\textsuperscript{+} at tetrahedral sites will
  stabilise the cubic garnet phase at room temperature, yielding an ionic conductivity
  $\sigma \geq 3.0 \times 10^{-4}$\,S\,cm\textsuperscript{-1}, consistent with
  the results reported by \cite{Geiger2011} and the in-house baseline of
  $\sigma = 1.1 \times 10^{-4}$\,S\,cm\textsuperscript{-1} for undoped LLZO
  batch NRI-2024-0041.

  \vspace{8pt}
  {\footnotesize\color{Graphite}
    \textbf{Related protocols:} NRI-SOP-EC-07 (EIS procedure),
    NRI-SOP-SY-12 (solid-state synthesis), NRI-SOP-XR-03 (XRD sample prep).
  }

\end{tcolorbox}
