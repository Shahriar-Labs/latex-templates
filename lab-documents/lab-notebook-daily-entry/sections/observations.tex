  \textbf{Derived ionic conductivity}

  \vspace{4pt}
  \begin{tcolorbox}[
    enhanced,
    colback=LightSteel,
    colframe=InkBlue,
    boxrule=0.5pt,
    arc=2pt,
    left=12pt, right=12pt,
    top=6pt, bottom=6pt,
  ]
    \[
      \sigma = \frac{\ell}{R_{\text{tot}} \cdot A}
             = \frac{0.852\,\text{cm}^{-1}}{461.0\,\Omega}
             = 1.848 \times 10^{-3}\,\text{S}\,\text{cm}^{-1}
    \]
    {\small\color{Graphite}
      Uncertainty estimate: $\pm 5\,\%$ (dominant contributor: Au electrode
      contact area variability). \textbf{Result exceeds hypothesis threshold.}
    }
  \end{tcolorbox}
