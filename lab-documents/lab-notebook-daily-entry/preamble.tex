% ── Ruled lines helper (notebook paper illusion) ────────────
\newcommand{\ruledblock}[1]{%
  \begin{tikzpicture}[remember picture]
    \foreach \n in {1,...,#1}{
      \draw[RuledLine, thin] (0, -\n*5.5mm) -- (\linewidth-2pt, -\n*5.5mm);
    }
  \end{tikzpicture}%
}

% ── Section label style ────────────────────────────────────
\newcommand{\sectionlabel}[2]{%
  \vspace{6pt}%
  \noindent
  \begin{tikzpicture}
    \node[
      fill=InkBlue,
      text=white,
      font=\bfseries\footnotesize,
      inner xsep=8pt,
      inner ysep=4pt,
      anchor=west
    ] (lbl) {#1};
    \node[
      right=6pt of lbl,
      text=Graphite,
      font=\small,
      anchor=west
    ] {#2};
    \draw[RuledLine, thin] (lbl.south east) ++(6pt,0) -- ++(\linewidth - \pgfkeysvalueof{/pgf/minimum width} - 60pt, 0);
  \end{tikzpicture}%
  \vspace{2pt}%
}

% ── Signature line ─────────────────────────────────────────
\newcommand{\sigline}[2]{%
  \begin{minipage}[t]{0.44\linewidth}
    \vspace{22pt}%
    {\color{Graphite}\rule{\linewidth}{0.5pt}}\\[2pt]
    {\footnotesize\color{Graphite}\textbf{#1}\\#2}
  \end{minipage}%
}
