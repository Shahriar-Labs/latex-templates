\newcommand{\bslchip}[1]{%
  \tikz[baseline=-1pt]{%
    \node[fill=BSLamber, text=white, font=\sffamily\bfseries\tiny,
          inner sep=2pt, rounded corners=2pt] {#1};%
  }%
}

\newcommand{\required}{\bslchip{REQUIRED}}

  font=\sffamily\bfseries\tiny, inner sep=2pt, rounded corners=2pt]{RECOMMENDED};}}

\newcommand{\biosymbol}{%
  \begin{tikzpicture}[scale=0.28, baseline=-3pt]
    \fill[BSLamber] (0,0) circle (1.1);
    \fill[white] (0,0) circle (0.35);
    \foreach \a in {0,120,240}{
      \begin{scope}[rotate=\a]
        \fill[white] (0,0.55) ellipse ({0.28} and {0.35});
      \end{scope}
    }
  \end{tikzpicture}%
}
