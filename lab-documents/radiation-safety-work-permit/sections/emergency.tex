\PermitHead{Spill Response}{FORM MIBR-RS-14}

\begin{minipage}[t]{0.48\linewidth}
\textbf{Minor spill} (contained, activity below 1 mCi, no personnel contamination):
\begin{enumerate}
\item Alert everyone in the room; restrict entry.
\item Cover the spill with absorbent pads; do not spread material.
\item Survey the perimeter to confirm the spill is contained.
\item Don gloves and clean from the outside inward into a labeled waste bag.
\item Re-survey; notify the RSO and log the event on Form MIBR-RS-14.
\end{enumerate}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.48\linewidth}
\textbf{Major spill} (activity above 1 mCi, personnel contamination, or airborne risk):
\begin{enumerate}
\item Evacuate the room immediately; close the door.
\item Post a person outside to prevent entry; do not attempt cleanup.
\item Call the RSO and Campus Security from the table above.
\item Contaminated individuals remove affected clothing and proceed to the nearest safety
shower or sink for skin decontamination.
\item Await RSO clearance before re-entry.
\end{enumerate}
\end{minipage}
