\begin{minipage}[t]{0.485\linewidth}
\subsecbar{A. General Orientation}{}
\begin{checklist}
\item Emergency exits and outdoor assembly point identified
\item Fire extinguisher and fire blanket locations shown
\item Eyewash station and safety shower demonstrated
\item SDS binder / online SDS portal access explained
\item Emergency gas and power shut-off points identified
\end{checklist}

\subsecbar{B. Hazard Communication}{}
\begin{checklist}
\item GHS pictogram and label system reviewed
\item Door hazard signage (chemical, biological, radiation) explained
\item Incompatible-chemical segregated storage rules reviewed
\item Biosafety Level 2 practices reviewed (if applicable)
\end{checklist}

\subsecbar{C. Personal Protective Equipment}{}
\begin{checklist}
\item Correct PPE for assigned bench tasks confirmed
\item PPE storage, laundering and disposal procedure explained
\item Prescription safety eyewear requirement discussed
\end{checklist}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.485\linewidth}
\subsecbar{D. Emergency Procedures}{}
\begin{checklist}
\item Chemical spill response procedure demonstrated
\item Fire alarm response and evacuation route walked
\item Sharps / needlestick injury protocol reviewed
\item Chemical splash and eye-exposure first-aid steps reviewed
\item Building emergency assembly point confirmed
\end{checklist}

\subsecbar{E. Waste Management}{}
\begin{checklist}
\item Segregation of chemical, biohazard and sharps waste explained
\item Waste labeling and full-container pickup procedure reviewed
\end{checklist}

\subsecbar{F. Equipment-Specific Training}{}
\begin{checklist}
\item Fume hood / biosafety cabinet operation demonstrated
\item Autoclave operation and required PPE reviewed
\item Centrifuge safe-loading and balancing procedure reviewed
\item Compressed gas cylinder handling and securing reviewed
\end{checklist}
\end{minipage}
