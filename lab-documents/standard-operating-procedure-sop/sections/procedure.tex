\section{Procedure}

\subsection{Receiving and Inspection}

\begin{procstep}{1}
Confirm the shipment matches the packing slip: catalog number, lot number, quantity, and
expiration date. Reject and quarantine any shipment with a damaged cold-pack or a
temperature-indicator strip showing excursion.
\end{procstep}

\begin{procstep}{2}
Record the reagent name, supplier lot number, receipt date, and assigned internal lot
number (format \texttt{NXB-YYYY-\#\#\#\#}) in Form QF-014-A within 1 hour of receipt.
\end{procstep}

\begin{procstep}{3}
Print and affix an internal barcode label to every individual container. The label must
include the internal lot number, receipt date, and assigned storage condition.
\end{procstep}

\subsection{Storage Assignment}

\begin{procstep}{4}
Consult the Safety Data Sheet (SDS) storage recommendation and assign the reagent to
COLD-04, FRZ-11, or ULT-02 accordingly. Light-sensitive reagents are additionally wrapped
in amber sleeves before placement.
\end{procstep}

\begin{procstep}{5}
Place the reagent in the shelf position designated for its category on the storage map
posted inside each unit door. First-in-first-out (FIFO) rotation applies within each
category.
\end{procstep}

\subsection{Ongoing Monitoring}

\begin{procstep}{6}
Verify the continuous logger reading against the unit's independent thermometer at the
start of each shift and initial Form QF-014-B. Investigate any deviation of more than
2\,\textdegree C from setpoint immediately.
\end{procstep}

\begin{procstep}{7}
Review the weekly logger export every Monday. Flag any excursion event exceeding 30
minutes to the Laboratory Supervisor for disposition of the affected lots.
\end{procstep}

\begin{cautionbox}[ -- Cold-Chain Excursion]
A confirmed excursion of more than 2 hours at 2--8\,\textdegree C, or any duration above
$-65$\,\textdegree C for ULT-02 contents, requires immediate quarantine of the affected
lot and notification of the QA Reviewer before any further use.
\end{cautionbox}
