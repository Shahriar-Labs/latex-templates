\section{Roles and Responsibilities}

\begin{tcolorbox}[
    enhanced, breakable, colback=paperwhite, colframe=inkblue!80,
    boxrule=0.6pt, arc=2pt, left=8pt, right=8pt, top=6pt, bottom=6pt
  ]
  \begin{tabularx}{\linewidth}{@{} l >{\RaggedRight}X @{}}
    \toprule
    \textbf{\color{inkblue}Role} & \textbf{\color{inkblue}Responsibility} \\
    \midrule
    Laboratory Technician &
      Executes receiving, labeling, and storage steps; records lot data in the reagent
      log within 1 hour of receipt. \\
    \addlinespace
    Laboratory Supervisor &
      Verifies weekly cold-chain logs; co-signs storage-excursion investigations;
      approves reagent disposal requests. \\
    \addlinespace
    QA Reviewer &
      Audits reagent log entries monthly; confirms this SOP is followed during internal
      inspections; initiates CAPA for repeat deviations. \\
    \addlinespace
    Document Control Officer &
      Maintains the master copy of this SOP; distributes controlled revisions; retires
      superseded versions per Section~6. \\
    \bottomrule
  \end{tabularx}
\end{tcolorbox}
