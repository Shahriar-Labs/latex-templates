\cossection{\faChartBar\quad Severity--Likelihood Risk Matrix}

\vspace{4pt}

{\small
Risk scores are the product of \textbf{Severity} (S, 1--5) and \textbf{Likelihood}
(L, 1--5). Colour banding follows the standard 5-tier scheme used across
Nexa Biomedical Research Ltd sites.
}

\vspace{8pt}

%% ---- Risk matrix grid (TikZ) --------------------------------
\begin{center}
\begin{tikzpicture}[
  cell/.style={
    draw=white, line width=1pt,
    minimum width=18mm, minimum height=12mm,
    inner sep=0pt,
    font=\bfseries\small,
    text=white
  },
  rlabel/.style={font=\footnotesize\bfseries, text=NavyDeep, align=center},
  clabel/.style={font=\footnotesize\bfseries, text=NavyDeep, align=center}
]

%% Define row colours by likelihood (bottom=1, top=5)
%% Matrix: rows = likelihood (1..5 bottom to top), cols = severity (1..5)
%% Score = row * col; colour bands: 1-4 Low, 5-9 Med, 10-14 High, 15-19 VHigh, 20-25 Crit

\def\cellwidth{18mm}
\def\cellheight{12mm}

\foreach \lik/\lirow in {1/0, 2/1, 3/2, 4/3, 5/4}{
  \foreach \sev/\sevcol in {1/0, 2/1, 3/2, 4/3, 5/4}{
    \pgfmathsetmacro{\score}{\lik * \sev}
    \pgfmathsetmacro{\scoreint}{int(\score)}
    % Choose colour based on score
    \pgfmathparse{int(\score) < 5 ? 1 : 0}
    \ifnum\pgfmathresult=1
      \def\fillcol{SageGreen}
    \else
      \pgfmathparse{int(\score) < 10 ? 1 : 0}
      \ifnum\pgfmathresult=1
        \def\fillcol{MidAmber}
      \else
        \pgfmathparse{int(\score) < 15 ? 1 : 0}
        \ifnum\pgfmathresult=1
          \def\fillcol{AmberAlert}
        \else
          \pgfmathparse{int(\score) < 20 ? 1 : 0}
          \ifnum\pgfmathresult=1
            \def\fillcol{DangerRed}
          \else
            \def\fillcol{NavyDeep}
          \fi
        \fi
      \fi
    \fi
    \fill[\fillcol]
      ({\sevcol * \cellwidth}, {\lirow * \cellheight})
      rectangle
      ({(\sevcol+1) * \cellwidth}, {(\lirow+1) * \cellheight});
    \draw[white, line width=0.8pt]
      ({\sevcol * \cellwidth}, {\lirow * \cellheight})
      rectangle
      ({(\sevcol+1) * \cellwidth}, {(\lirow+1) * \cellheight});
    \node[text=white, font=\bfseries\small] at
      ({(\sevcol + 0.5) * \cellwidth}, {(\lirow + 0.5) * \cellheight})
      {\scoreint};
  }
}

%% Axis labels — Severity (X)
\foreach \sev/\sevcol in {1/0, 2/1, 3/2, 4/3, 5/4}{
  \node[clabel] at ({(\sevcol+0.5)*\cellwidth}, {-0.55*\cellheight})
    {\sev};
}
\node[clabel] at ({2.5*\cellwidth}, {-1.1*\cellheight}) {\textbf{Severity (S)}};

%% Axis labels — Likelihood (Y)
\foreach \lik/\lirow in {1/0, 2/1, 3/2, 4/3, 5/4}{
  \node[rlabel] at ({-0.6*\cellwidth}, {(\lirow+0.5)*\cellheight})
    {\lik};
}
\node[rlabel, rotate=90] at ({-1.3*\cellwidth}, {2.5*\cellheight})
  {\textbf{Likelihood (L)}};

%% Box outline
\draw[NavyDeep, line width=1.5pt] (0,0) rectangle
  ({5*\cellwidth}, {5*\cellheight});

\end{tikzpicture}
\end{center}
