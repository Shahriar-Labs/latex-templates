\cossection{\faLock\quad Control Measures (Hierarchy of Control)}

\vspace{4pt}

{\small Controls are listed following the \textbf{COSHH hierarchy}: Elimination
$\rightarrow$ Substitution $\rightarrow$ Engineering $\rightarrow$
Administrative $\rightarrow$ PPE. Higher-order controls are always preferred.}

\vspace{6pt}

%% Level 1 & 2
\noindent
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[
    colback=NavyDeep!8, colframe=NavyDeep, boxrule=1pt,
    arc=4pt, left=8pt, right=8pt, top=6pt, bottom=6pt
  ]
    \textbf{\textcolor{NavyDeep}{\faTimesCircle\quad Elimination / Substitution}}

    \vspace{4pt}
    \begin{itemize}[leftmargin=*, itemsep=2pt, topsep=2pt]
      \item Thionyl chloride: \textit{no viable substitute} for acid chloride
            generation at this scale. Elimination is not feasible; minimise
            batch size to $\leq$50\,mL.
      \item DCM: replace with 2-methyltetrahydrofuran (2-MeTHF) where
            extraction yield permits (\textit{ongoing trial, Q2 2024}).
      \item TEA: consider polymer-supported base (Amberlyst A21) for
            scale-up reactions.
    \end{itemize}
  \end{tcolorbox}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[
    colback=SlateBlue!8, colframe=SlateBlue, boxrule=1pt,
    arc=4pt, left=8pt, right=8pt, top=6pt, bottom=6pt
  ]
    \textbf{\textcolor{SlateBlue}{\faCog\quad Engineering Controls}}

    \vspace{4pt}
    \begin{itemize}[leftmargin=*, itemsep=2pt, topsep=2pt]
      \item All volatile and toxic liquid handling in \textbf{Class~II fume
            cupboard} (face velocity $\geq$0.5\,m/s, tested 2024-01-18).
      \item Thionyl chloride reactions on \textbf{Schlenk line} with
            \textbf{NaOH acid-gas scrubber} ($\geq$4\,M, 5\,L).
      \item Local exhaust ventilation (LEV) at solvent evaporation workstation.
      \item Antistatic earthing straps on all large-volume THF vessels.
      \item Dedicated pyrophoric cabinet for NaH storage.
    \end{itemize}
  \end{tcolorbox}
\end{minipage}

%% Level 3 & 4
\noindent
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[
    colback=AmberAlert!8, colframe=AmberAlert, boxrule=1pt,
    arc=4pt, left=8pt, right=8pt, top=6pt, bottom=6pt
  ]
    \textbf{\textcolor{AmberAlert!80!black}{\faClipboardList\quad Administrative Controls}}

    \vspace{4pt}
    \begin{itemize}[leftmargin=*, itemsep=2pt, topsep=2pt]
      \item Two-person rule: thionyl chloride and NaH operations require a
            second competent person present in the laboratory.
      \item Peroxide test strip check on THF before each use (dip strip,
            record in lab notebook, discard if $>$25\,ppm).
      \item Maximum single-session scale limits (see inventory table).
      \item Monthly fume cupboard face-velocity log (Form NXB-EHS-04).
      \item All personnel must complete NXB COSHH induction (eLearning
            module \#3) and sign acknowledgement sheet.
      \item Emergency eyewash station tested weekly; deluge shower monthly.
    \end{itemize}
  \end{tcolorbox}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.48\linewidth}
  \begin{tcolorbox}[

  ]
    \textbf{\textcolor{SageGreen!70!black}{\faHardHat\quad Personal Protective Equipment (PPE)}}

    \vspace{4pt}
    \begin{tabularx}{\linewidth}{@{} l X @{}}
      \cyes & \textbf{Lab coat:} cotton, knee-length, closed front \\
      \cyes & \textbf{Gloves:} nitrile (inner, 0.3\,mm) + butyl rubber
              (outer) for DMSO and corrosive work \\
      \cyes & \textbf{Eye protection:} chemical-splash goggles (EN~166:2002)
              at all times \\
      \cyes & \textbf{Face shield:} mandatory for thionyl chloride
              and NaH additions \\
      \cyes & \textbf{Footwear:} closed-toe, impermeable; no synthetic
              fabrics \\
      \cno  & \textbf{Respirator:} not routinely required (fume hood used);
              half-face APF~20 available for spill response \\
    \end{tabularx}
  \end{tcolorbox}
\end{minipage}
