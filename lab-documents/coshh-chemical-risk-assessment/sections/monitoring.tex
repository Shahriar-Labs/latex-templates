\cossection{\faStethoscope\quad Health Surveillance and Exposure Monitoring}

\vspace{4pt}

{\small
All personnel working with listed substances on a regular basis
($\geq$4\,h/week) are enrolled on the Nexa Biomedical occupational health
programme administered by \textbf{Meridian Occupational Health Services Ltd}.
}

\cossection{\faTrash\quad Waste Disposal}

\vspace{4pt}

{\small
All chemical waste must be disposed of in accordance with the
\textbf{Environmental Protection Act 1990} and Nexa Biomedical Waste
Management Procedure NXB-EHS-WMP-03.
}

\vspace{6pt}

\noindent
\begin{tabularx}{\linewidth}{@{}
  >{\small\bfseries}p{3.2cm}

\midrule
\rowcolor{LightFog}
Halogenated solvents (DCM) &

\rowcolor{white}
Non-halogenated solvents (TEA, THF) &
Separate amber bottle, EWC 07\,01\,04* &
Vertex EHS Contractors \\
\rowcolor{LightFog}
Acid waste (thionyl chloride residues) &
Quench slowly onto ice/water in fume hood; neutralise to pH\,5--9;
transfer to labelled inorganic acid waste bottle &
Vertex EHS Contractors \\
\rowcolor{white}
Reactive metal waste (NaH) &
Quench individually with dry isopropanol in fume hood (small portions);
solid residue to labelled reactive metals container &
Vertex EHS Contractors \\
\rowcolor{LightFog}
Contaminated PPE &
Double-bag in yellow chemical-waste bags; label with hazard class &
Campus hazardous waste store \\
\bottomrule

\vspace{4pt}
\begin{tcolorbox}[infobox]
  \faBell\quad \textbf{THF Peroxide Disposal:} Peroxide-positive THF
  (\textgreater{}25\,ppm) must \textbf{NOT} be evaporated, distilled, or
  stored. Contact the EHS office (x3200) immediately for specialist disposal.
  Do not decant into standard solvent waste.
\end{tcolorbox}
