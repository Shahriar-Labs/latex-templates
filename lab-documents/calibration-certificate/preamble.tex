    \draw[certnavy,line width=1.1pt]
      ($(current page.north west)+(0.9cm,-0.9cm)$) rectangle
      ($(current page.south east)+(-0.9cm,0.9cm)$);
    \draw[certgold,line width=0.4pt]
      ($(current page.north west)+(1.05cm,-1.05cm)$) rectangle
