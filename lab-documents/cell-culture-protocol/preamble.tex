%% ── Convenience commands ──────────────────────────────────────────────────────
% Step number badge
\newcommand{\stepbadge}[1]{%
  \tikz[baseline=(n.base)]{%
    \node[circle, fill=ProtoTeal, text=white, inner sep=2.2pt,
          font=\bfseries\small] (n) {#1};%
  }%
}

% Protocol step with badge
\newcounter{protostep}
\newcommand{\protostep}[2]{%
  \stepcounter{protostep}%
  \par\vspace{4pt}%
  \noindent\stepbadge{\theprotostep}\enspace%
  \textbf{#1}\enspace #2\par%
}
\newcommand{\resetsteps}{\setcounter{protostep}{0}}

% Inline concentration / volume helpers
\newcommand{\conc}[2]{\textbf{#1}\,#2}      % e.g. \conc{10}{\ensuremath{\mu} M}
\newcommand{\vol}[2]{\textbf{#1}\,#2}        % e.g. \vol{200}{\ensuremath{\mu} L}

% Sidebar note pair (timing | note) using minipage
\newcommand{\siderowpair}[2]{%
  \par\noindent%
  \begin{minipage}[t]{0.48\linewidth}%
    \begin{tcolorbox}[timingbox]
      #1
    \end{tcolorbox}%
  \end{minipage}%
  \hfill%
  \begin{minipage}[t]{0.48\linewidth}%
    \begin{tcolorbox}[cautionbox]
      #2
    \end{tcolorbox}%
  \end{minipage}%
  \par%
}
