\section{Procedure A: Cell Passage (Subculture)}

\resetsteps

\siderowpair{%
  Total time: \textbf{30--45 min}\\
  Frequency: Every \textbf{2--3 days} (HEK293T) or \textbf{3--4 days} (CHO-K1)\\
  Best performed: \textbf{morning}, to allow equilibration
}{%
  Subculture at \textbf{70--80\% confluency} only.\\
  Over-confluent cells lose contact inhibition and transfection efficiency.\\
  Never allow HEK293T to exceed 90\% confluency.
}

\subsection{Preparation}

\protostep{Warm reagents.}{Place complete growth medium, DPBS, and trypsin--EDTA in the 37\,$^\circ$C water bath
  for at least 20\,min before use. Cold trypsin acts slowly and causes cell clumping.}

\protostep{Decontaminate the biosafety cabinet.}{Wipe down all interior surfaces with 70\% ethanol and allow
  to evaporate for 2\,min. Switch on UV lamp for 15\,min if the cabinet has been idle \textgreater\,1\,h.
  Never expose cells or open media to UV.}

\protostep{Inspect cells by microscopy.}{Using an inverted microscope at 10\texttimes{} magnification,
  confirm confluency (\textbf{70--80\%}), check for morphological abnormalities, and confirm the absence
  of contamination (turbid media, unusual granularity, floating debris).}

\subsection{Trypsinisation}

\protostep{Aspirate spent medium.}{Using a sterile serological pipette connected to the vacuum line,
  carefully aspirate the conditioned medium from the flask. Tilt the flask at a low angle to
  collect residual medium without disturbing the cell monolayer.}

\protostep{Wash with DPBS.}{Add \vol{5}{mL} (T-25) or \vol{10}{mL} (T-75) of Ca\textsuperscript{2+}/Mg\textsuperscript{2+}-free
  DPBS. Rock gently to cover the entire monolayer, then aspirate completely. This removes residual
  FBS that would inhibit trypsin activity.}

\protostep{Add trypsin--EDTA.}{Pipette \vol{1}{mL} (T-25) or \vol{3}{mL} (T-75) of 0.25\% trypsin--EDTA
  evenly over the cell monolayer. Return the flask to the incubator.}

\begin{tcolorbox}[timingbox]
  \textbf{HEK293T:} 2--3\,min at 37\,$^\circ$C (loosely adherent -- check every 60\,s).\\
  \textbf{CHO-K1:} 4--6\,min at 37\,$^\circ$C (more adherent -- gentle tap if needed).\\
  \emph{Do not exceed 5\,min (HEK293T) or 8\,min (CHO-K1) to avoid cell damage.}
\end{tcolorbox}

\protostep{Monitor detachment.}{Check under the microscope every 60\,s. When \textgreater\,90\% of cells
  are rounded and floating, remove the flask from the incubator. Gently tap the side of the flask
  once or twice to dislodge remaining adherent cells. \textbf{Do not vortex.}}

\protostep{Neutralise trypsin.}{Add \vol{4}{mL} (T-25) or \vol{10}{mL} (T-75) of complete growth medium
  containing 10\% FBS immediately. Pipette up and down 3--4 times to create a single-cell suspension.
  Transfer the entire volume to a 15\,mL conical tube.}

\subsection{Cell Counting \& Reseeding}

\protostep{Centrifuge.}{Spin at \textbf{300\,$\times$\,g} for 5\,min at room temperature. Carefully aspirate
  the supernatant without disturbing the cell pellet.}

\protostep{Resuspend the pellet.}{Add \vol{1}{mL} of fresh complete medium. Pipette gently to disperse
  the pellet into a uniform single-cell suspension.}

\protostep{Count viable cells.}{Mix \vol{10}{\ensuremath{\mu} L} of cell suspension with \vol{10}{\ensuremath{\mu} L}
  of 0.4\% Trypan Blue. Load \vol{10}{\ensuremath{\mu} L} into a Neubauer haemocytometer. Count viable
  (clear) and non-viable (blue) cells in all four outer quadrants. Calculate:
  \[
    \text{Viable cells/mL} = \frac{\text{viable cell count}}{4} \times 2 \times 10^{4}
  \]
  Target viability: \textbf{\textgreater\,95\%}. If viability is \textless\,85\%, investigate for contamination
  or suboptimal culture conditions before continuing.}

\protostep{Reseed into new flasks.}{Dilute the cell suspension to the recommended seeding density in
  pre-warmed complete medium and transfer to a new flask:}

\vspace{4pt}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lccc}
\toprule
\rowcolor{ProtoTeal!15}
\textbf{Cell Line} & \textbf{Split Ratio} & \textbf{Seeding Density} & \textbf{Flask} \\
\midrule
HEK293T & 1:4 -- 1:10 & $3\text{--}5 \times 10^{4}$ cells/cm\textsuperscript{2} & T-75 \\
\rowcolor{ProtoGray}
CHO-K1  & 1:3 -- 1:6  & $2\text{--}4 \times 10^{4}$ cells/cm\textsuperscript{2} & T-75 \\
\bottomrule
\end{tabular}
\end{center}

\vspace{4pt}

\protostep{Label and incubate.}{Label the flask with: cell line, passage number, date, operator initials,
  and complete medium formulation. Place upright in the incubator at 37\,$^\circ$C, 5\% CO\textsubscript{2}.
  \textbf{Do not stack flasks.}}

\begin{tcolorbox}[infobox, title=\faBookOpen\enspace Passage Record]
  Record each passage in the Cell Line Logbook (Form~MRD-CL-003). Entry must include: date,
  flask ID, passage number, cell count, viability, split ratio, and operator signature. Maximum
  recommended passage number: \textbf{P30} (HEK293T) / \textbf{P25} (CHO-K1). Discard and
  thaw a new vial from the master cell bank beyond these limits.
\end{tcolorbox}
