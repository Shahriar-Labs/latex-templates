\section{Procedure B: Cryopreservation \& Thawing}

\subsection{Cryopreservation}

\resetsteps

\siderowpair{%
  Total time: \textbf{60 min} active + overnight slow-freeze\\
  Optimal harvest: \textbf{log-phase cells at 70\% confluency}\\
  Target: \textbf{2--5\,$\times$\,10\textsuperscript{6} cells/vial}
}{%
  Use only \textbf{certified cryogenic vials} (1.8\,mL, external thread).\\
  \textbf{Never} use standard 1.5\,mL Eppendorf tubes.\\
  DMSO is cytotoxic at RT -- proceed quickly after addition.
}

\protostep{Harvest cells.}{Perform Steps 1--10 of Procedure~A (passage). After centrifugation,
  aspirate all supernatant and resuspend the pellet in \vol{1}{mL} of ice-cold CryoStor CS10
  freezing medium per $2.5\times10^{6}$ cells. Keep on ice throughout.}

\protostep{If preparing DMSO-based freeze medium manually,}{dissolve 10\% (v/v) DMSO in
  complete growth medium pre-chilled to 4\,$^\circ$C. Add DMSO \emph{dropwise} to medium --
  never add medium to neat DMSO. The mixture will warm; chill again before use. \textbf{Prepare
  fresh on the day of use.}}

\protostep{Aliquot into cryovials.}{Pipette \vol{1}{mL} of cell suspension per cryovial. Close
  caps firmly. Label each vial with: cell line, passage number, date, operator, vial number, and
  total vials in the batch (e.g.\ ``3 of 8'').}

\protostep{Initiate controlled-rate freezing.}{Place vials in a pre-chilled isopropanol
  Mr Frosty\texttrademark{}-style container at $-20\,$\,$^\circ$C for 30\,min, then transfer to
  $-80\,$\,$^\circ$C overnight ($\approx$16\,h). This achieves the optimal
  $-1\,^\circ$C/min cooling rate.}

\begin{tcolorbox}[cautionbox]
  Transfer vials to liquid nitrogen storage within \textbf{24\,h} of placing at $-80\,$\,$^\circ$C.
  Vials left at $-80\,$\,$^\circ$C for \textgreater\,2\,weeks suffer accelerated ice crystal
  damage and significantly reduced post-thaw viability.
\end{tcolorbox}

\protostep{Transfer to liquid nitrogen.}{Using insulated cryo-gloves and face shield, place vials
  into a labelled cryobox and submerge in the liquid nitrogen dewar. Record vial location (box,
  row, column) in the cryogenic inventory database (Meridian LIMS -- Module: CryoStore).}

\vspace{4pt}

\subsection{Thawing}

\resetsteps

\begin{tcolorbox}[timingbox]
  Total time: \textbf{20--30 min}. Thawing and dilution must be performed rapidly.
  Prolonged exposure to DMSO at 37\,$^\circ$C is highly cytotoxic.
\end{tcolorbox}

\vspace{4pt}

\protostep{Pre-warm media and equipment.}{Place \vol{9}{mL} of complete growth medium in a 15\,mL
  conical tube in the 37\,$^\circ$C water bath. Prepare a T-25 flask with \vol{4}{mL} of complete
  medium at 37\,$^\circ$C.}

\protostep{Retrieve cryovial from liquid nitrogen.}{Wearing insulated cryo-gloves and face shield,
  remove the vial from the dewar. Note the vial ID against the inventory. Carry the vial in a dry
  ice bucket to the biosafety cabinet -- do not allow premature thawing in transit.}

\protostep{Rapid thaw.}{Place the cryovial in the 37\,$^\circ$C water bath. Agitate gently by hand.
  Thaw until only a small ice crystal remains (approximately \textbf{60--90\,s}).
  Wipe the exterior of the vial with 70\% ethanol and transfer to the biosafety cabinet.}

\protostep{Dilute dropwise.}{Transfer the cell suspension \emph{dropwise} with a 1\,mL pipette
  into the pre-warmed \vol{9}{mL} medium in the conical tube. This 1:9 dilution reduces the DMSO
  concentration from 10\% to $\approx$1\%, preventing osmotic shock.}

\protostep{Centrifuge and remove DMSO.}{Spin at \textbf{300\,$\times$\,g} for 5\,min. Aspirate the
  supernatant completely. Resuspend the pellet in \vol{1}{mL} fresh complete medium.}

\protostep{Count viable cells.}{Perform trypan blue exclusion count as in Procedure~A Step~11.
  Expected post-thaw viability: \textbf{\textgreater\,85\%}. If viability is \textless\,70\%,
  inspect records for incorrect freezing procedure or temperature deviations.}

\protostep{Plate and equilibrate.}{Transfer the cell suspension to the pre-warmed T-25 flask.
  Distribute medium evenly. Place in the incubator. \textbf{Do not disturb for 24\,h.}
  Change medium after 24\,h to remove any residual DMSO and dead cells.}

\protostep{Record thaw event.}{Log the thaw date, operator, vial ID, post-thaw viability, flask
  ID, and passage number ($P_{\text{stock}} + 1$) in the Cell Line Logbook and LIMS.}

\begin{tcolorbox}[infobox, title=\faClone\enspace Cell Bank Hierarchy]
  \textbf{Master Cell Bank (MCB):} Vials from P3--P5, stored in two separate LN\textsubscript{2}
  locations. Access restricted to Core Facility Manager only.\\[4pt]
  \textbf{Working Cell Bank (WCB):} Vials from P5--P10, available to authorised users.
  Replenished from MCB when fewer than 5 vials remain.\\[4pt]
  \textbf{Experimental stocks:} Vials at P\textgreater\,10 for daily experiments. Do \emph{not}
  use for generating new bank stocks.
\end{tcolorbox}
