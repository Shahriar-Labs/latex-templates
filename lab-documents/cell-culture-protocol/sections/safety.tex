\section{Safety \& Biosafety Requirements}

\begin{minipage}[t]{0.475\linewidth}
  \begin{tcolorbox}[cautionbox, title=\faExclamationTriangle\enspace Biosafety Level]
    Both HEK293T and CHO-K1 cell lines are classified \textbf{BSL-2}. HEK293T cells
    contain stably integrated adenovirus sequences. All work must be performed inside
    a certified Class II Biosafety Cabinet. Liquid biological waste must be
    autoclaved at 121\,$^\circ$C / 15\,psi for 30\,min before disposal.
  \end{tcolorbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.475\linewidth}
  \begin{tcolorbox}[timingbox, title=\faLock\enspace Personal Protective Equipment]
    \begin{itemize}
      \item Lab coat (fluid-resistant, long sleeve)
      \item Nitrile gloves -- double-glove when handling cryovials
      \item Safety glasses or face shield
      \item Closed-toe shoes; no sandals permitted
      \item Change gloves immediately on contamination or tear
    \end{itemize}
  \end{tcolorbox}
\end{minipage}

\vspace{6pt}

\begin{tcolorbox}[infobox, title=\faInfoCircle\enspace Chemical Hazards]
  \begin{itemize}
    \item \textbf{DMSO:} penetration enhancer -- avoid skin contact; use in fume hood.
    \item \textbf{Trypan Blue:} potential carcinogen -- dispose as chemical waste; do not mouth-pipette.
    \item \textbf{Liquid Nitrogen:} cryogenic burns and asphyxiation risk -- use insulated cryo-gloves and
          work in ventilated area; never seal LN\textsubscript{2} in airtight containers.
    \item \textbf{Ethanol (70\%):} flammable -- keep away from open flame.
  \end{itemize}
\end{tcolorbox}
