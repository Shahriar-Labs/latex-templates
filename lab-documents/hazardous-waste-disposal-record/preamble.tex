%% Styled section label (replaces \section)
\newcommand{\hwsection}[1]{%
  \vspace{6pt}%
  \begin{tcolorbox}[sectionbar]
    #1
  \end{tcolorbox}%
  \vspace{2pt}%
}

%% Coloured rule
\newcommand{\hrulecolor}[2][SlateInk]{%
  {\color{#1}\rule{\linewidth}{#2}}%
}

%% EPA / DOT waste code badge
\newcommand{\epacode}[1]{%
  \tikz[baseline=-0.5ex]\node[
    fill=HazardAmber, text=white,
    rounded corners=2.5pt,
    inner sep=3pt,
    font=\bfseries\footnotesize
  ]{#1};%
}

%% Status pill (Complete / Pending / Overdue)
\newcommand{\statuspill}[2]{%
  \tikz[baseline=-0.5ex]\node[
    fill=#1, text=white,
    rounded corners=3pt,
    inner sep=3pt,
    font=\bfseries\footnotesize
  ]{#2};%
}
