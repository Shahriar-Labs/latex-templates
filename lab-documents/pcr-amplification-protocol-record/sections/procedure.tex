\section{Procedure}

\subsection{Colony Pick and Template Preparation}

\stepitem{Using a sterile 10\,\uL\ pipette tip, touch 8 well-isolated, single colonies from
transformation plate TXP-0729-A. Streak each tip briefly onto a numbered master patch plate for
archival, then eject the tip into 30\,\uL\ nuclease-free water in a labelled 0.2\,mL tube.}

\stepitem{Vortex briefly and heat lysates at 95\degC{} for 8 minutes in a benchtop heat block to
release template DNA and inactivate nucleases. Cool on ice for 2 minutes before use.}

\stepitem{Label 13 PCR strip tubes: \texttt{1--8} (colonies), \texttt{NTC}, \texttt{PC}
(positive control), \texttt{S1--S2} (spares). Keep on a chilled rack throughout setup.}

\subsection{Reaction Assembly}

\stepitem{Prepare master mix per Section~3 in the pre-PCR workstation. Vortex 3\,s and
centrifuge briefly to collect contents.}

\stepitem{Aliquot 18\,\uL\ master mix into each of the 13 tubes.}

\stepitem{Move tubes to the template-addition bench. Add 2\,\uL\ of the appropriate template
(boiled lysate, water for NTC, or diluted plasmid for PC) to each tube, changing tips between
every sample. Cap tubes immediately after each addition.}

\stepitem{Centrifuge the strip briefly (5\,s pulse-spin) to collect contents at the tube base
and eliminate air bubbles.}

\subsection{Amplification and Post-Run Handling}

\stepitem{Load the strip into the Nimbus GT-96, close the heated lid, and start program
\texttt{NX303\_COLPCR} (Section~4). Record start time in the run log.}

\stepitem{At completion, retrieve tubes promptly from the 4\degC{} hold (within 4 hours) and
proceed directly to gel loading, or store at 4\degC{} short-term / $-$20\degC{} for longer storage.}

\begin{cautionbox}
Do not leave amplified product at room temperature after the hold step --- prolonged exposure
promotes non-specific re-annealing artefacts on the gel that can be mistaken for additional
bands.
\end{cautionbox}
