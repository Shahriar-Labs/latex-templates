% Nucleotide barcode chip
\newcommand{\ntchip}[2]{%
  \tikz[baseline=-0.5ex]{\node[fill=#1, text=white, font=\ttfamily\bfseries\scriptsize,
    inner sep=1.6pt, minimum width=4.3mm, minimum height=4.3mm, rounded corners=1pt]{#2};}%
}
\newcommand{\bA}{\ntchip{BaseA}{A}}
\newcommand{\bC}{\ntchip{BaseC}{C}}
\newcommand{\bG}{\ntchip{BaseG}{G}}
\newcommand{\bT}{\ntchip{BaseT}{T}}

% QC status chips
\newcommand{\passchip}{%
  \tikz[baseline=-0.5ex]{\node[fill=SeqPassBg, draw=SeqPass, text=SeqPass,
    font=\sffamily\bfseries\scriptsize, inner sep=2.5pt, rounded corners=2pt]{\faCheckCircle\ PASS};}%
}
\newcommand{\warnstatchip}{%
  \tikz[baseline=-0.5ex]{\node[fill=SeqWarnBg, draw=SeqWarn, text=SeqWarn,
    font=\sffamily\bfseries\scriptsize, inner sep=2.5pt, rounded corners=2pt]{\faExclamationTriangle\ WATCH};}%
}
\newcommand{\failchip}{%
  \tikz[baseline=-0.5ex]{\node[fill=SeqFailBg, draw=SeqFail, text=SeqFail,
    font=\sffamily\bfseries\scriptsize, inner sep=2.5pt, rounded corners=2pt]{\faTimesCircle\ FAIL};}%
}
