\section{Step-by-Step Procedure}

\subsection{Cell Harvest and Lysis}
\begin{enumerate}
  \item \textbf{Adherent Cells:} Wash with cold PBS. Add 350 $\mu$L Buffer RLT directly to the dish, scrape, and transfer lysate to a 1.5 mL tube.
  \item \textbf{Suspension Cells:} Centrifuge cells at $300 \times g$ for 5 minutes. Resuspend pellet in 350 $\mu$L of Buffer RLT.
  \item \textbf{Homogenization:} Homogenize the lysate by passing it through a sterile 20-gauge needle attached to a 1 mL syringe 5 times, or by centrifuging through a QIAshredder column ($16,000 \times g$, 2 min).
\end{enumerate}

\subsection{RNA Binding and On-Column DNase Digestion}
\begin{enumerate}
  \item Add 350 $\mu$L 70\% ethanol to the homogenized lysate; mix by pipetting. Transfer 700 $\mu$L to an RNeasy spin column. Centrifuge at $\ge 8000 \times g$ for 15s; discard flow-through.
  \item Add 350 $\mu$L Buffer RW1; centrifuge at $\ge 8000 \times g$ for 15s; discard flow-through.
  \item Pipette DNase mix (10 $\mu$L DNase I stock + 70 $\mu$L Buffer RDD, mixed by inversion) directly onto the membrane. Incubate at room temperature for 15 min.
  \item Add 350 $\mu$L Buffer RW1; centrifuge at $\ge 8000 \times g$ for 15s; discard flow-through.
\end{enumerate}

\subsection{Column Washing and Elution}
\begin{enumerate}
  \item Add 500 $\mu$L Buffer RPE; centrifuge at $\ge 8000 \times g$ for 15s (discard flow-through). Add another 500 $\mu$L Buffer RPE; centrifuge for 2 min.
  \item Centrifuge empty column at $16,000 \times g$ for 1 minute to dry the membrane.
  \item Place column in a 1.5 mL tube. Add 30--50 $\mu$L RNase-free water. Incubate for 1 minute. Centrifuge at $\ge 8000 \times g$ for 1 minute to elute. Store at $-80^\circ$C.
\end{enumerate}
