% ---------- Chapter / section styling ----------
\renewcommand{\thechapter}{\Roman{chapter}}

\titleformat{\chapter}[display]
  {\normalfont\centering}
  {\color{uigold}\Large\bfseries BAB \thechapter}
  {8pt}
  {\centering\color{uinavy}\Huge\bfseries}
  [\vspace{4pt}\begin{center}\color{uigold}\rule{4cm}{1.2pt}\end{center}\vspace{18pt}]

  {\normalfont\large\bfseries\color{uinavy}}
  {\thesection}{0.8em}{}

  {\thesubsection}{0.8em}{}

\titlespacing*{\section}{0pt}{16pt}{8pt}
