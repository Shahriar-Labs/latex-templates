% ── Chapter / section formatting ───────────────────────────────
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\Huge\bfseries\color{edinburghblue}}
  {\filleft\fontsize{72}{72}\selectfont\thechapter}
  {0pt}
  {\Huge}
  [{\color{saltireblue}\titlerule[1.5pt]}\vspace{-\baselineskip}]
\titlespacing*{\chapter}{0pt}{40pt}{40pt}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{edinburghblue}}
  {\thesection}
  {1em}
  {}
\titlespacing*{\section}{0pt}{18pt}{8pt}

\titleformat{\subsection}
  {\normalfont\large\bfseries\color{edinburghblue!85!black}}
  {\thesubsection}
  {1em}
  {}
\titlespacing*{\subsection}{0pt}{14pt}{6pt}

% ── Headers & footers ──────────────────────────────────────────

  \fancyhead[R]{\small\color{edinburghblue!60}\thepage}
