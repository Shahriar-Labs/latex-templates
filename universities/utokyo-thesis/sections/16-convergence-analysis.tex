We now state the main theoretical result of this thesis. The proof is deferred to
Appendix~\ref{app:proofs} for brevity; we sketch the key steps here.

\begin{description}[leftmargin=2em,style=nextline]
  \item[Theorem 1 (NPG Convergence).] \emph{Under Assumptions~1--4, let $\{\theta_t\}_{t=1}^T$
    be generated by the NPG algorithm (Equation~\ref{eq:npg}) with $\Psi_t$ satisfying
    $\mathbb{E}[\|\Psi_t\|^2] \leq C_1 + C_2\|\nabla\mathcal{L}(\theta_t)\|^2$ for constants
    $C_1, C_2 > 0$, and with learning rate $\eta_t = \eta/\sqrt{T}$ for $\eta > 0$ sufficiently
    small. Then}
    \begin{equation}
      \frac{1}{T}\sum_{t=1}^{T} \mathbb{E}\bigl[\|\nabla\mathcal{L}(\theta_t)\|^2\bigr]
        \leq \frac{2\sqrt{\Delta}}{\sqrt{T}} + \mathcal{O}\!\left(\frac{1}{T}\right),
      \label{eq:thm1}
    \end{equation}
    \emph{where $\Delta = \mathcal{L}(\theta_1) - \mathcal{L}_*$.}
    \label{thm:convergence}
\end{description}

The proof proceeds in three steps. First, we use the $L$-smoothness of $\mathcal{L}$ to
bound the per-iteration decrease:
\begin{equation}
  \mathcal{L}(\theta_{t+1}) \leq \mathcal{L}(\theta_t)
    - \eta_t \langle\nabla\mathcal{L}(\theta_t),\, P_t^{-1/2}\Psi_t\rangle
    + \frac{L\eta_t^2}{2}\|P_t^{-1/2}\Psi_t\|^2.
  \label{eq:smoothness}
\end{equation}

Second, we bound the inner-product term in expectation, leveraging the fact that the
normalization $P_t^{-1/2}$ ensures that $\|\Psi_t\|$ is controlled regardless of the
conditioning of $P_t$:
\begin{equation}
  \mathbb{E}_t\bigl[\langle\nabla\mathcal{L}(\theta_t),\, P_t^{-1/2}\Psi_t\rangle\bigr]
    \geq \frac{1}{2}\|\nabla\mathcal{L}(\theta_t)\|^2 - \mathcal{O}\!\left(\frac{1}{\sqrt{t}}\right).
  \label{eq:inner}
\end{equation}

Third, we sum the per-iteration bounds over $t = 1,\dots,T$ and apply a telescoping argument,
yielding the $\mathcal{O}(1/\sqrt{T})$ rate in Equation~\ref{eq:thm1}. Table~\ref{tab:rates}
compares this result against existing bounds for Adam and SGD.
