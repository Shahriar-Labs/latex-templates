% --- Encoding & Fonts --------------------------------------------------
\usepackage[T1]{fontenc}
\usepackage{tgpagella}

% --- Geometry & Spacing ------------------------------------------------

            includefoot,includehead]{geometry}

\onehalfspacing

% --- Headers & Footers -------------------------------------------------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\scshape\color{softgrey}\leftmark}

  \fancyfoot[C]{\small\color{softgrey}\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% --- Chapter & Section Titles ------------------------------------------
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{anucrimson}}
  {\fontsize{56}{60}\selectfont\color{anugold}\thechapter}
  {0pt}
  {\Huge}
  [\vspace{6pt}{\color{anucrimson}\rule{\textwidth}{1.5pt}}]
\titlespacing*{\chapter}{0pt}{50pt}{36pt}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{anucrimson}}
  {\thesection}
  {1em}
  {}
\titlespacing*{\section}{0pt}{18pt plus 4pt minus 2pt}{10pt plus 2pt minus 2pt}

\titleformat{\subsection}
  {\normalfont\large\bfseries}
  {\thesubsection}
  {1em}
  {}
\titlespacing*{\subsection}{0pt}{14pt plus 2pt minus 2pt}{8pt plus 2pt minus 2pt}
