\chapter{Proof of Theorem~1}\label{app:proofs}

We provide a detailed proof of Theorem~\ref{thm:convergence}. Recall the standing assumptions:
$\mathcal{L}$ is $L$-smooth (Assumption~1), gradients are unbiased with bounded variance
(Assumptions~2--3), and $\mathcal{L}$ is bounded below (Assumption~4).

\begin{proof}
From the $L$-smoothness of $\mathcal{L}$ and the NPG update $\theta_{t+1} = \theta_t -
\eta_t P_t^{-1/2}\Psi_t$, a standard quadratic upper bound yields
\begin{equation}
  \mathcal{L}(\theta_{t+1}) \leq \mathcal{L}(\theta_t)
    - \eta_t \langle\nabla\mathcal{L}(\theta_t),\, P_t^{-1/2}\Psi_t\rangle
    + \frac{L\eta_t^2}{2}\|P_t^{-1/2}\Psi_t\|^2.
  \label{eq:pf1}
\end{equation}

Let $\mathbb{E}_t[\cdot] = \mathbb{E}[\cdot \mid \theta_t, P_t]$ denote the conditional
expectation. Taking $\mathbb{E}_t$ of both sides of~(\ref{eq:pf1}) and applying the tower
property:
\begin{align}
  \mathbb{E}_t[\mathcal{L}(\theta_{t+1})]
    &\leq \mathcal{L}(\theta_t)
      - \eta_t \mathbb{E}_t\bigl[\langle\nabla\mathcal{L}(\theta_t),\,
        P_t^{-1/2}\Psi_t\rangle\bigr]
      + \frac{L\eta_t^2}{2}\mathbb{E}_t\bigl[\|P_t^{-1/2}\Psi_t\|^2\bigr].
  \label{eq:pf2}
\end{align}

By the construction of the preconditioner, $P_t \succ 0$, so $P_t^{-1/2}$ is well-defined.
Moreover, by the boundedness condition on $\Psi_t$, there exist constants $C_1, C_2 > 0$ such
that
\begin{equation}
  \mathbb{E}_t[\|\Psi_t\|^2] \leq C_1 + C_2\|\nabla\mathcal{L}(\theta_t)\|^2.
  \label{eq:pf3}
\end{equation}

The critical step is bounding the inner product. Since $P_t^{-1/2}$ is $\mathcal{F}_{t-1}$-measurable
(it depends only on gradients up to $t-1$), and $\Psi_t$ is an unbiased estimator of the
normalized gradient direction, we have:
\begin{align}
  \mathbb{E}_t\bigl[\langle\nabla\mathcal{L}(\theta_t),\, P_t^{-1/2}\Psi_t\rangle\bigr]
    &= \langle\nabla\mathcal{L}(\theta_t),\,
        P_t^{-1/2}\,\mathbb{E}_t[\Psi_t]\rangle \nonumber\\
    &\geq \frac{\lambda_{\min}(P_t^{-1/2})}{2}\|\nabla\mathcal{L}(\theta_t)\|^2
        - \frac{1}{2}\|P_t^{-1/2}\|\,\mathbb{E}_t[\|\Psi_t - \mathbb{E}_t[\Psi_t]\|^2],
  \label{eq:pf4}
\end{align}
where $\lambda_{\min}(P_t^{-1/2})$ is the minimum eigenvalue of $P_t^{-1/2}$. By the
normalization property of NPG, $P_t^{-1/2}$ is bounded both above and away from zero
uniformly in $t$, so $\lambda_{\min}(P_t^{-1/2}) \geq \gamma$ for some $\gamma > 0$.

Substituting (\ref{eq:pf3}) and (\ref{eq:pf4}) into (\ref{eq:pf2}), setting $\eta_t =
\eta/\sqrt{T}$, and summing from $t=1$ to $T$:
\begin{align}
  \sum_{t=1}^{T} \mathbb{E}\bigl[\|\nabla\mathcal{L}(\theta_t)\|^2\bigr]
    &\leq \frac{2(\mathcal{L}(\theta_1) - \mathcal{L}_*)}{\gamma\eta\sqrt{T}}
        + \mathcal{O}(1),
\end{align}
where we have used the telescoping property and the lower bound $\mathcal{L}_*$. Dividing by
$T$ yields the claimed $\mathcal{O}(1/\sqrt{T})$ rate.
\end{proof}

{\small\textit{End of proof.}}
\end{center}
