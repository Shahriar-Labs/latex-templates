% ── Encoding and fonts ──────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella}
\usepackage{newpxmath}

% ── Geometry ────────────────────────────────────────────────────────
\usepackage[paper=letterpaper,inner=1in,outer=1in,top=1in,bottom=1in]{geometry}

% ── Spacing ─────────────────────────────────────────────────────────
\usepackage{setspace}
\onehalfspacing
\setlength{\parindent}{1.5em}
\setlength{\parskip}{0pt}

% ── Colors ──────────────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{ColumbiaBlue}{HTML}{1D4F91}
\definecolor{DarkGray}{HTML}{333333}
\definecolor{MidGray}{HTML}{777777}
\definecolor{LightGray}{HTML}{E8E8E8}

% ── Graphics and figures ────────────────────────────────────────────
\usepackage{graphicx}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,shapes,fit,backgrounds,calc,patterns}
\usepackage{caption}
\usepackage{subcaption}

% ── Tables ──────────────────────────────────────────────────────────
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{array}

% ── Mathematics ─────────────────────────────────────────────────────
\let\Bbbk\relax           % newpxmath already defines \Bbbk; avoid amssymb clash
\usepackage{amsmath,amssymb,amsfonts}
\let\openbox\relax        % newpxmath defines \openbox; let amsthm redefine cleanly
\usepackage{amsthm}       % provides the proof environment + \newtheorem
\usepackage{mathtools}

% ── Headers and footers ─────────────────────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE]{\small\itshape\color{MidGray}\leftmark}
\fancyhead[RO]{\small\itshape\color{MidGray}\rightmark}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% ── Chapter and section titles ──────────────────────────────────────
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries\centering}
  {\chaptertitlename\ \thechapter}
  {20pt}
  {\Huge}
  [{\color{ColumbiaBlue}\titlerule[2pt]}\vspace{10pt}]
\titlespacing*{\chapter}{0pt}{50pt}{40pt}

\titleformat{\section}
  {\normalfont\LARGE\bfseries}
  {\thesection}
  {1em}
  {}
\titlespacing*{\section}{0pt}{24pt}{12pt}

\titleformat{\subsection}
  {\normalfont\Large\bfseries}
  {\thesubsection}
  {1em}
  {}
\titlespacing*{\subsection}{0pt}{18pt}{8pt}

% ── Table of contents styling ───────────────────────────────────────
\usepackage{tocloft}
\renewcommand{\cftchapfont}{\bfseries}
\renewcommand{\cftsecfont}{}
\renewcommand{\cftchappagefont}{\bfseries}
\setlength{\cftbeforechapskip}{8pt}

% ── Lists ───────────────────────────────────────────────────────────
\usepackage{enumitem}
\setlist{nosep,leftmargin=*}

% ── Hyperlinks ──────────────────────────────────────────────────────
\usepackage[colorlinks=true,linkcolor=black,citecolor=ColumbiaBlue,urlcolor=ColumbiaBlue,bookmarksnumbered=true]{hyperref}

% ── Icons and symbols ───────────────────────────────────────────────
\usepackage{fontawesome5}

% ── Miscellaneous ───────────────────────────────────────────────────
\usepackage{ragged2e}
