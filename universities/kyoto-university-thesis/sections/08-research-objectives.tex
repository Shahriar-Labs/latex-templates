\section{Research Objectives}

\begin{keybox}[title=Research Questions]
\begin{enumerate}[leftmargin=*,itemsep=2pt]
\item Can a machine learning model trained on molecular descriptors
  reliably predict the $V_{oc}$-deficit reduction conferred by a candidate
  SAM interlayer?
\item Does Bayesian optimization over such a surrogate model identify
  previously unreported candidates that outperform literature benchmarks?
\item What is the microscopic origin --- bulk or interfacial --- of any
  observed efficiency and stability improvement?
\end{enumerate}
\end{keybox}
