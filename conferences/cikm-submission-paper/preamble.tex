% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, captions, citations (natbib)
% and hyperlinks, and the ACM format depends on them, so this file adds nothing that
% changes those. In particular: no geometry, no font packages, no titlesec, no caption
% settings, no cite package (it clashes with acmart's natbib), and no amssymb (acmart's
% math fonts already define its symbols, and loading it stops the build).

\usepackage{tabularx}
\usepackage{tikz}
\usetikzlibrary{positioning,calc,shapes.geometric,arrows.meta}

\definecolor{cikmblue}{HTML}{1A365D}
\definecolor{cikmgray}{HTML}{4A5568}
\definecolor{cikmred}{HTML}{C53030}
\definecolor{cikmbg}{HTML}{F7FAFC}
