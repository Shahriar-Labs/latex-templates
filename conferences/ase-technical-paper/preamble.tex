% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, captions, citations (natbib)
% and hyperlinks, and the ACM format depends on them, so this file adds nothing that
% changes those. In particular: no geometry, no font packages (acmart already loads its own
% math fonts), no titlesec, no caption settings, no enumitem, and no amssymb (its symbols are
% already defined, and loading it stops the build).

\usepackage{multirow}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,calc,shapes.geometric,backgrounds}
\usepackage{pgfplots}
\pgfplotsset{compat=1.17}

\definecolor{aseaccent}{RGB}{163,52,44}
\definecolor{asegray}{RGB}{95,95,100}
\definecolor{aselight}{RGB}{238,241,245}
