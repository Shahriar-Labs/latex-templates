% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, captions, author block and
% hyperlinks, and the ACM format depends on them, so this file adds nothing that changes
% those. In particular: no geometry, no font packages, no authblk, no caption settings, and
% no amssymb (acmart's math fonts already define its symbols, and loading it stops the
% build). cleveref has to come after hyperref, which acmart loads first, so it is safe here.

\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.17}
\usetikzlibrary{positioning,calc,shapes.geometric,backgrounds,patterns}
\usepackage[capitalize]{cleveref}

\definecolor{poplblue}{RGB}{31,58,102}
\definecolor{poplgray}{RGB}{90,90,90}
\definecolor{poplrule}{RGB}{200,200,200}
\definecolor{poplaccent}{RGB}{176,58,46}
