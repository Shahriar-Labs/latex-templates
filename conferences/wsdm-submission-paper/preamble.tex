% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, captions, citations (natbib)
% and hyperlinks, and the ACM format depends on them, so this file adds nothing that
% changes those. In particular: no geometry, no font packages, no titlesec, no caption
% settings, and no amssymb (acmart's math fonts already define its symbols, and loading it
% stops the build).

\usepackage{tikz}
\usetikzlibrary{positioning,calc,shapes.geometric,decorations.pathmorphing,backgrounds,patterns}
\usepackage{pgfplots}
\pgfplotsset{compat=1.17}

\definecolor{wsdmteal}{RGB}{31,122,140}
\definecolor{wsdmgray}{RGB}{95,95,100}
