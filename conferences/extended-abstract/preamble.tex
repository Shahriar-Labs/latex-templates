% ─── Geometry ───────────────────────────────────────────────────────────────
\usepackage[
  a4paper,
  top=18mm, bottom=18mm,
  left=15mm, right=15mm,
  columnsep=6mm
]{geometry}

% ─── Fonts & encoding ───────────────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{tgpagella}          % Palatino-style serif (elegant, classic)
\usepackage{microtype}

% ─── Colour ─────────────────────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{accent}{HTML}{1A4E8C}   % steel-blue accent
\definecolor{muted}{HTML}{555555}    % body-text neutral
\definecolor{rule}{HTML}{C0C8D8}     % light rule

% ─── Core packages ──────────────────────────────────────────────────────────
\usepackage{amsmath,amssymb}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{enumitem}
\usepackage{graphicx}
\usepackage{caption}
\usepackage{titlesec}
\usepackage{parskip}
\usepackage{hyperref}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% ─── Spacing fix (required when parskip is loaded) ──────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{3.5pt plus 1pt minus 1pt}

% ─── Section headings ────────────────────────────────────────────────────────
\titleformat{\section}
  {\normalfont\normalsize\bfseries\color{accent}}
  {}
  {0pt}
  {\MakeUppercase}
  [\vspace{-2pt}\color{rule}\hrule height 0.6pt\vspace{2pt}]

\titlespacing*{\section}{0pt}{7pt}{3pt}

% ─── Caption style ───────────────────────────────────────────────────────────
\captionsetup{
  font={small,color=muted},
  labelfont={bf,color=accent},
  skip=4pt
}

% ─── Hyperref ────────────────────────────────────────────────────────────────
\hypersetup{
  colorlinks=true,
  urlcolor=accent,
  citecolor=accent,
  linkcolor=accent,
  pdftitle={Adaptive Gradient Pruning for Energy-Efficient Neural Inference},
  pdfauthor={Elena Vasquez; Dmitri Orlov; Jun-Yi Chen}
}

% ─── Title macros ────────────────────────────────────────────────────────────
\makeatletter
\newcommand{\papertitle}[1]{\def\@papertitle{#1}}
\newcommand{\paperauthors}[1]{\def\@paperauthors{#1}}
\newcommand{\paperaffil}[1]{\def\@paperaffil{#1}}
\newcommand{\paperemail}[1]{\def\@paperemail{#1}}

\newcommand{\maketitleblock}{%
  \twocolumn[{%
    \begin{center}
      {\LARGE\bfseries\color{accent} \@papertitle\par}
      \vspace{6pt}
      {\normalsize \@paperauthors\par}
      \vspace{2pt}
      {\small\color{muted} \@paperaffil\par}
      \vspace{2pt}
      {\small\color{muted} \texttt{\@paperemail}\par}
    \end{center}
    \vspace{2pt}
    \color{rule}\hrule height 1pt
    \vspace{8pt}
  }]
}
\makeatother

% ─── Abstract environment ─────────────────────────────────────────────────────
\renewenvironment{abstract}{%
  \small\color{muted}\itshape
  \textbf{\color{accent}Abstract.}\ \ignorespaces
}{%
  \par
}

% ─── Keywords macro ───────────────────────────────────────────────────────────
\newcommand{\keywords}[1]{%
  \vspace{2pt}
  {\small\color{muted}\textbf{Keywords:} #1\par}
  \vspace{6pt}
  \color{rule}\hrule height 0.4pt
  \vspace{6pt}
}

% ─── References style ────────────────────────────────────────────────────────
\usepackage[numbers,sort&compress]{natbib}
\setlength{\bibsep}{2pt}
\renewcommand{\bibfont}{\small}
