% Packages, colours and commands for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, text colour, section headings, spacing and
% hyperlinks, and the ACM format depends on them, so this file adds nothing that changes
% those. In particular: no geometry, no titlesec, no \parskip or \parindent changes, no
% global \color, and no amssymb (acmart's math fonts already define its symbols, and
% loading it stops the build).

\usepackage{tikz}
\usetikzlibrary{positioning,calc,shapes.geometric,backgrounds,arrows.meta}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\definecolor{inkblack}{RGB}{26,26,26}
\definecolor{steelblue}{RGB}{40,71,109}
\definecolor{mutedteal}{RGB}{58,124,120}
\definecolor{papergray}{RGB}{110,110,110}
