% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, captions, citations (natbib),
% author block and hyperlinks, and the ACM format depends on them, so this file adds
% nothing that changes those. In particular: no geometry, no font packages, no authblk, no
% caption settings, and no amssymb (acmart's math fonts already define its symbols, and
% loading it stops the build).

\usepackage{tikz}
\usetikzlibrary{positioning,calc,shapes.geometric,arrows.meta,backgrounds}

\definecolor{sigirteal}{RGB}{0,110,110}
\definecolor{sigirgray}{RGB}{95,95,100}
