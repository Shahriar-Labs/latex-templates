% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, citations (natbib), running
% heads and hyperlinks, and MICRO asks you not to change the template's settings, so this
% file adds nothing that changes those. In particular: no geometry, no font packages, no
% titlesec, no fancyhdr, no cite package (it clashes with acmart's natbib), and no amssymb
% (acmart's math fonts already define its symbols, and loading it stops the build).

\usepackage{tikz}
\usetikzlibrary{positioning,calc}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\definecolor{archrust}{RGB}{176,90,42}
\definecolor{archgray}{RGB}{100,105,112}
