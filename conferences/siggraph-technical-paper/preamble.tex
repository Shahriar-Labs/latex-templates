% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, citations (natbib) and
% hyperlinks, and the ACM format depends on them, so this file adds nothing that changes
% those. In particular: no geometry, no titlesec, no cite package (it clashes with acmart's
% natbib), and no amssymb or amsfonts (their symbols are already defined, and loading them
% stops the build). tikz is on ACM's TAPS list of accepted packages; check anything you add.

\usepackage{tikz}
\usetikzlibrary{positioning,calc}

\definecolor{primarycolor}{rgb}{0.1,0.25,0.45}
\definecolor{secondarycolor}{rgb}{0.15,0.15,0.15}
\definecolor{accentcolor}{rgb}{0.65,0.15,0.15}
\definecolor{lightgray}{rgb}{0.96,0.96,0.96}
