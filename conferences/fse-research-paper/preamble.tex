% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, section headings, citations (natbib) and
% hyperlinks, and the ACM format depends on them, so this file adds nothing that changes
% those. In particular: no geometry, no font packages (acmart already loads its own math
% fonts), no titlesec, no cite package (it clashes with acmart's natbib), and no amssymb or
% amsfonts (their symbols are already defined, and loading them stops the build).

\usepackage{colortbl}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{positioning, calc, shapes.geometric}

\definecolor{primarycolor}{rgb}{0.08,0.20,0.38}
\definecolor{secondarycolor}{rgb}{0.50,0.00,0.00}
\definecolor{neutraldark}{rgb}{0.15,0.15,0.15}
