% Packages, colours and custom commands for this template.
% ============================================================
%  ICSE Technical Paper Template
%  International Conference on Software Engineering
%  Edit and compile online at https://letx.app
%  Local:  latexmk -pdf main.tex
% ============================================================

% ── Encoding & fonts ────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{times}
\usepackage{microtype}

% ── Page geometry ───────────────────────────────────────────
\usepackage[
  letterpaper,
  top=25mm, bottom=25mm,
  left=20mm, right=20mm,
  columnsep=8mm
]{geometry}

% ── Colours ─────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{accent}{HTML}{1A5276}
\definecolor{muted}{HTML}{555555}
\definecolor{rule}{HTML}{C0C8D8}

% ── Core packages ───────────────────────────────────────────
\usepackage{amsmath,amssymb}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{enumitem}
\usepackage{graphicx}
\usepackage{caption}
\usepackage{subcaption}
\usepackage{titlesec}
\usepackage{parskip}
\usepackage{hyperref}
\usepackage{listings}
\usepackage{float}

% ── Spacing ─────────────────────────────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{3pt plus 1pt minus 1pt}

% ── Section headings ────────────────────────────────────────
\titleformat{\section}
  {\normalfont\normalsize\bfseries}
  {\thesection}{1em}{}
  [\vspace{-2pt}\color{rule}\hrule height 0.4pt\vspace{2pt}]

\titlespacing*{\section}{0pt}{8pt}{3pt}

\titleformat{\subsection}
  {\normalfont\normalsize\bfseries\itshape}
  {\thesubsection}{1em}{}

\titlespacing*{\subsection}{0pt}{6pt}{2pt}

% ── Caption style ───────────────────────────────────────────
\captionsetup{
  font={small,color=muted},
  labelfont={bf},
  skip=4pt
}

% ── Hyperref ────────────────────────────────────────────────
\hypersetup{
  colorlinks=true,
  urlcolor=accent,
  citecolor=accent,
  linkcolor=accent,
  pdftitle={ICSE 2026 Technical Paper},
  pdfauthor={}
}

% ── Code listings ───────────────────────────────────────────
\lstset{
  basicstyle=\footnotesize\ttfamily,
  breaklines=true,
  frame=single,
  rulecolor=\color{rule},
  numbers=left,
  numberstyle=\tiny\color{muted},
  xleftmargin=1em,
  framexleftmargin=1em
}

% ── Title macros ────────────────────────────────────────────
\makeatletter
\newcommand{\papertitle}[1]{\def\@papertitle{#1}}
\newcommand{\paperauthors}[1]{\def\@paperauthors{#1}}
\newcommand{\paperaffil}[1]{\def\@paperaffil{#1}}
\newcommand{\paperemail}[1]{\def\@paperemail{#1}}

\newcommand{\maketitleblock}{%
  \twocolumn[{%
    \begin{center}
      {\LARGE\bfseries \@papertitle\par}
      \vspace{8pt}
      {\normalsize \@paperauthors\par}
      \vspace{3pt}
      {\small\color{muted} \@paperaffil\par}
      \vspace{3pt}
      {\small\color{muted} \texttt{\@paperemail}\par}
    \end{center}
    \vspace{4pt}
    \color{rule}\hrule height 1pt
    \vspace{10pt}
  }]
}
\makeatother

% ── Abstract environment ────────────────────────────────────
\renewenvironment{abstract}{%
  \small
  \textbf{ABSTRACT}\ \ignorespaces
}{%
  \par
  \vspace{8pt}
  \color{rule}\hrule height 0.4pt
  \vspace{8pt}
}

% ── Keywords macro ──────────────────────────────────────────
\newcommand{\keywords}[1]{%
  {\small\textbf{CCS Concepts:}\enspace #1\par}
}

% ── ACM Reference format ────────────────────────────────────
\newcommand{\acmreference}[1]{%
  {\small\textbf{ACM Reference Format:}\enspace #1\par}
}
