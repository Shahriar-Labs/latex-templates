% Packages and colours for this template, on top of what acmart already loads.
%
% acmart sets its own fonts, page geometry, running heads, section headings, line numbers
% (the `review` option) and hyperlinks, and the ACM format depends on them, so this file
% adds nothing that changes those. In particular: no geometry, no font packages, no
% fancyhdr, no titlesec, no lineno, and no amssymb (acmart's math fonts already define its
% symbols, and loading it stops the build).

\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{positioning,calc,shapes.geometric,decorations.pathmorphing,backgrounds,patterns}

\definecolor{primarycolor}{HTML}{1A365D}
\definecolor{secondarycolor}{HTML}{2B6CB0}
\definecolor{accentcolor}{HTML}{2D3748}
