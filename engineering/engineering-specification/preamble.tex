\newcommand{\docctrl}[5]{%
  \noindent%
  \begin{tcolorbox}[
    enhanced,
    sharp corners,
    colback=accent,
    colframe=accent,
    boxrule=0pt,
    left=0pt,right=0pt,top=0pt,bottom=0pt,
    before skip=0pt,after skip=14pt,
  ]%
  \begin{tikzpicture}
    \node[inner sep=0pt,outer sep=0pt,anchor=north west] (box) at (0,0) {%
      \begin{minipage}{\dimexpr\linewidth-10pt\relax}
        \vspace{10pt}
        \begin{minipage}{0.62\linewidth}
          \color{white}\LARGE\bfseries #1
          \par\vspace{2pt}
          \color{accentlight!50!white}\normalsize #2
        \end{minipage}%
        \hfill%
        \begin{minipage}{0.34\linewidth}
          \raggedleft
          \color{white}\footnotesize
          \textbf{Document:} #3\\[1pt]
          \textbf{Revision:} #4\\[1pt]
          \textbf{Date:} #5
        \end{minipage}
        \vspace{10pt}
      \end{minipage}%
    };
    \draw[accentlight,line width=4pt] (box.south west) -- (box.south east);
  \end{tikzpicture}
  \end{tcolorbox}%
}
