% =============================================================================
% CALCULATION ENGINE --- values below are derived automatically
% =============================================================================
\pgfmathsetmacro{\TxLineLoss}{\TxFeeder+\TxConnector}
\pgfmathsetmacro{\RxLineLoss}{\RxFeeder+\RxConnector}
\pgfmathsetmacro{\FSPL}{92.45+20*ln(\FreqGHz)/ln(10)+20*ln(\PathKm)/ln(10)}
\pgfmathsetmacro{\PropagationLoss}{\FSPL+\GasLoss+\PolLoss+\OtherLoss}
\pgfmathsetmacro{\EIRP}{\TxPower-\TxLineLoss+\TxGain}
\pgfmathsetmacro{\ReceivedPower}{\EIRP-\PropagationLoss+\RxGain-\RxLineLoss}
\pgfmathsetmacro{\NoiseFloor}{-174+10*(ln(\BandwidthMHz)/ln(10)+6)+\NoiseFigure}
\pgfmathsetmacro{\RxThreshold}{\NoiseFloor+\RequiredCN+\ImplementationLoss}
\pgfmathsetmacro{\CarrierNoise}{\ReceivedPower-\NoiseFloor}
\pgfmathsetmacro{\FadeMargin}{\ReceivedPower-\RxThreshold}
\pgfmathsetmacro{\MarginSurplus}{\FadeMargin-\RequiredMargin}
\pgfmathsetmacro{\RainResidual}{\FadeMargin-\RainAttenuation}
\pgfmathsetmacro{\FresnelRadius}{sqrt((0.299792458/\FreqGHz)*(\PathKm*1000/4))}
\pgfmathsetmacro{\FresnelClearance}{0.6*\FresnelRadius}
