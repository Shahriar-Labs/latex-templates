\subsection{Numerical checks}
\begin{align*}
L_{fs} &= 92.45+20\log_{10}(\FreqGHz)+20\log_{10}(\PathKm)
       = \dB{\FSPL},\\
\mathrm{EIRP} &= \TxPower-\plainone{\TxLineLoss}+\TxGain
       = \dBm{\EIRP},\\
P_r &= \plainone{\EIRP}-\plainone{\PropagationLoss}
       +\RxGain-\plainone{\RxLineLoss}
       = \dBm{\ReceivedPower},\\
N &= -174+10\log_{10}(\BandwidthMHz\times10^6)+\NoiseFigure
       = \dBm{\NoiseFloor},\\
P_{th} &= \plainone{\NoiseFloor}+\RequiredCN+\ImplementationLoss
       = \dBm{\RxThreshold},\\
M_f &= (\plainone{\ReceivedPower})-(\plainone{\RxThreshold})
       = \dB{\FadeMargin}.
\end{align*}
