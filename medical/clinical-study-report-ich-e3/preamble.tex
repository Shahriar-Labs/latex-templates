\setlength{\parindent}{0pt}

\usepackage{titlesec}
\titleformat{\section}
  {\Large\bfseries\color{accent}}
  {\thesection}
  {1em}
  {}
  [{\color{grayrule}\titlerule}]
\titleformat{\subsection}
  {\large\bfseries\color{secondary}}
  {\thesubsection}
  {1em}
  {}
\titleformat{\subsubsection}
  {\normalsize\bfseries\color{secondary}}
  {\thesubsubsection}
  {1em}
  {}
