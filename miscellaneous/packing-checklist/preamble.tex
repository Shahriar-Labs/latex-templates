% ── Custom checkbox ──
\newcommand{\chkbox}{%
  \begin{tikzpicture}[baseline=-0.45ex]
    \draw[accent!50, line width=0.35pt, rounded corners=0.3pt, fill=white]
      (0,0) rectangle (0.38,0.38);
  \end{tikzpicture}%
}
