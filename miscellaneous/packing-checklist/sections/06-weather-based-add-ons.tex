\section{\faCloud\ \ Weather-Based Add-ons}

{\footnotesize\color{medium} Tick the forecast(s) that apply to your trip:\par}

\begin{hlbox}
\textbf{\color{accent}\faSun\ \ Hot \& Sunny Weather}
\begin{checklist}
  \item Extra sunscreen (SPF 50+)
  \item Wide-brimmed sun hat
  \item Polarised sunglasses
  \item Lightweight, breathable tops
\end{checklist}

\vspace{2pt}
\textbf{\color{accent}\faUmbrella\ \ Rainy Weather}
\begin{checklist}
  \item Rain jacket / waterproof shell
  \item Compact travel umbrella
  \item Waterproof shoes / boots
  \item Quick-dry clothing
  \item Waterproof pouch for electronics
\end{checklist}

\vspace{2pt}
\textbf{\color{accent}\faSnowflake\ \ Cold Weather}
\begin{checklist}
  \item Thermal base layers (top \& bottom)
  \item Warm gloves \& beanie
  \item Insulated jacket / coat
  \item Wool-blend socks
\end{checklist}

\vspace{2pt}
\textbf{\color{accent}\faWater\ \ Beach Trip}
\begin{checklist}
  \item Swimsuit ($\times$2)
  \item Beach towel
  \item Flip-flops / sandals
  \item Snorkel \& mask
  \item Waterproof phone pouch
\end{checklist}
\end{hlbox}
