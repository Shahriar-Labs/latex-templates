% ── Claim macro ─────────────────────────────────────────────
\newcounter{claimcnt}
\setcounter{claimcnt}{0}
\newcommand{\claim}[1]{%
  \stepcounter{claimcnt}%
  \noindent\hangindent=1.8em\hangafter=1%
  \textbf{\theclaimcnt.}\enspace#1\par\vspace{5pt}%
}

% ── Title block macro ───────────────────────────────────────
\newcommand{\titlerow}[2]{%
  \noindent\begin{tabularx}{\linewidth}{%
    @{}>{{\color{muted}\small\bfseries}}l@{\hspace{7mm}}>{\raggedright\arraybackslash}X@{}%
  }%
    #1 & #2 \\
  \end{tabularx}%
  \vspace{2pt}%
}
