% ── Core layout ──────────────────────────────────────────────
\usepackage[
  top=2.4cm, bottom=2.8cm,
  left=2.8cm, right=2.8cm
]{geometry}

% ── Typography (classic serif via TeX Gyre Pagella) ──────────
\usepackage{tgpagella}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}

% ── Colour palette ───────────────────────────────────────────
\usepackage{xcolor}
\definecolor{accent}{HTML}{1E3A5F}   % deep navy — single accent
\definecolor{rulegray}{HTML}{B0B8C4} % subtle rule colour
\definecolor{bodytext}{HTML}{1A1A1A} % near-black body

% ── Packages ─────────────────────────────────────────────────
\usepackage{microtype}
\usepackage{setspace}
\usepackage{enumitem}
\usepackage{tabularx}
\usepackage{booktabs}
\usepackage{array}
\usepackage{parskip}
\usepackage{titlesec}
\usepackage{hyperref}
\usepackage{lastpage}
\usepackage{fancyhdr}
\usepackage{calc}
\usepackage{mdframed}

% No parskip/parindent drift in tables
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt}

% ── Hyperref setup ───────────────────────────────────────────
\hypersetup{
  colorlinks=false,
  pdfborder={0 0 0},
  pdftitle={Mutual Non-Disclosure Agreement},
  pdfauthor={LetX},
}

% ── Body colour ──────────────────────────────────────────────
\color{bodytext}

% ── Page style ───────────────────────────────────────────────
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{%
  \color{rulegray}\footnotesize
  Mutual Non-Disclosure Agreement \quad|\quad
  Page \thepage\ of \pageref{LastPage}%
}
\fancyfoot[R]{%
  \color{rulegray}\footnotesize Confidential
}

% ── Section / clause style (numbered, flush left) ────────────
\titleformat{\section}[block]
  {\normalfont\normalsize\bfseries\color{accent}}
  {\thesection.}{0.5em}{}[{%
    \vspace{2pt}\color{rulegray}\hrule height 0.4pt\vspace{4pt}}]

\titlespacing*{\section}{0pt}{14pt}{4pt}

\setcounter{secnumdepth}{2}

% ── Sub-item list style ──────────────────────────────────────
\setlist[enumerate,1]{
  label=(\alph*),
  leftmargin=1.8em,
  itemsep=2pt,
  topsep=4pt,
}
\setlist[itemize,1]{
  leftmargin=1.8em,
  itemsep=2pt,
  topsep=4pt,
  label=\textcolor{accent}{\textbullet},
}

% ── Decorative top rule macro ────────────────────────────────
\newcommand{\ndarule}{%
  {\color{accent}\hrule height 2.5pt}%
  \vspace{1pt}%
  {\color{rulegray}\hrule height 0.6pt}%
}

% ── Signature block macro ────────────────────────────────────
%  \sigblock{Party Label}{Name}{Title}{Date line}
\newcommand{\sigblock}[4]{%
  \begin{minipage}[t]{0.44\textwidth}
    \vspace{0pt}
    {\small\bfseries\color{accent} #1}\\[18pt]
    \hrule\vspace{4pt}
    {\small Signature}\\[10pt]
    \textbf{#2}\\[2pt]
    {\small #3}\\[10pt]
    \hrule\vspace{4pt}
    {\small Date: #4}
  \end{minipage}%
}

% ── Defined term style ───────────────────────────────────────
\newcommand{\term}[1]{\textbf{``#1''}}
