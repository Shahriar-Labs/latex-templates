% --- Font & Encoding Configuration ---
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lato}
\renewcommand{\familydefault}{\sfdefault}

% --- Math Support ---
\usepackage{amsmath, amssymb}

% --- Margins & Layout ---
\usepackage[top=0.65in, bottom=0.65in, left=0.75in, right=0.75in]{geometry}

% --- Colors ---
\usepackage[table]{xcolor}
\definecolor{accent}{HTML}{0F766E}        % Deep Teal Primary Accent
\definecolor{neutraldark}{HTML}{1E293B}   % Slate Charcoal for Text
\definecolor{neutrallight}{HTML}{F8FAFC}  % Soft Warm White for Tables/Cards
\definecolor{muted}{HTML}{64748B}         % Slate Muted for Footers/Subtitles
\definecolor{border}{HTML}{E2E8F0}        % Soft Border Color

% --- Heading Styles (titlesec) ---
\usepackage{titlesec}
\titleformat{\section}{\color{accent}\Large\bfseries}{}{0pt}{}[\color{accent}\vspace{2pt}\hrule height 1.2pt]
\titleformat{\subsection}{\color{accent}\large\bfseries}{}{0pt}{}
\titlespacing*{\section}{0pt}{1.8ex}{1ex}
\titlespacing*{\subsection}{0pt}{1.5ex}{0.8ex}

% --- Table Styling ---
\usepackage{tabularx}
\usepackage{booktabs}
\renewcommand{\arraystretch}{1.2}
\arrayrulecolor{border}

% --- Lists ---
\usepackage{enumitem}
\setlist[itemize]{label={\color{accent}\tiny$\blacksquare$}, leftmargin=*, itemsep=3pt, topsep=1pt, parsep=0pt}

% --- Header & Footer ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[R]{\color{muted}\small Page \thepage\ of 2}
\fancyfoot[L]{\color{muted}\small Project Proposal \textbullet\ E-Commerce Platform Redesign}

% --- Icons & Links ---
\usepackage{fontawesome5}
\usepackage[colorlinks=true, linkcolor=accent, urlcolor=accent]{hyperref}

% --- Spacing Rules (Rule 3) ---
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% Helper for spacing between paragraphs since parskip is 0pt
\newcommand{\parbreak}{\vspace{1ex}\par}
