% ---- Timeline circled-number command ----
\newcommand{\circnum}[1]{%
  \tikz[baseline=(char.base)]{%
    \node[shape=circle, fill=accent, draw=accent2, line width=0.7pt,
          text=white, font=\bfseries\small, inner sep=0pt, minimum size=20pt] (char) {#1};%
  }%
}

  leftmargin=2.1cm,
  labelsep=9pt,
  itemsep=15pt,
  topsep=8pt
}
